\section{Ergodic random fields}

The shift group acts by the deterministic kernels $\hat\theta_i(A \mid \omega) = \1_A(\theta_i\omega)$
of (14.4), so that $\Prob_\Theta$ of (14.1) and $\mathcal{I}(\mu)$ of (14.3) are the
$\Pi$-invariant measures and the $\sigma$-algebra $\mathcal{I}_\Pi(\mu)$ of Corollary (7.4) for
$\Pi = \set{\hat\theta_i}$. Everything below is stated for a measurable action of a group $G$ on
$(\Cfg, \mathcal{F})$, specialised to a subgroup $\Theta$ of the transformation group where said.

\begin{definition}[Invariant fields and invariant events, Georgii (14.1)--(14.4)]
    \label{def:erg-invariant}
    \uses{def:ext-ae-invariant}
    \lean{ProbabilityTheory.Kernel.smulKernel,
      ProbabilityTheory.Kernel.smulInvariantMeasure_iff_forall_invariant,
      ProbabilityTheory.Kernel.measurableSet_aeInvariantSigmaAlgebraFamily_smul}
    \leanok

    For a group $G$ acting measurably on $(\Cfg, \mathcal{F})$ and $g \in G$, let
    $\hat g(A \mid \omega) = \1_A(g\omega)$ be the deterministic kernel of $g$ (Georgii's
    $\hat\theta_i$). A measure is $G$-invariant iff it is invariant under every $\hat g$, and for
    a finite invariant $\mu$ the $\mu$-a.s.\ invariant $\sigma$-algebra of
    $\Pi = \set{\hat g : g \in G}$ is
    \[\mathcal{I}(\mu) = \set{A \in \mathcal{F} : g^{-1}A = A \ \mu\text{-a.e. for all } g \in G}.\]
\end{definition}

\begin{theorem}[Ergodicity is extremality, Georgii (14.5)(a) and (14.6)]
    \label{thm:erg-145a}
    \uses{def:erg-invariant, thm:ext-74}
    \lean{ProbabilityTheory.Kernel.ergodicSMul_iff_mem_extremePoints}
    \leanok

    Let a group $G$, with no countability assumption, act measurably on $(\Cfg, \mathcal{F})$,
    and let $\mu$ be a $G$-invariant probability measure. Then $\mu$ is \emph{ergodic}, i.e.\
    trivial on $\mathcal{I}(\mu)$, iff $\mu$ is an extreme point of the convex set of
    $G$-invariant probability measures.
\end{theorem}

\begin{definition}[The invariant $\sigma$-algebra and the shift group, Georgii (14.1), (14.2)]
    \label{def:erg-invariant-events}
    \uses{def:erg-invariant}
    \lean{MeasurableSpace.smulInvariants,
      MeasureTheory.GibbsMeasure.invariantEvents,
      MeasureTheory.GibbsMeasure.invariantFields,
      MeasureTheory.GibbsMeasure.shiftGroup,
      MeasureTheory.GibbsMeasure.mem_invariantFields_shiftGroup}
    \leanok

    For a group $G$ acting measurably on $(\Cfg, \mathcal{F})$, the \emph{invariant
    $\sigma$-algebra} is $\mathcal{I} = \set{A \in \mathcal{F} : gA = A \text{ for all } g \in G}$,
    and $\Prob_G$ is the set of $G$-invariant probability measures. For an additive group $S$,
    the shift group $\Theta = \set{\theta_j : j \in S}$ is the subgroup of the transformation
    group generated by the shifts, and $\mu \in \Prob_\Theta$ iff $\mu$ is preserved by every
    $\theta_j$.
\end{definition}

\begin{lemma}[Georgii, Remark (14.3)]
    \label{lem:erg-143}
    \uses{def:erg-invariant-events}
    \lean{MeasurableSpace.measurable_invariants_iff,
      MeasureTheory.GibbsMeasure.measurable_invariantEvents_iff,
      ProbabilityTheory.Kernel.exists_measurableSet_invariants_ae_eq,
      ProbabilityTheory.Kernel.exists_measurable_invariants_ae_eq}
    \leanok

    (1) A function into a space with measurable singletons is $\mathcal{I}$-measurable iff it is
    measurable and $G$-invariant. (2) For countable $G$ and an invariant probability measure
    $\mu$, every $\mu$-a.s.\ invariant event $A$ (function $f \geq 0$) agrees $\mu$-a.e.\ with a
    strictly invariant one, namely $\bigcup_g g^{-1}A$ ($\sup_g f \circ g$); so $\mathcal{I}(\mu)$
    is the $\mu$-completion of $\mathcal{I}$.
\end{lemma}

\begin{theorem}[Georgii (14.5) and Definition (14.6)]
    \label{thm:erg-145}
    \uses{def:erg-invariant-events, lem:erg-143, thm:erg-145a, thm:ext-invariant-density}
    \lean{MeasureTheory.GibbsMeasure.ergodicSMul_iff_mem_trivialOn_invariantEvents,
      MeasureTheory.GibbsMeasure.mem_extremePoints_invariantFields_iff_mem_trivialOn,
      MeasureTheory.GibbsMeasure.mem_invariantFields_iff_exists_withDensity,
      MeasureTheory.GibbsMeasure.eq_of_forall_measurableSet_invariantEvents_eq,
      MeasureTheory.GibbsMeasure.exists_measurableSet_invariantEvents_eq_one_eq_zero,
      MeasureTheory.GibbsMeasure.mutuallySingular_of_mem_extremePoints_invariantFields,
      ProbabilityTheory.Kernel.smulInvariantMeasure_iff_exists_withDensity,
      ProbabilityTheory.Kernel.eq_of_forall_measurableSet_invariants_eq,
      ProbabilityTheory.Kernel.exists_measurableSet_invariants_eq_one_eq_zero}
    \leanok

    Let $\Theta$ be a countable subgroup of the transformation group and $\mathcal{I}$ its
    invariant $\sigma$-algebra. A measure $\mu \in \Prob_\Theta$ is \emph{ergodic} if it is
    trivial on $\mathcal{I}$.
    \begin{enumerate}
        \item[(a)] $\mu \in \Prob_\Theta$ is extreme in $\Prob_\Theta$ iff it is ergodic.
        \item[(b)] For $\mu \in \Prob_\Theta$ and a probability measure $\nu \ll \mu$:
              $\nu \in \Prob_\Theta$ iff $\nu = f\mu$ with $f$ $\mathcal{I}$-measurable.
        \item[(c)] Each $\mu \in \Prob_\Theta$ is determined within $\Prob_\Theta$ by its
              restriction to $\mathcal{I}$.
        \item[(d)] Distinct $\mu, \nu \in \mathrm{ex}\,\Prob_\Theta$ are mutually singular on
              $\mathcal{I}$: some $A \in \mathcal{I}$ has $\mu(A) = 1$, $\nu(A) = 0$.
    \end{enumerate}
\end{theorem}
\begin{proof}
    \leanok

    (a) is Theorem \ref{thm:erg-145a}, with Lemma \ref{lem:erg-143}(2) identifying triviality
    on $\mathcal{I}(\mu)$ with triviality on $\mathcal{I}$. (b): for invariant $\nu \ll \mu$ the
    density is a.s.\ invariant by Theorem \ref{thm:ext-invariant-density}, hence has an
    $\mathcal{I}$-measurable version; the converse is a change of variables. (c): apply (b) to
    $\mu$ and $\nu$ relative to $\frac12(\mu + \nu)$; the two densities are
    $\mathcal{I}$-measurable with the same integrals over $\mathcal{I}$, so they agree. (d): for
    extreme $\mu \neq \nu$, (c) gives $A \in \mathcal{I}$ with $\mu(A) \neq \nu(A)$, and
    triviality of both on $\mathcal{I}$ forces the values $1$ and $0$.
\end{proof}

\begin{proposition}[Georgii (14.9)]
    \label{prop:erg-149}
    \uses{def:erg-invariant-events, def:ql-tail}
    \lean{MeasureTheory.GibbsMeasure.exists_measurableSet_tail_measure_symmDiff_eq_zero,
      MeasureTheory.GibbsMeasure.ergodicSMul_of_mem_trivialOn_tail,
      MeasureTheory.GibbsMeasure.shiftGroup_exists_disjoint_sites_preimage,
      MeasureTheory.exists_measurableSet_iInf_measure_symmDiff_eq_zero}
    \leanok

    Let $\Theta$ be a group of transformations moving every finite volume off itself: for each
    $\Lambda \in \Fin$ some $\tau \in \Theta$ has $\tau_*^{-1}\Lambda \cap \Lambda = \emptyset$
    (the shift group of an infinite site group, or any group containing a shift of infinite
    order). For $\mu \in \Prob_\Theta$ and $A \in \mathcal{I}$ there is $B \in \Tail$ with
    $\mu(A \mathbin{\triangle} B) = 0$. In particular a tail-trivial $\mu \in \Prob_\Theta$ is
    ergodic.
\end{proposition}
\begin{proof}
    \leanok

    Choose local events $B_n$ with $\mu(A \mathbin{\triangle} B_n) \le 2^{-n}$, shift each by a
    $\tau_n \in \Theta$ moving its volume off itself, so that $\tau_n B_n$ is measurable for the
    tail of the exhaustion at stage $n$ and, by invariance, still within $2^{-n}$ of $A$; then
    $B = \limsup_n \tau_n B_n$ is a tail event and Borel--Cantelli gives
    $\mu(A \mathbin{\triangle} B) = 0$.
\end{proof}

\begin{theorem}[Georgii (14.15)]
    \label{thm:erg-1415}
    \uses{def:erg-invariantG, thm:erg-145, prop:erg-149, thm:ext-invariant-density}
    \lean{MeasureTheory.GibbsMeasure.isExtreme_invariantG,
      MeasureTheory.GibbsMeasure.extremePoints_invariantG,
      MeasureTheory.GibbsMeasure.mem_extremePoints_invariantG_iff_ergodicSMul,
      MeasureTheory.GibbsMeasure.mem_invariantG_of_absolutelyContinuous,
      MeasureTheory.GibbsMeasure.eq_of_forall_measurableSet_invariantEvents_eq_of_mem_invariantG,
      MeasureTheory.GibbsMeasure.exists_measurableSet_invariantEvents_eq_one_eq_zero_of_mem_extremePoints_invariantG}
    \leanok

    Let $\Theta$ be as in Proposition \ref{prop:erg-149} and $\gamma$ a specification.
    \begin{enumerate}
        \item[(a)] $\mathrm{ex}\,\mathcal{G}_\Theta(\gamma) = \mathcal{G}_\Theta(\gamma) \cap
              \mathrm{ex}\,\Prob_\Theta$: $\mu \in \mathcal{G}_\Theta(\gamma)$ is extreme there
              iff it is ergodic.
        \item[(b)] If $\mu \in \mathcal{G}_\Theta(\gamma)$, $\nu \in \Prob_\Theta$ and
              $\nu \ll \mu$, then $\nu \in \mathcal{G}_\Theta(\gamma)$.
        \item[(c)] $\mathcal{G}_\Theta(\gamma)$ is a face of $\Prob_\Theta$.
    \end{enumerate}
    Parts (c), (d) of Theorem \ref{thm:erg-145} hold with $\mathcal{G}_\Theta(\gamma)$ in place
    of $\Prob_\Theta$.
\end{theorem}
\begin{proof}
    \leanok

    (b): by Theorem \ref{thm:erg-145}(b), $\nu = f\mu$ with $f$ $\mathcal{I}$-measurable; by
    Proposition \ref{prop:erg-149} the events of $\mathcal{I}$ are $\mu$-a.s.\ invariant for
    every kernel $\gamma_\Lambda$, so $f\mu$ is again $\gamma$-invariant by
    Theorem \ref{thm:ext-invariant-density}. (c) is (b) applied to the two summands of a convex
    combination, and (a) is (c): the extreme points of a face are the extreme points of the
    ambient set that lie in it.
\end{proof}

\begin{theorem}[Georgii (14.12)]
    \label{thm:erg-1412}
    \uses{def:erg-invariant-events, prop:erg-149, thm:erg-145}
    \lean{MeasureTheory.GibbsMeasure.tileAverage,
      MeasureTheory.GibbsMeasure.exists_tendsto_extremePoints_invariantFields_shiftGroup,
      MeasureTheory.GibbsMeasure.closure_setOf_mem_extremePoints_invariantFields_shiftGroup}
    \leanok

    Let the site set be a countable infinite abelian group with a sequence of tilings
    $(C_n, H_n)$ --- $C_n$ a set of representatives of the subgroup $H_n$ --- whose boundary
    fractions vanish, e.g.\ the cubes of $\mathbb{Z}^d$. In the topology of local convergence,
    $\mathrm{ex}\,\Prob_\Theta$ is dense in $\Prob_\Theta$: every $\mu \in \Prob_\Theta$ is the
    limit of a sequence of ergodic shift-invariant fields.
\end{theorem}
\begin{proof}
    \leanok

    Let $\mu_n$ be the product over the tiles of the marginals of $\mu$ on $C_n$, and
    $v_n = |C_n|^{-1}\sum_{j \in C_n} \theta_j(\mu_n)$. For a local event in the volume $\Delta$,
    only the $j$ for which $\Delta - j$ crosses a tile boundary contribute to
    $v_n(A) - \mu(A)$, a fraction tending to zero, so $v_n \to \mu$. Each $v_n$ is shift
    invariant since $\theta_h(\mu_n) = \mu_n$ for $h \in H_n$, and ergodic: for $A \in \mathcal{I}$,
    Proposition \ref{prop:erg-149} gives a tail event $B$ with $v_n(A \mathbin{\triangle} B) = 0$,
    and $\mu_n$ is trivial on the tail as a product over independent tiles, so $v_n(A) = \mu_n(B)
    \in \set{0,1}$.
\end{proof}

\begin{theorem}[Georgii (14.A3), (14.A5): the mean ergodic theorems]
    \label{thm:erg-14A3}
    \uses{def:erg-invariant-events}
    \lean{MeasureTheory.tendsto_eLpNorm_inv_card_smul_sum_sub_condExp_two,
      MeasureTheory.tendsto_eLpNorm_inv_card_smul_sum_sub_condExp_one,
      MeasureTheory.tendsto_integral_norm_inv_card_smul_sum_sub_condExp,
      MeasureTheory.Lp.tendsto_inv_card_smul_sum_compMeasurePreservingₗᵢ_smul_condExpL2}
    \leanok

    Let a countable group $G$ act on $(\Cfg, \mu)$ by measure-preserving maps, and let $(F_k)$
    be a net of nonempty finite sets with $|gF_k \mathbin{\triangle} F_k| / |F_k| \to 0$ for every
    $g$. For $R_k f = |F_k|^{-1}\sum_{i \in F_k} f \circ (i\,\cdot)$: if $f \in L^2(\mu)$ then
    $R_k f \to \mu(f \mid \mathcal{I})$ in $L^2$, and if $\mu$ is finite and $f \in L^1(\mu)$
    then in $L^1$.
\end{theorem}
\begin{proof}
    \leanok

    In $L^2$ the Koopman operators are isometries; $L^2 = \mathcal{K} \oplus \overline{\mathrm{span}}\{Sx - x\}$
    with $\mathcal{K}$ their common fixed space, the averages fix $\mathcal{K}$, contract a
    coboundary by $|gF_k \mathbin{\triangle} F_k|/|F_k|$, and are equicontinuous, so they converge
    to the projection onto $\mathcal{K}$; countability of $G$ identifies $\mathcal{K}$ with
    $L^2(\mathcal{I})$ and the projection with $\mu(\cdot \mid \mathcal{I})$. The $L^1$ statement
    follows by density of simple functions and the contraction $\|R_k f\|_1 \le \|f\|_1$.
\end{proof}

\begin{lemma}[Georgii (14.19)]
    \label{lem:erg-1419}
    \lean{MeasureTheory.tendsto_ae_condExp_of_antitone_of_dominated,
      MeasureTheory.condExp_iInf_le_liminf_condExp_of_antitone,
      MeasureTheory.tendsto_ae_condExp_of_monotone}
    \leanok

    Let $\mu$ be finite, $(\mathcal{A}_n)$ a decreasing sequence of sub-$\sigma$-algebras,
    $(f_n)$ with $|f_n| \le g$ for an integrable $g$ and $f_n \to f_\infty$ a.e. Then
    $\mu(f_n \mid \mathcal{A}_n) \to \mu(f_\infty \mid \bigcap_n \mathcal{A}_n)$ a.e.
\end{lemma}
\begin{proof}
    \leanok

    For $g_N = \inf_{n \ge N} f_n$ the backward martingale theorem gives
    $\mu(g_N \mid \bigcap \mathcal{A}_n) = \lim_n \mu(g_N \mid \mathcal{A}_n) \le
    \liminf_n \mu(f_n \mid \mathcal{A}_n)$; $g_N \uparrow f_\infty$ and monotone convergence of
    conditional expectations give the lower bound, and $-f_n$ the upper one.
\end{proof}

\begin{lemma}[Georgii (14.A6), the maximal inequality]
    \label{lem:erg-14A6}
    \lean{MeasureTheory.ergodicMaximalFunction,
      MeasureTheory.mul_measure_lt_ergodicMaximalFunction_le,
      MeasureTheory.mul_measure_lt_ergodicMaximalFunction_le_of_monotone,
      MeasureTheory.measure_exists_lt_abs_div_card_le_cube,
      Finset.exists_subset_pairwiseDisjoint_subset_biUnion}
    \leanok

    Let an additive commutative group act on $(\Cfg, \mu)$ by measure-preserving maps and let
    $(F_n)$ be an increasing sequence of finite sets with $|F_n - F_n + F_n| \le C|F_n|$. For
    measurable $f \ge 0$, $R_n f = |F_n|^{-1}\sum_{i \in F_n} f \circ \theta_i$ and $c > 0$,
    \[ c\,\mu\bigl(\sup_n R_n f > c\bigr) \le C \int f\,\mathrm{d}\mu . \]
    For increasing cubes in $\mathbb{Z}^d$ one may take $C = 3^d$, which is Georgii's
    $\mu(\sup_n |R_n f| > c) \le 3^d\mu(|f|)/c$ for real integrable $f$.
\end{lemma}
\begin{proof}
    \leanok

    For a finite window $\Delta$ and a configuration $\omega$, each site $i \in \Delta$ at which
    the truncated maximal average exceeds $c$ has a witness $F_{n(i)} + i$; the greedy selection
    of Mathlib's Vitali covering lemma, largest rank first, yields pairwise disjoint witnesses
    whose enlargements $F_n - F_n + F_n + i$ cover the exceptional set, so
    $c\,|\Delta_\omega| \le C \sum_{j \in \Delta + F_N} f(\theta_j\omega)$. Integrate, use
    invariance, let $|\Delta| \to \infty$ along a F\o lner sequence of windows and $N \to \infty$.
\end{proof}

\begin{theorem}[Georgii (14.A8), the individual ergodic theorem]
    \label{thm:erg-14A8}
    \uses{thm:erg-14A3, lem:erg-14A6}
    \lean{MeasureTheory.ae_tendsto_inv_card_smul_sum_vadd_condExp,
      MeasureTheory.ae_forall_tendsto_inv_card_smul_sum_vadd_condExp,
      MeasureTheory.ae_tendsto_inv_card_smul_sum_vadd_condExp_cube,
      MeasureTheory.tendsto_card_vadd_cube_symmDiff_div_card}
    \leanok

    Let a countable additive group act on a finite measure space by measure-preserving maps and
    let $(F_n)$ be an increasing F\o lner sequence of nonempty finite sets with
    $|F_n - F_n + F_n| \le C|F_n|$, $C < \infty$. For integrable $f$,
    $R_n f \to \mu(f \mid \mathcal{I})$ almost everywhere; for countably many $f$ simultaneously.
    Increasing cubes in $\mathbb{Z}^d$ are such a sequence with $C = 3^d$.
\end{theorem}
\begin{proof}
    \leanok

    For a simple $g$ with $\|f - g\|_1 < \varepsilon^2/(C+1)$ and $N$ with
    $\|R_N g - \mu(g\mid\mathcal{I})\|_1$ small (Theorem \ref{thm:erg-14A3}),
    $R_n g - R_n(R_N g) \to 0$ everywhere as an average of coboundaries, and the maximal
    inequality (Lemma \ref{lem:erg-14A6}) applied to $f - g$ and to $R_N g - \mu(g\mid\mathcal{I})$,
    with Markov's inequality on $\mu(f - g \mid \mathcal{I})$, bound
    $\mu(\limsup_n |R_n f - \mu(f\mid\mathcal{I})| \ge 4\varepsilon)$ by $3\varepsilon$.
\end{proof}
