\section{Ergodic random fields}

The shift group acts by the deterministic kernels $\hat\theta_i(A \mid \omega) = \1_A(\theta_i\omega)$
of (14.4), so that $\Prob_\Theta$ of (14.1) and $\mathcal{I}(\mu)$ of (14.3) are the
$\Pi$-invariant measures and the $\sigma$-algebra $\mathcal{I}_\Pi(\mu)$ of Corollary (7.4) for
$\Pi = \set{\hat\theta_i}$. Everything below is stated for a measurable action of a group $G$ on
$(\Cfg, \mathcal{F})$, specialised to a subgroup $\Theta$ of the transformation group where said.

\begin{definition}[Invariant fields and invariant events, Georgii (14.1)--(14.4)]
    \label{def:erg-invariant}
    \uses{def:ext-ae-invariant}
    \lean{ProbabilityTheory.Kernel.smulKernel,
      ProbabilityTheory.Kernel.smulInvariantMeasure_iff_forall_invariant,
      ProbabilityTheory.Kernel.measurableSet_aeInvariantSigmaAlgebraFamily_smul}
    \leanok

    For a group $G$ acting measurably on $(\Cfg, \mathcal{F})$ and $g \in G$, let
    $\hat g(A \mid \omega) = \1_A(g\omega)$ be the deterministic kernel of $g$ (Georgii's
    $\hat\theta_i$). A measure is $G$-invariant iff it is invariant under every $\hat g$, and for
    a finite invariant $\mu$ the $\mu$-a.s.\ invariant $\sigma$-algebra of
    $\Pi = \set{\hat g : g \in G}$ is
    \[\mathcal{I}(\mu) = \set{A \in \mathcal{F} : g^{-1}A = A \ \mu\text{-a.e. for all } g \in G}.\]
\end{definition}

\begin{theorem}[Ergodicity is extremality, Georgii (14.5)(a) and (14.6)]
    \label{thm:erg-145a}
    \uses{def:erg-invariant, thm:ext-74}
    \lean{ProbabilityTheory.Kernel.ergodicSMul_iff_mem_extremePoints}
    \leanok

    Let a group $G$, with no countability assumption, act measurably on $(\Cfg, \mathcal{F})$,
    and let $\mu$ be a $G$-invariant probability measure. Then $\mu$ is \emph{ergodic}, i.e.\
    trivial on $\mathcal{I}(\mu)$, iff $\mu$ is an extreme point of the convex set of
    $G$-invariant probability measures.
\end{theorem}

\begin{definition}[The invariant $\sigma$-algebra and the shift group, Georgii (14.1), (14.2)]
    \label{def:erg-invariant-events}
    \uses{def:erg-invariant}
    \lean{MeasurableSpace.invariants,
      MeasureTheory.GibbsMeasure.invariantEvents,
      MeasureTheory.GibbsMeasure.invariantFields,
      MeasureTheory.GibbsMeasure.shiftGroup,
      MeasureTheory.GibbsMeasure.mem_invariantFields_shiftGroup}
    \leanok

    For a group $G$ acting measurably on $(\Cfg, \mathcal{F})$, the \emph{invariant
    $\sigma$-algebra} is $\mathcal{I} = \set{A \in \mathcal{F} : gA = A \text{ for all } g \in G}$,
    and $\Prob_G$ is the set of $G$-invariant probability measures. For an additive group $S$,
    the shift group $\Theta = \set{\theta_j : j \in S}$ is the subgroup of the transformation
    group generated by the shifts, and $\mu \in \Prob_\Theta$ iff $\mu$ is preserved by every
    $\theta_j$.
\end{definition}

\begin{lemma}[Georgii, Remark (14.3)]
    \label{lem:erg-143}
    \uses{def:erg-invariant-events}
    \lean{MeasurableSpace.measurable_invariants_iff,
      MeasureTheory.GibbsMeasure.measurable_invariantEvents_iff,
      ProbabilityTheory.Kernel.exists_measurableSet_invariants_ae_eq,
      ProbabilityTheory.Kernel.exists_measurable_invariants_ae_eq}
    \leanok

    (1) A function into a space with measurable singletons is $\mathcal{I}$-measurable iff it is
    measurable and $G$-invariant. (2) For countable $G$ and an invariant probability measure
    $\mu$, every $\mu$-a.s.\ invariant event $A$ (function $f \geq 0$) agrees $\mu$-a.e.\ with a
    strictly invariant one, namely $\bigcup_g g^{-1}A$ ($\sup_g f \circ g$); so $\mathcal{I}(\mu)$
    is the $\mu$-completion of $\mathcal{I}$.
\end{lemma}

\begin{theorem}[Georgii (14.5) and Definition (14.6)]
    \label{thm:erg-145}
    \uses{def:erg-invariant-events, lem:erg-143, thm:erg-145a, thm:ext-invariant-density}
    \lean{MeasureTheory.GibbsMeasure.ergodicSMul_iff_mem_trivialOn_invariantEvents,
      MeasureTheory.GibbsMeasure.mem_extremePoints_invariantFields_iff_mem_trivialOn,
      MeasureTheory.GibbsMeasure.mem_invariantFields_iff_exists_withDensity,
      MeasureTheory.GibbsMeasure.eq_of_forall_measurableSet_invariantEvents_eq,
      MeasureTheory.GibbsMeasure.exists_measurableSet_invariantEvents_eq_one_eq_zero,
      MeasureTheory.GibbsMeasure.mutuallySingular_of_mem_extremePoints_invariantFields,
      ProbabilityTheory.Kernel.smulInvariantMeasure_iff_exists_withDensity,
      ProbabilityTheory.Kernel.eq_of_forall_measurableSet_invariants_eq,
      ProbabilityTheory.Kernel.exists_measurableSet_invariants_eq_one_eq_zero}
    \leanok

    Let $\Theta$ be a countable subgroup of the transformation group and $\mathcal{I}$ its
    invariant $\sigma$-algebra. A measure $\mu \in \Prob_\Theta$ is \emph{ergodic} if it is
    trivial on $\mathcal{I}$.
    \begin{enumerate}
        \item[(a)] $\mu \in \Prob_\Theta$ is extreme in $\Prob_\Theta$ iff it is ergodic.
        \item[(b)] For $\mu \in \Prob_\Theta$ and a probability measure $\nu \ll \mu$:
              $\nu \in \Prob_\Theta$ iff $\nu = f\mu$ with $f$ $\mathcal{I}$-measurable.
        \item[(c)] Each $\mu \in \Prob_\Theta$ is determined within $\Prob_\Theta$ by its
              restriction to $\mathcal{I}$.
        \item[(d)] Distinct $\mu, \nu \in \mathrm{ex}\,\Prob_\Theta$ are mutually singular on
              $\mathcal{I}$: some $A \in \mathcal{I}$ has $\mu(A) = 1$, $\nu(A) = 0$.
    \end{enumerate}
\end{theorem}
\begin{proof}
    \leanok

    (a) is Theorem \ref{thm:erg-145a}, with Lemma \ref{lem:erg-143}(2) identifying triviality
    on $\mathcal{I}(\mu)$ with triviality on $\mathcal{I}$. (b): for invariant $\nu \ll \mu$ the
    density is a.s.\ invariant by Theorem \ref{thm:ext-invariant-density}, hence has an
    $\mathcal{I}$-measurable version; the converse is a change of variables. (c): apply (b) to
    $\mu$ and $\nu$ relative to $\frac12(\mu + \nu)$; the two densities are
    $\mathcal{I}$-measurable with the same integrals over $\mathcal{I}$, so they agree. (d): for
    extreme $\mu \neq \nu$, (c) gives $A \in \mathcal{I}$ with $\mu(A) \neq \nu(A)$, and
    triviality of both on $\mathcal{I}$ forces the values $1$ and $0$.
\end{proof}
