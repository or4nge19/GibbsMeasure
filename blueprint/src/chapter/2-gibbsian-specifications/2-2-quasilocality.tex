\section{Quasilocality}

Throughout, $S$ is the parameter set, $(E, \mathcal{E})$ a measurable space, $\Cfg = E^S$ the
configuration space, $\Fin$ the finite subsets of $S$, and $\mathcal{F}_\Lambda$ the cylinder
$\sigma$-algebra generated by the coordinates in $\Lambda$. The section isolates the condition on a
specification expressing that no finite-volume conditional probability reads off the state of the
system at infinity.

\subsection*{Macroscopic observables: the tail \texorpdfstring{$\sigma$}{sigma}-algebra}

\begin{definition}[Tail $\sigma$-algebra, Georgii (2.19)]
    \label{def:ql-tail}
    \uses{def:cylinder-event}
    \lean{MeasureTheory.GibbsMeasure.tailSigmaAlgebra}
    \leanok

    The \textbf{tail $\sigma$-algebra} (or $\sigma$-algebra at infinity) is
    \[\Tail \;=\; \bigcap_{\Lambda \in \Fin} \mathcal{F}_{S \setminus \Lambda}.\]
    A bounded $\Tail$-measurable function is a \textbf{macroscopic observable}.
\end{definition}

\begin{lemma}[Tail events are boundary events]
    \label{lem:ql-tail-boundary}
    \uses{def:ql-tail}
    \lean{MeasureTheory.GibbsMeasure.measurableSet_cylinderEvents_compl_of_measurableSet_tail}
    \leanok

    For every $\Lambda \in \Fin$, $\Tail \subseteq \mathcal{F}_{S\setminus\Lambda}$.
\end{lemma}
\begin{proof}
    \leanok

    $\Tail$ is the infimum of the $\mathcal{F}_{S\setminus\Lambda}$, hence below each of them.
\end{proof}

\begin{lemma}[Specifications preserve macroscopic observables, Georgii after (2.19)]
    \label{lem:ql-spec-preserves-tail}
    \uses{def:ql-tail, def:markov-spec}
    \lean{Specification.ae_eq_of_tail}
    \leanok

    Let $\xi : \Cfg \to X$ be $\Tail$-measurable, with $X$ a measurable space whose singletons are
    measurable. Then for every specification $\gamma$, every $\Lambda \in \Fin$ and every
    $\omega \in \Cfg$, the measure $\gamma_\Lambda(\cdot \mid \omega)$ is carried by
    $\{\sigma : \xi(\sigma) = \xi(\omega)\}$.
\end{lemma}
\begin{proof}
    \leanok

    By Lemma \ref{lem:ql-tail-boundary} the event $B = \{\xi = \xi(\omega)\}$ lies in
    $\mathcal{F}_{S\setminus\Lambda}$, and $\omega \in B$. Properness of $\gamma_\Lambda$ gives
    $\gamma_\Lambda(B \mid \omega) = 1_B(\omega)\,\gamma_\Lambda(\Cfg \mid \omega) = 1$, so the
    complement of $B$ is null.
\end{proof}

\begin{remark}
    \label{rmk:ql-tail-family}
    \uses{lem:ql-spec-preserves-tail}
    The same conclusion holds for a family $(\gamma^x)_{x \in X}$ of specifications indexed by the
    codomain of $\xi$: for every $x$, $\Lambda$ and $\omega$, the measure
    $\gamma^x_\Lambda(\cdot \mid \omega)$ is carried by $\{\xi = \xi(\omega)\}$. This is the form
    used in Proposition \ref{prop:ql-glued}, and the single-specification statement is the case of a
    constant family.
\end{remark}

\subsection*{Microscopic observables: local and quasilocal functions}

\begin{definition}[Local functions, Georgii (2.20)(a)]
    \label{def:ql-local}
    \uses{def:cylinder-event}
    \lean{MeasureTheory.GibbsMeasure.localFunctionsOn, MeasureTheory.GibbsMeasure.localFunctions, MeasureTheory.GibbsMeasure.mem_localFunctions}
    \leanok

    For $\Lambda \in \Fin$, $\mathcal{L}_\Lambda$ is the space of bounded
    $\mathcal{F}_\Lambda$-measurable functions $\Cfg \to \R$, and
    \[\mathcal{L} \;=\; \bigcup_{\Lambda \in \Fin} \mathcal{L}_\Lambda\]
    is the space of bounded \textbf{local} (cylinder) functions. Both are subalgebras of the Banach
    algebra $\ell^\infty(\Cfg; \R)$; since the family $(\mathcal{L}_\Lambda)_\Lambda$ is directed,
    the subalgebra it generates is the displayed union.
\end{definition}

\begin{definition}[Quasilocal functions, Georgii (2.20)(b)]
    \label{def:ql-quasilocal}
    \uses{def:ql-local}
    \lean{MeasureTheory.GibbsMeasure.quasilocalFunctions, MeasureTheory.GibbsMeasure.measurable_of_mem_quasilocalFunctions}
    \leanok

    A bounded function $f : \Cfg \to \R$ is \textbf{quasilocal} if it is a uniform limit of local
    functions. The space $\overline{\mathcal{L}}$ of bounded quasilocal functions is the closure of
    $\mathcal{L}$ in the sup-norm. Every quasilocal function is measurable.
\end{definition}

\begin{definition}[Oscillation, Georgii (2.22)]
    \label{def:ql-osc}
    \lean{MeasureTheory.GibbsMeasure.oscOutside, MeasureTheory.GibbsMeasure.IsQuasilocalFun,
      MeasureTheory.GibbsMeasure.isQuasilocalFun_iff_tendsto_oscOutside}
    \leanok

    For $f : \Cfg \to \R$ and $s \subseteq S$ put
    \[\osc_s(f) \;=\; \sup\set{|f(\zeta) - f(\eta)| \;:\; \zeta, \eta \in \Cfg,\;
      \zeta_s = \eta_s} \;\in\; [0,\infty],\]
    the oscillation of $f$ under changes outside $s$; it is antitone in $s$, and $\osc_\Lambda(f)$
    denotes its value at $s = \Lambda \in \Fin$.

    $f$ is \textbf{quasilocal in the sense of (2.22)} if for every $\varepsilon > 0$ there is a
    finite $\Delta$ with $|f(\zeta) - f(\eta)| \leq \varepsilon$ whenever $\zeta_\Delta =
    \eta_\Delta$; this requires neither boundedness nor measurability of $f$, and is equivalent to
    $\osc_\Lambda(f) \to 0$ along $\Fin$.
\end{definition}

\begin{theorem}[Oscillation criterion, Georgii Remark (2.21)(1)]
    \label{thm:ql-osc-criterion}
    \uses{def:ql-quasilocal, def:ql-osc}
    \lean{MeasureTheory.GibbsMeasure.mem_quasilocalFunctions_iff, MeasureTheory.GibbsMeasure.tendsto_oscOutside_of_mem_quasilocalFunctions, MeasureTheory.GibbsMeasure.mem_quasilocalFunctions_of_tendsto_oscOutside,
      MeasureTheory.GibbsMeasure.mem_quasilocalFunctions_iff_isQuasilocalFun}
    \leanok

    A bounded measurable $f : \Cfg \to \R$ is quasilocal if and only if
    \[\lim_{\Lambda \in \Fin} \osc_\Lambda(f) = 0,\]
    the limit being taken along the directed set $\Fin$; equivalently, if and only if it is
    quasilocal in the $\varepsilon$--$\Delta$ sense of (2.22).
\end{theorem}
\begin{proof}
    \leanok

    If $\|f - g\| < \varepsilon$ with $g \in \mathcal{L}_{\Lambda_0}$, then for $\Lambda \supseteq
    \Lambda_0$ and $\zeta_\Lambda = \eta_\Lambda$ one has $g(\zeta) = g(\eta)$, whence
    $|f(\zeta) - f(\eta)| \le 2\varepsilon$; so $\osc_\Lambda(f) \to 0$. Conversely, fix
    $\eta \in \Cfg$ and set $g_\Lambda(\omega) = f(\omega_\Lambda \eta_{S\setminus\Lambda})$. Each
    $g_\Lambda$ is local and $\|f - g_\Lambda\| \le \osc_\Lambda(f)$, so $f$ is a uniform limit of
    local functions.
\end{proof}

\begin{remark}
    \label{rmk:ql-tail-not-quasilocal}
    \uses{def:ql-tail, thm:ql-osc-criterion}
    A non-constant $\Tail$-measurable $f$ satisfies $\osc_\Lambda(f) = \sup f - \inf f > 0$ for
    every $\Lambda \in \Fin$, so by Theorem \ref{thm:ql-osc-criterion} it is not quasilocal. In
    Examples \ref{ex:ql-2-27} and \ref{ex:ql-4-16} the criterion is contradicted directly, with an
    explicit pair of configurations.
\end{remark}

\begin{proposition}[Uniformly continuous observables are quasilocal, Georgii (2.21)(2)]
    \label{prop:ql-unif-cont}
    \uses{def:ql-quasilocal, thm:ql-osc-criterion}
    \lean{MeasureTheory.GibbsMeasure.mem_quasilocalFunctions_of_uniformContinuous}
    \leanok

    Let $E$ carry a uniform structure and $\Cfg = E^S$ the product uniformity --- no metric on
    $\Cfg$, no countability of $S$. Every bounded, measurable, uniformly continuous
    $f : \Cfg \to \R$ is quasilocal.
\end{proposition}
\begin{proof}
    \leanok

    Given $\varepsilon > 0$ choose an entourage on which $f$ oscillates by less than $\varepsilon$;
    by definition of the product uniformity it contains all pairs agreeing on some finite
    $\Lambda_0$. Hence $\osc_\Lambda(f) \le \varepsilon$ for $\Lambda \supseteq \Lambda_0$, and
    Theorem \ref{thm:ql-osc-criterion} applies.
\end{proof}

\begin{proposition}[Discrete compact spin space, Georgii (2.21)(3)]
    \label{prop:ql-continuous-iff}
    \uses{def:ql-quasilocal, prop:ql-unif-cont}
    \lean{MeasureTheory.GibbsMeasure.mem_quasilocalFunctions_iff_continuous, MeasureTheory.GibbsMeasure.continuous_of_mem_localFunctions}
    \leanok

    If $E$ is discrete and compact and $\Cfg$ carries the product topology, then a bounded
    measurable $f : \Cfg \to \R$ is quasilocal if and only if it is continuous. Compactness is used
    only for the implication ``continuous $\Rightarrow$ quasilocal''.
\end{proposition}
\begin{proof}
    \leanok

    A local function factors through finitely many discrete coordinates, hence is continuous, and
    the continuous bounded functions form a closed subset for the sup-norm; this gives one
    direction. For the other, a continuous function on the compact space $\Cfg$ is uniformly
    continuous, so Proposition \ref{prop:ql-unif-cont} applies.
\end{proof}

\subsection*{Quasilocal specifications}

\begin{definition}[Action of a specification on observables]
    \label{def:ql-action}
    \uses{def:spec}
    \lean{Specification.action, Specification.action_apply, Specification.measurable_action, Specification.action_mem_localFunctionsOn_compl}
    \leanok

    For a specification $\gamma$, $\Lambda \in \Fin$ and a bounded $f$, the observable
    $\gamma_\Lambda f$ is
    \[(\gamma_\Lambda f)(\eta) \;=\; \int f(\sigma)\, \gamma_\Lambda(\mathrm{d}\sigma \mid \eta).\]
    It is again bounded, by $\|f\|$, and for measurable $f$ it is
    $\mathcal{F}_{S\setminus\Lambda}$-measurable.
\end{definition}

\begin{lemma}[The action is a contraction]
    \label{lem:ql-action-contraction}
    \uses{def:ql-action}
    \lean{Specification.dist_action_le}
    \leanok

    For bounded measurable $f, g$ and every $\Lambda \in \Fin$,
    $\|\gamma_\Lambda f - \gamma_\Lambda g\| \le \|f - g\|$.
\end{lemma}
\begin{proof}
    \leanok

    Pointwise, $|(\gamma_\Lambda(f-g))(\eta)| \le \int |f - g| \,\mathrm{d}\gamma_\Lambda(\cdot\mid\eta)
    \le \|f-g\|$, because $\gamma_\Lambda(\cdot \mid \eta)$ is a probability measure.
\end{proof}

\begin{definition}[Quasilocal specification, Georgii (2.23)]
    \label{def:ql-quasilocal-spec}
    \uses{def:spec, def:ql-quasilocal, def:ql-action}
    \lean{Specification.IsQuasilocal}
    \leanok

    A specification $\gamma$ is \textbf{quasilocal} if $f \in \overline{\mathcal{L}}$ implies
    $\gamma_\Lambda f \in \overline{\mathcal{L}}$ for every $\Lambda \in \Fin$.
\end{definition}

\begin{lemma}[It suffices to test local observables, Georgii after (2.23)]
    \label{lem:ql-check-local}
    \uses{def:ql-quasilocal-spec, lem:ql-action-contraction, def:ql-local}
    \lean{Specification.isQuasilocal_iff_forall_mem_localFunctions}
    \leanok

    $\gamma$ is quasilocal if and only if $\gamma_\Lambda f \in \overline{\mathcal{L}}$ for every
    $\Lambda \in \Fin$ and every \emph{local} $f$.
\end{lemma}
\begin{proof}
    \leanok

    $\overline{\mathcal{L}}$ is closed, and by Lemma \ref{lem:ql-action-contraction} the map
    $f \mapsto \gamma_\Lambda f$ is $1$-Lipschitz; so it carries the closure of $\mathcal{L}$ into
    the closure of $\gamma_\Lambda(\mathcal{L}) \subseteq \overline{\mathcal{L}}$.
\end{proof}

\begin{proposition}[Independent specifications are quasilocal, Georgii after (2.23)]
    \label{prop:ql-isssd}
    \uses{def:ql-quasilocal-spec, def:isssd-spec, lem:ql-check-local}
    \lean{Specification.IsResampling.isQuasilocal, Specification.isQuasilocal_isssd, Specification.IsResampling.action_mem_localFunctionsOn}
    \leanok

    Let $\gamma$ be a \textbf{resampling} specification, i.e.\ one with
    $\gamma_\Lambda(\cdot \mid \eta) = \mu_\Lambda \circ \juxtBy{\eta}^{-1}$ for some measure
    $\mu_\Lambda$ on $E^\Lambda$. Then $\gamma_\Lambda$ maps $\mathcal{L}_\Delta$ into
    $\mathcal{L}_{\Delta \setminus \Lambda}$, and $\gamma$ is quasilocal. In particular every
    independent specification $\ISSSD(\nu)$, and every inhomogeneous independent specification, is
    quasilocal.
\end{proposition}
\begin{proof}
    \leanok

    If $f$ is $\mathcal{F}_\Delta$-measurable then, writing the kernel as the image of a fixed
    measure under $\zeta \mapsto \juxtBy{\eta}(\zeta)$, the integral
    $\int f(\juxtBy{\eta}\zeta)\, \mu_\Lambda(\mathrm{d}\zeta)$ depends on $\eta$ only through the
    coordinates in $\Delta \setminus \Lambda$. Hence $\gamma_\Lambda f \in
    \mathcal{L}_{\Delta\setminus\Lambda} \subseteq \overline{\mathcal{L}}$, and
    Lemma \ref{lem:ql-check-local} concludes.
\end{proof}

\subsection*{Georgii (2.24): quasilocality of \texorpdfstring{$\lambda$}{lambda}-specifications}

\begin{proposition}[Georgii (2.24)(a)]
    \label{prop:ql-modification}
    \uses{def:ql-quasilocal-spec, def:modif}
    \lean{Specification.IsQuasilocal.modification}
    \leanok

    Let $\gamma$ be a quasilocal specification and $\rho$ a modifier of $\gamma$ such that every
    $\rho_\Lambda$ is finite and, as a real-valued function, is a bounded quasilocal observable,
    i.e.\ lies in $\overline{\mathcal{L}}$. Then $\rho\gamma$ is quasilocal.
\end{proposition}
\begin{proof}
    \leanok

    For local $f$ one has $(\rho\gamma)_\Lambda f = \gamma_\Lambda(\rho_\Lambda f)$ by definition of
    the modification as a density kernel. Since $\overline{\mathcal{L}}$ is an algebra,
    $\rho_\Lambda f \in \overline{\mathcal{L}}$, and quasilocality of $\gamma$ gives
    $\gamma_\Lambda(\rho_\Lambda f) \in \overline{\mathcal{L}}$. Conclude with
    Lemma \ref{lem:ql-check-local}.
\end{proof}

\begin{definition}[Boltzmann factor of a bounded observable]
    \label{def:ql-boltzmann}
    \uses{def:ql-quasilocal}
    \lean{Specification.boltzmann, Specification.boltzmann_mem_quasilocalFunctions, MeasureTheory.GibbsMeasure.exp_mem_quasilocalFunctions}
    \leanok

    For a bounded observable $H$ put $h = e^{-H}$. If $H \in \overline{\mathcal{L}}$ then
    $h \in \overline{\mathcal{L}}$ and $h \ge e^{-\|H\|} > 0$. More generally the exponential of a
    quasilocal observable is quasilocal, $\overline{\mathcal{L}}$ being a closed subalgebra.
\end{definition}

\begin{theorem}[Georgii (2.24)(b)]
    \label{thm:ql-gibbsian-quasilocal}
    \uses{prop:ql-modification, def:ql-boltzmann, prop:ql-isssd}
    \lean{Specification.isQuasilocal_lambdaSpecification,
      Potential.isQuasilocal_gibbsSpecificationOfSigmaFiniteAdmissible,
      Specification.IsResampling.isQuasilocal_modification_relNorm,
      Specification.isQuasilocal_modification_premodifierNorm}
    \leanok

    Let $\lambda \in \Meas(E, \mathcal{E})$ be $\sigma$-finite and non-zero. The
    $\lambda$-specification of a $\lambda$-admissible premodifier of Boltzmann form
    $\rho_\Lambda = e^{-H_\Lambda}$ with Hamiltonians $H_\Lambda$ quasilocal in the sense of
    (2.22) --- not necessarily bounded --- is quasilocal; in particular, for a summable potential
    $\Phi$ that is $\lambda$-admissible and has quasilocal scaled Hamiltonians
    $\beta H_\Lambda^\Phi$, the Gibbsian specification $\gamma^\Phi$ is quasilocal.

    If instead the reference specification $\gamma$ is a resampling specification, e.g.\
    $\ISSSD(\nu)$, and $H_\Lambda \in \overline{\mathcal{L}}$, then the normalised modification of
    $\gamma$ by $e^{-H_\Lambda}$ is quasilocal.
\end{theorem}
\begin{proof}
    \leanok

    For the bounded case write $Z_\Lambda = \gamma_\Lambda h_\Lambda$ with
    $h_\Lambda = e^{-H_\Lambda}$. Then $Z_\Lambda \ge e^{-\|H_\Lambda\|} > 0$, so $Z_\Lambda$ is
    bounded away from $0$ and, being the $\gamma_\Lambda$-action of a quasilocal observable, is
    itself quasilocal by Proposition \ref{prop:ql-isssd}. A closed subalgebra of bounded functions
    is stable under inversion of elements bounded away from $0$, so
    $\rho_\Lambda = h_\Lambda / Z_\Lambda$ is a bounded quasilocal density and
    Proposition \ref{prop:ql-modification} applies.

    For unbounded quasilocal $H_\Lambda$ over a $\sigma$-finite $\lambda$, approximate
    $e^{-H_\Lambda}$ by $e^{-u}$ with $u$ local, at the cost
    $\lambda_\Lambda|\rho_\Lambda - e^{-u}| \le (e^\varepsilon - 1)\lambda_\Lambda\rho_\Lambda$,
    and divide by the partition function.
\end{proof}

\begin{remark}[Georgii (2.24)(c)]
    \label{rmk:ql-2-24-c}
    \uses{def:ql-quasilocal-spec}
    Conversely, if $E$ is finite, $\lambda$ is counting measure and $\gamma$ is quasilocal, then the
    unique $\lambda$-modification $\rho$ with $\gamma = \rho\lambda$ satisfies
    $\rho_\Lambda \in \overline{\mathcal{L}}$ for every $\Lambda$, by
    \[\rho_\Lambda \;=\; \sum_{\zeta \in E^\Lambda} 1_{\{\sigma_\Lambda = \zeta\}}\,
    \gamma_\Lambda 1_{\{\sigma_\Lambda = \zeta\}}.\]
\end{remark}

\begin{example}[Georgii (2.25): absolutely summable potentials]
    \label{ex:ql-2-25}
    \uses{thm:ql-gibbsian-quasilocal, def:ql-quasilocal}
    \lean{Potential.hamiltonianLp_mem_quasilocalFunctions, Potential.boltzmannLp_mem_quasilocalFunctions, Potential.isQuasilocal_gibbsSpecificationOfAbsolutelySummable}
    \leanok

    Let $S$ be countable and $\Phi \in \Bspace$ absolutely summable. Every Hamiltonian
    $H_\Lambda^\Phi$ is a bounded quasilocal observable --- it is the sum, convergent in sup-norm,
    of the local interaction terms $\Phi_A$ with $A \cap \Lambda \ne \emptyset$ --- hence so is
    $h_\Lambda^\Phi = e^{-\beta H_\Lambda^\Phi}$; and therefore the Gibbsian specification
    $\gamma^\Phi$ relative to a single-spin probability measure $\nu$ is quasilocal, for every
    inverse temperature $\beta$.
\end{example}


\subsection*{Non-quasilocal specifications}

\begin{proposition}[Gluing a family of specifications, Georgii Remark (2.26)]
    \label{prop:ql-glued}
    \uses{def:spec, def:gibbs-meas, lem:ql-spec-preserves-tail}
    \lean{Specification.IsMeasurableFamily, Specification.glued, Specification.isGibbsMeasure_glued}
    \leanok

    Let $(X, \MCX)$ be a measurable space with measurable singletons, $\xi : \Cfg \to X$ be
    $\Tail$-measurable, and $(\gamma^x)_{x \in X}$ a family of specifications such that
    $(\omega, x) \mapsto \gamma^x_\Lambda(A \mid \omega)$ is jointly measurable. Then
    \[\gamma_\Lambda(A \mid \omega) \;=\; \gamma^{\xi(\omega)}_\Lambda(A \mid \omega)\]
    is again a specification, and every random field $\mu$ with $\mu(\xi = x) = 1$ and
    $\mu \in \Gibbs{\gamma^x}$ satisfies $\mu \in \Gibbs{\gamma}$.
\end{proposition}
\begin{proof}
    \leanok

    Properness and the Markov property are pointwise in $\omega$. For consistency: by
    Lemma \ref{lem:ql-spec-preserves-tail}, $\gamma^{\xi(\omega)}_\Delta(\cdot \mid \omega)$ is
    carried by $\{\xi = \xi(\omega)\}$, on which the inner kernel $\gamma^{\xi(\sigma)}_\Lambda$ is
    the fixed kernel $\gamma^{\xi(\omega)}_\Lambda$; consistency of $\gamma^{\xi(\omega)}$ then
    applies. The Gibbs statement is the same argument with $\mu$-almost sure constancy of $\xi$.
\end{proof}

\begin{example}[Georgii (2.27): Bernoulli fields on $\{0,1\}^{\mathbb{N}}$]
    \label{ex:ql-2-27}
    \uses{prop:ql-glued, def:ql-quasilocal-spec, thm:ql-osc-criterion}
    \lean{MeasureTheory.GibbsMeasure.Exchangeable.gammaEx, MeasureTheory.GibbsMeasure.Exchangeable.isGibbsMeasure_bernoulliField, MeasureTheory.GibbsMeasure.Exchangeable.not_countable_gibbsMeasures, MeasureTheory.GibbsMeasure.Exchangeable.not_isQuasilocal_gammaEx}
    \leanok

    Take $E = \{0,1\}$, $S = \mathbb{N}$, $X = [0,1]$, and let $\mu^x$ be the Bernoulli field with
    parameter $x$, a Gibbs measure for the independent specification $\gamma^x = \ISSSD(\lambda^x)$
    with $\lambda^x = x\delta_1 + (1-x)\delta_0$. The tail function
    \[\xi \;=\; \liminf_{n\to\infty} \frac{1}{n}\sum_{i < n} \sigma_i\]
    separates their supports, by the strong law of large numbers. Gluing along $\xi$ as in
    Proposition \ref{prop:ql-glued} produces a specification $\gamma$ with
    $\mu^x \in \Gibbs{\gamma}$ for every $x \in [0,1]$; in particular $\Gibbs{\gamma}$ is
    uncountable, and $\gamma$ is not quasilocal.

    Dobrushin's interdependence matrix of this $\gamma$ vanishes identically, so
    $c(\gamma) = 0 < 1$: quasilocality cannot be dropped from Dobrushin's uniqueness condition, and
    non-quasilocality of $\gamma$ follows by contradiction with the uniqueness theorem.
\end{example}

\begin{example}[Georgii (4.16): a non-quasilocal specification with no Gibbs measure]
    \label{ex:ql-4-16}
    \uses{def:ql-quasilocal-spec, def:gibbs-meas, thm:ql-osc-criterion}
    \lean{MeasureTheory.GibbsMeasure.Example416.specification, MeasureTheory.GibbsMeasure.Example416.GP_eq_empty, MeasureTheory.GibbsMeasure.Example416.not_isQuasilocal_specification, MeasureTheory.GibbsMeasure.Example416.mem_GP_of_finite}
    \leanok

    On $E = \{0,1\}$ with $S$ countably infinite, let $\gamma_\Lambda(\cdot \mid \omega)$ be the
    uniform distribution on the ``spikes'' $\omega^a$, $a \in \Lambda$, when $\omega$ vanishes off
    $\Lambda$, and the Dirac mass at $0_\Lambda\omega$ otherwise. This is a proper, consistent
    specification with $\Gibbs{\gamma} = \emptyset$, and it is not quasilocal; it is Georgii's
    witness that quasilocality cannot be dropped from the existence theorems (4.17) and (4.22). For
    \emph{finite} $S$ the same kernels do admit a Gibbs measure, so infiniteness of $S$ is needed.

    Non-quasilocality: $f = 1_{\{\sigma_a = 1\}}$ is local, but
    $(\gamma_{\{a\}} f)(\eta) = 1_{\{\eta = 0 \text{ off } \{a\}\}}$, whose oscillation outside any
    finite $\Delta$ is $1$ --- put a spike at a site outside $\Delta \cup \{a\}$ --- contradicting
    Theorem \ref{thm:ql-osc-criterion}. Emptiness of $\Gibbs{\gamma}$: a Gibbs measure would give
    mass $0$ to every configuration with two or more $1$s, to the zero configuration, and (letting
    $|\Lambda| \to \infty$ in $\gamma_\Lambda(\{\omega^a\} \mid \cdot) \le |\Lambda|^{-1}$) to every
    single spike; these exhaust $\Cfg$.
\end{example}
