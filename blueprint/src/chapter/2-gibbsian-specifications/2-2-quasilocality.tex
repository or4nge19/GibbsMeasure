\section{Quasilocality}

Throughout, $S$ is the parameter set, $(E, \mathcal{E})$ a measurable space, $\Cfg = E^S$,
$\Fin$ the finite subsets of $S$, and $\mathcal{F}_\Lambda$ the cylinder $\sigma$-algebra
generated by the coordinates in $\Lambda$.

\subsection*{Macroscopic observables: the tail \texorpdfstring{$\sigma$}{sigma}-algebra}

\begin{definition}[Tail $\sigma$-algebra, Georgii (2.19)]
    \label{def:ql-tail}
    \uses{def:cylinder-event}
    \lean{MeasureTheory.GibbsMeasure.tailSigmaAlgebra}
    \leanok

    The \textbf{tail $\sigma$-algebra} is
    \[\Tail \;=\; \bigcap_{\Lambda \in \Fin} \mathcal{F}_{S \setminus \Lambda}.\]
    A bounded $\Tail$-measurable function is a \textbf{macroscopic observable}.
\end{definition}

\begin{lemma}[Tail events are boundary events]
    \label{lem:ql-tail-boundary}
    \uses{def:ql-tail}
    \lean{MeasureTheory.GibbsMeasure.measurableSet_cylinderEvents_compl_of_measurableSet_tail}
    \leanok

    For every $\Lambda \in \Fin$, $\Tail \subseteq \mathcal{F}_{S\setminus\Lambda}$.
\end{lemma}
\begin{proof}
    \leanok

    $\Tail$ is the infimum of the $\mathcal{F}_{S\setminus\Lambda}$.
\end{proof}

\begin{lemma}[Specifications preserve macroscopic observables, Georgii after (2.19)]
    \label{lem:ql-spec-preserves-tail}
    \uses{def:ql-tail, def:markov-spec}
    \lean{Specification.ae_eq_of_tail}
    \leanok

    Let $X$ be a measurable space with measurable singletons, $\xi : \Cfg \to X$
    $\Tail$-measurable, and $(\gamma^x)_{x \in X}$ a family of specifications. For every $x \in X$,
    $\Lambda \in \Fin$ and $\omega \in \Cfg$, the measure $\gamma^x_\Lambda(\cdot \mid \omega)$ is
    carried by $\{\sigma : \xi(\sigma) = \xi(\omega)\}$.
\end{lemma}
\begin{proof}
    \leanok

    By Lemma \ref{lem:ql-tail-boundary} the event $B = \{\xi = \xi(\omega)\}$ lies in
    $\mathcal{F}_{S\setminus\Lambda}$, and $\omega \in B$. Properness gives
    $\gamma^x_\Lambda(B \mid \omega) = 1_B(\omega)\,\gamma^x_\Lambda(\Cfg \mid \omega) = 1$.
\end{proof}

\begin{remark}
    \label{rmk:ql-tail-family}
    \uses{lem:ql-spec-preserves-tail}
    A single specification is the case of a constant family in
    Lemma \ref{lem:ql-spec-preserves-tail}; the family form is the one used in
    Proposition \ref{prop:ql-glued}.
\end{remark}

\subsection*{Microscopic observables: local and quasilocal functions}

\begin{definition}[Local functions, Georgii (2.20)(a)]
    \label{def:ql-local}
    \uses{def:cylinder-event}
    \lean{MeasureTheory.GibbsMeasure.localFunctionsOn, MeasureTheory.GibbsMeasure.localFunctions, MeasureTheory.GibbsMeasure.mem_localFunctions}
    \leanok

    For $\Lambda \in \Fin$, $\mathcal{L}_\Lambda$ is the space of bounded
    $\mathcal{F}_\Lambda$-measurable functions $\Cfg \to \R$, and
    \[\mathcal{L} \;=\; \bigcup_{\Lambda \in \Fin} \mathcal{L}_\Lambda\]
    is the space of bounded \textbf{local} functions. Both are subalgebras of
    $\ell^\infty(\Cfg; \R)$; the family $(\mathcal{L}_\Lambda)_\Lambda$ being directed, the
    subalgebra it generates is the union.
\end{definition}

\begin{definition}[Quasilocal functions, Georgii (2.20)(b)]
    \label{def:ql-quasilocal}
    \uses{def:ql-local}
    \lean{MeasureTheory.GibbsMeasure.quasilocalFunctions, MeasureTheory.GibbsMeasure.measurable_of_mem_quasilocalFunctions}
    \leanok

    A bounded $f : \Cfg \to \R$ is \textbf{quasilocal} if it is a uniform limit of local
    functions; $\overline{\mathcal{L}}$ denotes the sup-norm closure of $\mathcal{L}$. Every
    quasilocal function is measurable.
\end{definition}

\begin{definition}[Oscillation, Georgii (2.22)]
    \label{def:ql-osc}
    \lean{MeasureTheory.GibbsMeasure.oscOutside, MeasureTheory.GibbsMeasure.IsQuasilocalFun,
      MeasureTheory.GibbsMeasure.isQuasilocalFun_iff_tendsto_oscOutside}
    \leanok

    For $f : \Cfg \to \R$ and $s \subseteq S$ put
    \[\osc_s(f) \;=\; \sup\set{|f(\zeta) - f(\eta)| \;:\; \zeta, \eta \in \Cfg,\;
      \zeta_s = \eta_s} \;\in\; [0,\infty];\]
    it is antitone in $s$, and $\osc_\Lambda(f)$ denotes its value at $s = \Lambda \in \Fin$.

    An arbitrary $f : \Cfg \to \R$ is \textbf{quasilocal in the sense of (2.22)} if for every
    $\varepsilon > 0$ there is $\Delta \in \Fin$ with $|f(\zeta) - f(\eta)| \leq \varepsilon$
    whenever $\zeta_\Delta = \eta_\Delta$; equivalently, $\osc_\Lambda(f) \to 0$ along $\Fin$.
\end{definition}

\begin{theorem}[Oscillation criterion, Georgii Remark (2.21)(1)]
    \label{thm:ql-osc-criterion}
    \uses{def:ql-quasilocal, def:ql-osc}
    \lean{MeasureTheory.GibbsMeasure.mem_quasilocalFunctions_iff, MeasureTheory.GibbsMeasure.tendsto_oscOutside_of_mem_quasilocalFunctions, MeasureTheory.GibbsMeasure.mem_quasilocalFunctions_of_tendsto_oscOutside,
      MeasureTheory.GibbsMeasure.mem_quasilocalFunctions_iff_isQuasilocalFun}
    \leanok

    A bounded measurable $f : \Cfg \to \R$ is quasilocal if and only if
    \[\lim_{\Lambda \in \Fin} \osc_\Lambda(f) = 0\]
    along the directed set $\Fin$, i.e.\ if and only if it is quasilocal in the sense of (2.22).
\end{theorem}
\begin{proof}
    \leanok

    If $\|f - g\| < \varepsilon$ with $g \in \mathcal{L}_{\Lambda_0}$, then for $\Lambda \supseteq
    \Lambda_0$ and $\zeta_\Lambda = \eta_\Lambda$ one has $g(\zeta) = g(\eta)$, whence
    $|f(\zeta) - f(\eta)| \le 2\varepsilon$. Conversely, fix $\eta \in \Cfg$ and set
    $g_\Lambda(\omega) = f(\omega_\Lambda \eta_{S\setminus\Lambda})$; each $g_\Lambda$ is local
    and $\|f - g_\Lambda\| \le \osc_\Lambda(f)$.
\end{proof}

\begin{remark}
    \label{rmk:ql-tail-not-quasilocal}
    \uses{def:ql-tail, thm:ql-osc-criterion}
    A non-constant $\Tail$-measurable $f$ has $\osc_\Lambda(f) = \sup f - \inf f > 0$ for every
    $\Lambda \in \Fin$, hence is not quasilocal by Theorem \ref{thm:ql-osc-criterion}. See
    Examples \ref{ex:ql-2-27} and \ref{ex:ql-4-16}.
\end{remark}

\begin{proposition}[Uniformly continuous observables are quasilocal, Georgii (2.21)(2)]
    \label{prop:ql-unif-cont}
    \uses{def:ql-quasilocal, thm:ql-osc-criterion}
    \lean{MeasureTheory.GibbsMeasure.mem_quasilocalFunctions_of_uniformContinuous}
    \leanok

    Let $E$ be a uniform space and $\Cfg = E^S$ carry the product uniformity, with no metric on
    $\Cfg$ and no countability assumption on $S$. Every bounded, measurable, uniformly continuous
    $f : \Cfg \to \R$ is quasilocal.
\end{proposition}
\begin{proof}
    \leanok

    Given $\varepsilon > 0$, an entourage on which $f$ oscillates by less than $\varepsilon$
    contains all pairs agreeing on some finite $\Lambda_0$. Hence $\osc_\Lambda(f) \le \varepsilon$
    for $\Lambda \supseteq \Lambda_0$, and Theorem \ref{thm:ql-osc-criterion} applies.
\end{proof}

\begin{proposition}[Discrete compact spin space, Georgii (2.21)(3)]
    \label{prop:ql-continuous-iff}
    \uses{def:ql-quasilocal, prop:ql-unif-cont}
    \lean{MeasureTheory.GibbsMeasure.mem_quasilocalFunctions_iff_continuous, MeasureTheory.GibbsMeasure.continuous_of_mem_localFunctions}
    \leanok

    Let $E$ be discrete and $\Cfg$ carry the product topology. Every local function is
    continuous, and if $E$ is moreover compact, a bounded measurable $f : \Cfg \to \R$ is
    quasilocal if and only if it is continuous.
\end{proposition}
\begin{proof}
    \leanok

    A local function factors through finitely many discrete coordinates. The bounded continuous
    functions are sup-norm closed, so quasilocal functions are continuous. Conversely, a
    continuous function on the compact space $\Cfg$ is uniformly continuous, and
    Proposition \ref{prop:ql-unif-cont} applies.
\end{proof}

\subsection*{Quasilocal specifications}

\begin{definition}[Action of a specification on observables]
    \label{def:ql-action}
    \uses{def:spec}
    \lean{Specification.action, Specification.action_apply, Specification.measurable_action, Specification.action_mem_localFunctionsOn_compl}
    \leanok

    For a specification $\gamma$, $\Lambda \in \Fin$ and a bounded $f$,
    \[(\gamma_\Lambda f)(\eta) \;=\; \int f(\sigma)\, \gamma_\Lambda(\mathrm{d}\sigma \mid \eta).\]
    $\gamma_\Lambda f$ is bounded by $\|f\|$, and for measurable $f$ it is
    $\mathcal{F}_{S\setminus\Lambda}$-measurable.
\end{definition}

\begin{lemma}[The action is a contraction]
    \label{lem:ql-action-contraction}
    \uses{def:ql-action}
    \lean{Specification.dist_action_le}
    \leanok

    For bounded measurable $f, g$ and every $\Lambda \in \Fin$,
    $\|\gamma_\Lambda f - \gamma_\Lambda g\| \le \|f - g\|$.
\end{lemma}
\begin{proof}
    \leanok

    $|(\gamma_\Lambda(f-g))(\eta)| \le \int |f - g| \,\mathrm{d}\gamma_\Lambda(\cdot\mid\eta)
    \le \|f-g\|$, as $\gamma_\Lambda(\cdot \mid \eta)$ is a probability measure.
\end{proof}

\begin{definition}[Quasilocal specification, Georgii (2.23)]
    \label{def:ql-quasilocal-spec}
    \uses{def:spec, def:ql-quasilocal, def:ql-action}
    \lean{Specification.IsQuasilocal}
    \leanok

    A specification $\gamma$ is \textbf{quasilocal} if $\gamma_\Lambda f \in
    \overline{\mathcal{L}}$ for every $f \in \overline{\mathcal{L}}$ and every $\Lambda \in \Fin$.
\end{definition}

\begin{lemma}[It suffices to test local observables, Georgii after (2.23)]
    \label{lem:ql-check-local}
    \uses{def:ql-quasilocal-spec, lem:ql-action-contraction, def:ql-local}
    \lean{Specification.isQuasilocal_iff_forall_mem_localFunctions}
    \leanok

    $\gamma$ is quasilocal if and only if $\gamma_\Lambda f \in \overline{\mathcal{L}}$ for every
    $\Lambda \in \Fin$ and every $f \in \mathcal{L}$.
\end{lemma}
\begin{proof}
    \leanok

    By Lemma \ref{lem:ql-action-contraction}, $f \mapsto \gamma_\Lambda f$ is $1$-Lipschitz, so it
    carries $\overline{\mathcal{L}}$ into the closure of
    $\gamma_\Lambda(\mathcal{L}) \subseteq \overline{\mathcal{L}}$.
\end{proof}

\begin{proposition}[Independent specifications are quasilocal, Georgii after (2.23)]
    \label{prop:ql-isssd}
    \uses{def:ql-quasilocal-spec, def:isssd-spec, lem:ql-check-local}
    \lean{Specification.IsResampling.isQuasilocal, Specification.isQuasilocal_isssd, Specification.IsResampling.action_mem_localFunctionsOn}
    \leanok

    Let $\gamma$ be a \textbf{resampling} specification, i.e.\
    $\gamma_\Lambda(\cdot \mid \eta) = \mu_\Lambda \circ \juxtBy{\eta}^{-1}$ for some measure
    $\mu_\Lambda$ on $E^\Lambda$. Then $\gamma_\Lambda$ maps $\mathcal{L}_\Delta$ into
    $\mathcal{L}_{\Delta \setminus \Lambda}$, and $\gamma$ is quasilocal. In particular every
    independent specification $\ISSSD(\nu)$, and every inhomogeneous independent specification, is
    quasilocal.
\end{proposition}
\begin{proof}
    \leanok

    For $\mathcal{F}_\Delta$-measurable $f$,
    $\int f(\juxtBy{\eta}\zeta)\, \mu_\Lambda(\mathrm{d}\zeta)$ depends on $\eta$ only through
    $\eta_{\Delta \setminus \Lambda}$. Hence $\gamma_\Lambda f \in
    \mathcal{L}_{\Delta\setminus\Lambda} \subseteq \overline{\mathcal{L}}$, and
    Lemma \ref{lem:ql-check-local} concludes.
\end{proof}

\subsection*{Georgii (2.24): quasilocality of \texorpdfstring{$\lambda$}{lambda}-specifications}

\begin{proposition}[Georgii (2.24)(a)]
    \label{prop:ql-modification}
    \uses{def:ql-quasilocal-spec, def:modif}
    \lean{Specification.IsQuasilocal.modification}
    \leanok

    Let $\gamma$ be a quasilocal specification and $\rho$ a modifier of $\gamma$ such that every
    $\rho_\Lambda$ is finite and lies in $\overline{\mathcal{L}}$. Then $\rho\gamma$ is quasilocal.
\end{proposition}
\begin{proof}
    \leanok

    For local $f$, $(\rho\gamma)_\Lambda f = \gamma_\Lambda(\rho_\Lambda f)$. Since
    $\overline{\mathcal{L}}$ is an algebra, $\rho_\Lambda f \in \overline{\mathcal{L}}$, so
    $\gamma_\Lambda(\rho_\Lambda f) \in \overline{\mathcal{L}}$; conclude with
    Lemma \ref{lem:ql-check-local}.
\end{proof}

\begin{definition}[Boltzmann factor of a bounded observable]
    \label{def:ql-boltzmann}
    \uses{def:ql-quasilocal}
    \lean{Specification.boltzmann, Specification.boltzmann_mem_quasilocalFunctions, MeasureTheory.GibbsMeasure.exp_mem_quasilocalFunctions}
    \leanok

    For a bounded observable $H$ put $h = e^{-H}$. Then $h \ge e^{-\|H\|} > 0$, and
    $h \in \overline{\mathcal{L}}$ whenever $H \in \overline{\mathcal{L}}$; more generally
    $\overline{\mathcal{L}}$, a closed subalgebra, is stable under $\exp$.
\end{definition}

\begin{theorem}[Georgii (2.24)(b)]
    \label{thm:ql-gibbsian-quasilocal}
    \uses{prop:ql-modification, def:ql-boltzmann, prop:ql-isssd}
    \lean{Specification.isQuasilocal_lambdaSpecification,
      Potential.isQuasilocal_gibbsSpecificationOfSigmaFiniteAdmissible,
      Specification.IsResampling.isQuasilocal_modification_relNorm,
      Specification.isQuasilocal_modification_premodifierNorm}
    \leanok

    Let $\lambda \in \Meas(E, \mathcal{E})$ be $\sigma$-finite and non-zero. The
    $\lambda$-specification of a $\lambda$-admissible premodifier $\rho_\Lambda = e^{-H_\Lambda}$
    whose Hamiltonians $H_\Lambda$, not necessarily bounded, are quasilocal in the sense of
    (2.22) is quasilocal. In particular, for a summable potential $\Phi$ that is
    $\lambda$-admissible and whose scaled Hamiltonians $\beta H_\Lambda^\Phi$ are quasilocal in
    the sense of (2.22), the Gibbsian specification $\gamma^\Phi$ is quasilocal.

    If the reference specification $\gamma$ is a resampling specification, e.g.\ $\ISSSD(\nu)$
    with $\nu$ a probability measure, and $H_\Lambda \in \overline{\mathcal{L}}$ for every
    $\Lambda$, then the normalised modification of $\gamma$ by $e^{-H_\Lambda}$ is quasilocal.
\end{theorem}
\begin{proof}
    \leanok

    For bounded $H_\Lambda$ write $Z_\Lambda = \gamma_\Lambda h_\Lambda$ with
    $h_\Lambda = e^{-H_\Lambda}$. Then $Z_\Lambda \ge e^{-\|H_\Lambda\|} > 0$ and $Z_\Lambda$ is
    quasilocal by Proposition \ref{prop:ql-isssd}. A closed subalgebra of bounded functions is
    stable under inversion of elements bounded away from $0$, so
    $\rho_\Lambda = h_\Lambda / Z_\Lambda$ is a bounded quasilocal density and
    Proposition \ref{prop:ql-modification} applies.

    For unbounded $H_\Lambda$ over $\sigma$-finite $\lambda$, approximate $e^{-H_\Lambda}$ by
    $e^{-u}$ with $u$ local, at the cost
    $\lambda_\Lambda|\rho_\Lambda - e^{-u}| \le (e^\varepsilon - 1)\lambda_\Lambda\rho_\Lambda$,
    and divide by the partition function.
\end{proof}

\begin{remark}[Georgii (2.24)(c)]
    \label{rmk:ql-2-24-c}
    \uses{def:ql-quasilocal-spec}
    Conversely, if $E$ is finite, $\lambda$ is counting measure and $\gamma$ is quasilocal, then the
    unique $\lambda$-modification $\rho$ with $\gamma = \rho\lambda$ satisfies
    $\rho_\Lambda \in \overline{\mathcal{L}}$ for every $\Lambda$, by
    \[\rho_\Lambda \;=\; \sum_{\zeta \in E^\Lambda} 1_{\{\sigma_\Lambda = \zeta\}}\,
    \gamma_\Lambda 1_{\{\sigma_\Lambda = \zeta\}}.\]
\end{remark}

\begin{example}[Georgii (2.25): absolutely summable potentials]
    \label{ex:ql-2-25}
    \uses{thm:ql-gibbsian-quasilocal, def:ql-quasilocal}
    \lean{Potential.hamiltonianLp_mem_quasilocalFunctions, Potential.boltzmannLp_mem_quasilocalFunctions, Potential.isQuasilocal_gibbsSpecificationOfAbsolutelySummable}
    \leanok

    Let $S$ be countable and $\Phi \in \Bspace$. Every Hamiltonian $H_\Lambda^\Phi$ is a bounded
    quasilocal observable, being the sup-norm convergent sum of the local terms $\Phi_A$,
    $A \cap \Lambda \ne \emptyset$; hence so is $h_\Lambda^\Phi = e^{-\beta H_\Lambda^\Phi}$, and
    the Gibbsian specification $\gamma^\Phi$ relative to a single-spin probability measure $\nu$ is
    quasilocal for every $\beta \in \R$.
\end{example}


\subsection*{Non-quasilocal specifications}

\begin{proposition}[Gluing a family of specifications, Georgii Remark (2.26)]
    \label{prop:ql-glued}
    \uses{def:spec, def:gibbs-meas, lem:ql-spec-preserves-tail}
    \lean{Specification.IsMeasurableFamily, Specification.glued, Specification.isGibbsMeasure_glued}
    \leanok

    Let $(X, \MCX)$ be a measurable space with measurable singletons, $\xi : \Cfg \to X$
    $\Tail$-measurable, and $(\gamma^x)_{x \in X}$ a family of specifications such that
    $(x, \omega) \mapsto \gamma^x_\Lambda(A \mid \omega)$ is
    $\MCX \otimes \mathcal{F}_{S \setminus \Lambda}$-measurable for every $\Lambda \in \Fin$ and
    measurable $A$. Then
    \[\gamma_\Lambda(A \mid \omega) \;=\; \gamma^{\xi(\omega)}_\Lambda(A \mid \omega)\]
    is a specification, and every probability measure $\mu$ with $\mu(\xi = x) = 1$ and
    $\mu \in \Gibbs{\gamma^x}$ lies in $\Gibbs{\gamma}$.
\end{proposition}
\begin{proof}
    \leanok

    Properness and the Markov property are pointwise in $\omega$. For consistency,
    Lemma \ref{lem:ql-spec-preserves-tail} shows that $\gamma^{\xi(\omega)}_\Delta(\cdot \mid \omega)$
    is carried by $\{\xi = \xi(\omega)\}$, on which the inner kernel $\gamma^{\xi(\sigma)}_\Lambda$
    equals $\gamma^{\xi(\omega)}_\Lambda$; consistency of $\gamma^{\xi(\omega)}$ applies. The
    Gibbs statement is the same argument with $\mu$-almost sure constancy of $\xi$.
\end{proof}

\begin{example}[Georgii (2.27): Bernoulli fields on $\{0,1\}^{\mathbb{N}}$]
    \label{ex:ql-2-27}
    \uses{prop:ql-glued, def:ql-quasilocal-spec, thm:ql-osc-criterion}
    \lean{MeasureTheory.GibbsMeasure.Exchangeable.gammaEx, MeasureTheory.GibbsMeasure.Exchangeable.isGibbsMeasure_bernoulliField, MeasureTheory.GibbsMeasure.Exchangeable.not_countable_gibbsMeasures, MeasureTheory.GibbsMeasure.Exchangeable.not_isQuasilocal_gammaEx}
    \leanok

    Take $E = \{0,1\}$, $S = \mathbb{N}$, $X = [0,1]$, and let $\mu^x$ be the Bernoulli field with
    parameter $x$, a Gibbs measure for $\gamma^x = \ISSSD(\lambda^x)$,
    $\lambda^x = x\delta_1 + (1-x)\delta_0$. The tail function
    \[\xi \;=\; \liminf_{n\to\infty} \frac{1}{n}\sum_{i < n} \sigma_i\]
    satisfies $\mu^x(\xi = x) = 1$ by the strong law of large numbers. Gluing along $\xi$ as in
    Proposition \ref{prop:ql-glued} gives a specification $\gamma$ with $\mu^x \in \Gibbs{\gamma}$
    for every $x \in [0,1]$; so $\Gibbs{\gamma}$ is uncountable, and $\gamma$ is not quasilocal.

    Dobrushin's interdependence matrix of $\gamma$ vanishes, so $c(\gamma) = 0 < 1$;
    non-quasilocality follows from Theorem \ref{thm:dob-uniqueness}, whose quasilocality
    hypothesis therefore cannot be dropped.
\end{example}

\begin{example}[Georgii (4.16): a non-quasilocal specification with no Gibbs measure]
    \label{ex:ql-4-16}
    \uses{def:ql-quasilocal-spec, def:gibbs-meas, thm:ql-osc-criterion}
    \lean{MeasureTheory.GibbsMeasure.Example416.specification, MeasureTheory.GibbsMeasure.Example416.GP_eq_empty, MeasureTheory.GibbsMeasure.Example416.not_isQuasilocal_specification, MeasureTheory.GibbsMeasure.Example416.mem_GP_of_finite}
    \leanok

    Let $E = \{0,1\}$ and $S$ be countably infinite. Let $\gamma_\Lambda(\cdot \mid \omega)$ be the
    uniform distribution on the spikes $\omega^a$, $a \in \Lambda$, when $\omega$ vanishes off
    $\Lambda$, and the Dirac mass at $0_\Lambda\omega$ otherwise. Then $\gamma$ is a
    specification, $\Gibbs{\gamma} = \emptyset$, and $\gamma$ is not quasilocal; hence
    quasilocality cannot be dropped from Theorems \ref{thm:exg-4-17} and \ref{thm:exg-4-22}. For
    non-empty \emph{finite} $S$ the same kernels admit a Gibbs measure, the uniform distribution
    on all spikes.

    Non-quasilocality: $f = 1_{\{\sigma_a = 1\}}$ is local, but
    $(\gamma_{\{a\}} f)(\eta) = 1_{\{\eta = 0 \text{ off } \{a\}\}}$ has oscillation $1$ outside
    every finite $\Delta$ (place a spike outside $\Delta \cup \{a\}$), contradicting
    Theorem \ref{thm:ql-osc-criterion}. Emptiness of $\Gibbs{\gamma}$: a Gibbs measure gives mass
    $0$ to every configuration with two or more $1$s, to the zero configuration, and, letting
    $|\Lambda| \to \infty$ in $\gamma_\Lambda(\{\omega^a\} \mid \cdot) \le |\Lambda|^{-1}$, to
    every single spike; these exhaust $\Cfg$.
\end{example}
