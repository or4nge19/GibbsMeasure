\section{Quasilocality}

Throughout, $S$ is the parameter set --- countable, by Georgii's standing convention, though most
of the statements below need no cardinality assumption --- $(E, \mathcal{E})$ a measurable space,
$\Cfg = E^S$ the configuration space, $\Fin$ the finite subsets of $S$, and
$\mathcal{F}_\Lambda$ the cylinder $\sigma$-algebra generated by the coordinates in $\Lambda$.

Georgii's aim in \S2.2 is to isolate the condition on a specification $\gamma$ expressing that
\emph{a microscopic part of the system has only a microscopic horizon}: no finite-volume
conditional probability may read off the state of the system at infinity.

\subsection*{Macroscopic observables: the tail \texorpdfstring{$\sigma$}{sigma}-algebra}

\begin{definition}[Tail $\sigma$-algebra, Georgii (2.19)]
    \label{def:ql-tail}
    \uses{def:cylinder-event}
    \lean{MeasureTheory.GibbsMeasure.tailSigmaAlgebra}
    \leanok

    The \textbf{tail $\sigma$-algebra} (or $\sigma$-algebra at infinity) is
    \[\Tail \;=\; \bigcap_{\Lambda \in \Fin} \mathcal{F}_{S \setminus \Lambda}.\]
    A bounded $\Tail$-measurable function is called a \emph{macroscopic observable}: its value is
    unaffected by the behaviour of any finite set of spins.
\end{definition}

\begin{lemma}[Tail events are boundary events]
    \label{lem:ql-tail-boundary}
    \uses{def:ql-tail}
    \lean{MeasureTheory.GibbsMeasure.measurableSet_cylinderEvents_compl_of_measurableSet_tail}
    \leanok

    For every $\Lambda \in \Fin$, $\Tail \subseteq \mathcal{F}_{S\setminus\Lambda}$; that is, every
    tail event is measurable for the boundary $\sigma$-algebra of every finite volume.
\end{lemma}
\begin{proof}
    \leanok

    $\Tail$ is by definition the infimum of the $\mathcal{F}_{S\setminus\Lambda}$, so it is below
    each of them.
\end{proof}

\begin{lemma}[Specifications preserve macroscopic observables, Georgii after (2.19)]
    \label{lem:ql-spec-preserves-tail}
    \uses{def:ql-tail, def:proper-spec}
    \lean{Specification.ae_eq_of_tail}
    \leanok

    Let $\xi : \Cfg \to X$ be $\Tail$-measurable, with $X$ a measurable space whose singletons are
    measurable. Then for every specification $\gamma$, every $\Lambda \in \Fin$ and every
    $\omega \in \Cfg$, the measure $\gamma_\Lambda(\cdot \mid \omega)$ is carried by
    $\{\sigma : \xi(\sigma) = \xi(\omega)\}$.

    This is the precise content of Georgii's assertion that $\gamma_\Lambda f = f$ for bounded
    $\Tail$-measurable $f$, i.e.\ that every specification preserves the macroscopic observables.
    The passage from the displayed almost-sure statement to the identity $\gamma_\Lambda f = f$ is
    not itself a separate declaration in the library.
\end{lemma}
\begin{proof}
    \leanok

    By Lemma \ref{lem:ql-tail-boundary} the event $B = \{\xi = \xi(\omega)\}$ lies in
    $\mathcal{F}_{S\setminus\Lambda}$, and $\omega \in B$. Properness of $\gamma_\Lambda$ gives
    $\gamma_\Lambda(B \mid \omega) = 1_B(\omega)\,\gamma_\Lambda(\Cfg \mid \omega) = 1$, so the
    complement of $B$ is null.
\end{proof}

\begin{remark}
    \label{rmk:ql-tail-family}
    The Lean statement quantifies over a family $(\gamma^x)_{x \in X}$ of specifications indexed by
    the codomain of $\xi$ (this is the form needed for Proposition \ref{prop:ql-glued} below); the
    statement above for a single specification is the case of a constant family. The hypothesis
    that singletons of $X$ are measurable is not a strengthening of Georgii's, whose condition
    (2.26)(i) already speaks of the events $\{\xi = x\}$.
\end{remark}

\subsection*{Microscopic observables: local and quasilocal functions}

\begin{definition}[Local functions, Georgii (2.20)(a)]
    \label{def:ql-local}
    \uses{def:cylinder-event}
    \lean{MeasureTheory.GibbsMeasure.localFunctionsOn, MeasureTheory.GibbsMeasure.localFunctions, MeasureTheory.GibbsMeasure.mem_localFunctions}
    \leanok

    For $\Lambda \in \Fin$, $\mathcal{L}_\Lambda$ is the space of bounded
    $\mathcal{F}_\Lambda$-measurable functions $\Cfg \to \R$, and
    \[\mathcal{L} \;=\; \bigcup_{\Lambda \in \Fin} \mathcal{L}_\Lambda\]
    is the space of bounded \textbf{local} (cylinder) functions. In Lean both are subalgebras of the
    Banach algebra $\ell^\infty(\Cfg; \R)$ of bounded functions, and the union is realised as the
    supremum of the directed family $(\mathcal{L}_\Lambda)_\Lambda$; that this supremum is the plain
    union is the lemma \texttt{mem\_localFunctions}. Georgii's $\mathcal{L}_\Lambda$ is a linear
    space; the Lean object is the algebra, which is the same set.
\end{definition}

\begin{definition}[Quasilocal functions, Georgii (2.20)(b)]
    \label{def:ql-quasilocal}
    \uses{def:ql-local}
    \lean{MeasureTheory.GibbsMeasure.quasilocalFunctions, MeasureTheory.GibbsMeasure.measurable_of_mem_quasilocalFunctions}
    \leanok

    A bounded function $f : \Cfg \to \R$ is \textbf{quasilocal} if it is a uniform limit of local
    functions. The space $\overline{\mathcal{L}}$ of bounded quasilocal functions is the closure of
    $\mathcal{L}$ in the sup-norm. Every quasilocal function is measurable.
\end{definition}

\begin{definition}[Oscillation, Georgii (2.22)]
    \label{def:ql-osc}
    \lean{MeasureTheory.GibbsMeasure.oscOutside}
    \leanok

    For $f : \Cfg \to \R$ and $\Lambda \in \Fin$ put
    \[\osc_\Lambda(f) \;=\; \sup\set{|f(\zeta) - f(\eta)| \;:\; \zeta, \eta \in \Cfg,\;
      \zeta_\Lambda = \eta_\Lambda},\]
    the oscillation of $f$ under changes outside $\Lambda$. The Lean definition takes values in
    $[0,\infty]$ and is stated for an arbitrary $s \subseteq S$ (\texttt{\_root\_.oscOutside});
    the displayed $\osc_\Lambda$ is that definition at $s = \Lambda$. It is antitone in $\Lambda$.
\end{definition}

\begin{theorem}[Oscillation criterion, Georgii Remark (2.21)(1)]
    \label{thm:ql-osc-criterion}
    \uses{def:ql-quasilocal, def:ql-osc}
    \lean{MeasureTheory.GibbsMeasure.mem_quasilocalFunctions_iff, MeasureTheory.GibbsMeasure.tendsto_oscOutside_of_mem_quasilocalFunctions, MeasureTheory.GibbsMeasure.mem_quasilocalFunctions_of_tendsto_oscOutside}
    \leanok

    A bounded measurable $f : \Cfg \to \R$ is quasilocal if and only if
    \[\lim_{\Lambda \in \Fin} \osc_\Lambda(f) = 0,\]
    the limit being taken along the directed set $\Fin$.
\end{theorem}
\begin{proof}
    \leanok

    If $\|f - g\| < \varepsilon$ with $g \in \mathcal{L}_{\Lambda_0}$, then for $\Lambda \supseteq
    \Lambda_0$ and $\zeta_\Lambda = \eta_\Lambda$ one has $g(\zeta) = g(\eta)$, whence
    $|f(\zeta) - f(\eta)| \le 2\varepsilon$; so $\osc_\Lambda(f) \to 0$. Conversely, fix
    $\eta \in \Cfg$ and set $g_\Lambda(\omega) = f(\omega_\Lambda \eta_{S\setminus\Lambda})$. Each
    $g_\Lambda$ is local and $\|f - g_\Lambda\| \le \osc_\Lambda(f)$, so $f$ is a uniform limit of
    local functions.
\end{proof}

\begin{remark}
    \label{rmk:ql-tail-not-quasilocal}
    Georgii draws from (2.22) the conclusion that a \emph{non-constant tail-measurable function is
    never quasilocal}: its oscillation outside any finite volume is the full oscillation of the
    function. This corollary is not recorded as a separate declaration in the library; the two
    places where it is needed --- Examples \ref{ex:ql-2-27} and \ref{ex:ql-4-16} --- each contradict
    Theorem \ref{thm:ql-osc-criterion} directly with an explicit pair of configurations.
\end{remark}

\begin{proposition}[Uniformly continuous observables are quasilocal, Georgii (2.21)(2)]
    \label{prop:ql-unif-cont}
    \uses{def:ql-quasilocal, thm:ql-osc-criterion}
    \lean{MeasureTheory.GibbsMeasure.mem_quasilocalFunctions_of_uniformContinuous}
    \leanok

    Let $E$ carry a uniform structure and $\Cfg = E^S$ the product uniformity. Every bounded,
    measurable, uniformly continuous $f : \Cfg \to \R$ is quasilocal.

    Georgii states this for a metric $d$ on $\Cfg$ inducing the product topology. The Lean statement
    is phrased instead with a uniform structure on $E$ and the product uniformity on $\Cfg$ --- no
    metric and no countability of $S$ --- and it carries measurability of $f$ as an explicit
    hypothesis, where Georgii relies on his standing convention that observables are measurable.
\end{proposition}
\begin{proof}
    \leanok

    Given $\varepsilon > 0$ choose an entourage on which $f$ oscillates by less than $\varepsilon$;
    by definition of the product uniformity it contains all pairs agreeing on some finite
    $\Lambda_0$. Hence $\osc_\Lambda(f) \le \varepsilon$ for $\Lambda \supseteq \Lambda_0$, and
    Theorem \ref{thm:ql-osc-criterion} applies.
\end{proof}

\begin{proposition}[Discrete compact spin space, Georgii (2.21)(3)]
    \label{prop:ql-continuous-iff}
    \uses{def:ql-quasilocal, prop:ql-unif-cont}
    \lean{MeasureTheory.GibbsMeasure.mem_quasilocalFunctions_iff_continuous, MeasureTheory.GibbsMeasure.continuous_of_mem_localFunctions}
    \leanok

    If $E$ is discrete and compact and $\Cfg$ carries the product topology, then a bounded
    measurable $f : \Cfg \to \R$ is quasilocal if and only if it is continuous.

    Georgii assumes $E$ finite; ``discrete and compact'' is the same condition phrased
    topologically. The implication ``quasilocal $\Rightarrow$ continuous'' is proved in Lean without
    compactness: local functions on a discrete $E$ are continuous, and continuity is preserved under
    uniform limits.
\end{proposition}
\begin{proof}
    \leanok

    A local function factors through finitely many discrete coordinates, hence is continuous, and
    the continuous bounded functions form a closed subset for the sup-norm; this gives one
    direction. For the other, a continuous function on the compact space $\Cfg$ is uniformly
    continuous, so Proposition \ref{prop:ql-unif-cont} applies.
\end{proof}

\subsection*{Quasilocal specifications}

\begin{definition}[Action of a specification on observables]
    \label{def:ql-action}
    \uses{def:spec}
    \lean{Specification.action, Specification.action_apply, Specification.measurable_action, Specification.action_mem_localFunctionsOn_compl}
    \leanok

    For a specification $\gamma$, $\Lambda \in \Fin$ and a bounded $f$, the observable
    $\gamma_\Lambda f$ is
    \[(\gamma_\Lambda f)(\eta) \;=\; \int f(\sigma)\, \gamma_\Lambda(\mathrm{d}\sigma \mid \eta).\]
    It is again bounded (by $\|f\|$, since $\gamma_\Lambda$ is Markov), and for measurable $f$ it is
    $\mathcal{F}_{S\setminus\Lambda}$-measurable.
\end{definition}

\begin{lemma}[The action is a contraction]
    \label{lem:ql-action-contraction}
    \uses{def:ql-action}
    \lean{Specification.dist_action_le}
    \leanok

    For bounded measurable $f, g$ and every $\Lambda \in \Fin$,
    $\|\gamma_\Lambda f - \gamma_\Lambda g\| \le \|f - g\|$.
\end{lemma}
\begin{proof}
    \leanok

    Pointwise, $|(\gamma_\Lambda(f-g))(\eta)| \le \int |f - g| \,\mathrm{d}\gamma_\Lambda(\cdot\mid\eta)
    \le \|f-g\|$, because $\gamma_\Lambda(\cdot \mid \eta)$ is a probability measure.
\end{proof}

\begin{definition}[Quasilocal specification, Georgii (2.23)]
    \label{def:ql-quasilocal-spec}
    \uses{def:spec, def:ql-quasilocal, def:ql-action}
    \lean{Specification.IsQuasilocal}
    \leanok

    A specification $\gamma$ is \textbf{quasilocal} if $f \in \overline{\mathcal{L}}$ implies
    $\gamma_\Lambda f \in \overline{\mathcal{L}}$ for every $\Lambda \in \Fin$. No topology on $E$
    enters this definition.
\end{definition}

\begin{lemma}[It suffices to test local observables, Georgii after (2.23)]
    \label{lem:ql-check-local}
    \uses{def:ql-quasilocal-spec, lem:ql-action-contraction, def:ql-local}
    \lean{Specification.isQuasilocal_iff_forall_mem_localFunctions}
    \leanok

    $\gamma$ is quasilocal if and only if $\gamma_\Lambda f \in \overline{\mathcal{L}}$ for every
    $\Lambda \in \Fin$ and every \emph{local} $f$.
\end{lemma}
\begin{proof}
    \leanok

    $\overline{\mathcal{L}}$ is closed, and by Lemma \ref{lem:ql-action-contraction} the map
    $f \mapsto \gamma_\Lambda f$ is $1$-Lipschitz; so it carries the closure of $\mathcal{L}$ into
    the closure of $\gamma_\Lambda(\mathcal{L}) \subseteq \overline{\mathcal{L}}$.
\end{proof}

\begin{proposition}[Independent specifications are quasilocal, Georgii after (2.23)]
    \label{prop:ql-isssd}
    \uses{def:ql-quasilocal-spec, def:isssd-spec, lem:ql-check-local}
    \lean{Specification.IsResampling.isQuasilocal, Specification.isQuasilocal_isssd, Specification.IsResampling.action_mem_localFunctionsOn}
    \leanok

    Every independent specification $\ISSSD(\nu)$ is quasilocal.

    The Lean statement is stronger than Georgii's remark in two ways. First, it is proved for every
    \emph{resampling} specification --- one whose kernels are $\gamma_\Lambda(\cdot \mid \eta) =
    \mu_\Lambda \circ \juxtBy{\eta}^{-1}$ for some measure $\mu_\Lambda$ on $E^\Lambda$ --- which
    includes the inhomogeneous independent specifications, not only $\ISSSD(\nu)$. Second, the
    conclusion is exact rather than approximate: $\gamma_\Lambda$ maps $\mathcal{L}_\Delta$ into
    $\mathcal{L}_{\Delta \setminus \Lambda}$, so local observables stay local.
\end{proposition}
\begin{proof}
    \leanok

    If $f$ is $\mathcal{F}_\Delta$-measurable then, writing the kernel as the image of a fixed
    measure under $\zeta \mapsto \juxtBy{\eta}(\zeta)$, the integral
    $\int f(\juxtBy{\eta}\zeta)\, \mu_\Lambda(\mathrm{d}\zeta)$ depends on $\eta$ only through the
    coordinates in $\Delta \setminus \Lambda$. Hence $\gamma_\Lambda f \in
    \mathcal{L}_{\Delta\setminus\Lambda} \subseteq \overline{\mathcal{L}}$, and
    Lemma \ref{lem:ql-check-local} concludes.
\end{proof}

\subsection*{Georgii (2.24): quasilocality of \texorpdfstring{$\lambda$}{lambda}-specifications}

\begin{proposition}[Georgii (2.24)(a)]
    \label{prop:ql-modification}
    \uses{def:ql-quasilocal-spec, def:modif}
    \lean{Specification.IsQuasilocal.modification}
    \leanok

    Let $\gamma$ be a quasilocal specification and $\rho$ a modifier of $\gamma$ such that every
    $\rho_\Lambda$ is finite and, as a real-valued function, is a \emph{bounded} quasilocal
    observable. Then $\rho\gamma$ is quasilocal.

    This is not literally Georgii's (2.24)(a). He takes $\gamma = \rho\lambda$ for an a priori
    measure $\lambda \in \Meas(E, \mathcal{E})$ and concludes from either (i) each $\rho_\Lambda$
    local, for arbitrary $\lambda$, or (ii) $\lambda$ finite and each $\rho_\Lambda$ quasilocal; in
    neither branch is $\rho_\Lambda$ assumed bounded. The Lean version replaces the reference
    $\lambda$-specification by an arbitrary quasilocal $\gamma$ --- Georgii's branch (ii) is the case
    where $\gamma$ is the independent specification of the normalised finite $\lambda$, so this is a
    genuine generalisation of the reference --- at the price of requiring each $\rho_\Lambda$ to be
    bounded, i.e.\ to lie in $\overline{\mathcal{L}}$, whereas Georgii allows unbounded quasilocal
    densities. Branch (i), local $\rho_\Lambda$ against an infinite $\lambda$, is not covered.
\end{proposition}
\begin{proof}
    \leanok

    For local $f$ one has $(\rho\gamma)_\Lambda f = \gamma_\Lambda(\rho_\Lambda f)$ by definition of
    the modification as a density kernel. Since $\overline{\mathcal{L}}$ is an algebra,
    $\rho_\Lambda f \in \overline{\mathcal{L}}$, and quasilocality of $\gamma$ gives
    $\gamma_\Lambda(\rho_\Lambda f) \in \overline{\mathcal{L}}$. Conclude with
    Lemma \ref{lem:ql-check-local}.
\end{proof}

\begin{definition}[Boltzmann factor of a bounded observable]
    \label{def:ql-boltzmann}
    \uses{def:ql-quasilocal}
    \lean{Specification.boltzmann, Specification.boltzmann_mem_quasilocalFunctions, MeasureTheory.GibbsMeasure.exp_mem_quasilocalFunctions}
    \leanok

    For a bounded observable $H$ put $h = e^{-H}$. If $H \in \overline{\mathcal{L}}$ then
    $h \in \overline{\mathcal{L}}$, and $h \ge e^{-\|H\|} > 0$. More generally the exponential of a
    quasilocal observable is quasilocal, since $\overline{\mathcal{L}}$ is a closed subalgebra.
\end{definition}

\begin{theorem}[Georgii (2.24)(b), bounded-Hamiltonian case]
    \label{thm:ql-gibbsian-quasilocal}
    \uses{prop:ql-modification, def:ql-boltzmann, prop:ql-isssd}
    \lean{Specification.IsResampling.isQuasilocal_modification_relNorm, Specification.isQuasilocal_modification_premodifierNorm}
    \leanok

    Let $\gamma$ be a resampling specification (for instance $\ISSSD(\nu)$) and let
    $(H_\Lambda)_{\Lambda \in \Fin}$ be bounded quasilocal observables. If the normalised densities
    \[\rho_\Lambda \;=\; \frac{e^{-H_\Lambda}}{\gamma_\Lambda\!\left(e^{-H_\Lambda}\right)}\]
    form a modifier of $\gamma$, then the resulting Gibbsian specification $\rho\gamma$ is
    quasilocal.

    Georgii states (2.24)(b) for an arbitrary a priori measure $\lambda \in \Meas(E,\mathcal{E})$
    and for quasilocal, not necessarily bounded, Hamiltonians $H_\Lambda^\Phi$. The Lean statement
    requires $H_\Lambda \in \overline{\mathcal{L}}$, i.e.\ bounded. By Georgii's own (2.25)(ii),
    bounded Hamiltonians force $\lambda$ to be finite, which is exactly the case a Markov (hence
    resampling) reference specification captures; so the two restrictions match.
\end{theorem}
\begin{proof}
    \leanok

    Write $Z_\Lambda = \gamma_\Lambda h_\Lambda$ with $h_\Lambda = e^{-H_\Lambda}$. Then
    $Z_\Lambda \ge e^{-\|H_\Lambda\|} > 0$, so $Z_\Lambda$ is bounded away from $0$ and, being the
    $\gamma_\Lambda$-action of a quasilocal observable, is itself quasilocal by
    Proposition \ref{prop:ql-isssd}. A closed subalgebra of bounded functions is stable under
    inversion of elements bounded away from $0$, so $\rho_\Lambda = h_\Lambda / Z_\Lambda$ is a
    bounded quasilocal density, and Proposition \ref{prop:ql-modification} applies. This replaces
    Georgii's explicit $\varepsilon$-estimate $\lambda_\Lambda|\rho_\Lambda^\Phi - g| \le
    2(e^\varepsilon - 1)$ by the algebraic closure properties of $\overline{\mathcal{L}}$.
\end{proof}

\begin{remark}[Georgii (2.24)(c)]
    \label{rmk:ql-2-24-c}
    Georgii's converse --- if $E$ is finite, $\lambda$ is counting measure and $\gamma$ is
    quasilocal, then the unique $\lambda$-modification $\rho$ with $\gamma = \rho\lambda$ satisfies
    $\rho_\Lambda \in \overline{\mathcal{L}}$ for all $\Lambda$, via
    $\rho_\Lambda = \sum_{\zeta \in E^\Lambda} 1_{\{\sigma_\Lambda = \zeta\}}\,
    \gamma_\Lambda 1_{\{\sigma_\Lambda = \zeta\}}$ --- is not formalised. The library has no
    declaration for it.
\end{remark}

\begin{example}[Georgii (2.25): absolutely summable potentials]
    \label{ex:ql-2-25}
    \uses{thm:ql-gibbsian-quasilocal, def:ql-quasilocal}
    \lean{Potential.hamiltonianLp_mem_quasilocalFunctions, Potential.boltzmannLp_mem_quasilocalFunctions, Potential.isQuasilocal_gibbsSpecificationOfAbsolutelySummable}
    \leanok

    For $\Phi \in \Bspace$, i.e.\ absolutely summable, every Hamiltonian $H_\Lambda^\Phi$ is a
    bounded quasilocal observable (it is the sum, convergent in sup-norm, of the local interaction
    terms $\Phi_A$ with $A \cap \Lambda \ne \emptyset$), hence so is $h_\Lambda^\Phi =
    e^{-\beta H_\Lambda^\Phi}$; and therefore the Gibbsian specification $\gamma^\Phi$ relative to a
    single-spin probability measure $\nu$ is quasilocal, for every inverse temperature $\beta$.

    Georgii's Example (2.25) is stated for a \emph{uniformly convergent} $\lambda$-admissible
    potential; absolute summability is strictly stronger, so the Lean theorem covers a smaller
    class. The Lean statement also assumes $S$ countable. The two refinements Georgii records are
    not formalised: (2.25)(i), that finite range makes each $\rho_\Lambda^\Phi$ local, and
    (2.25)(ii), the stronger conclusion $\rho_\Lambda^\Phi \in \overline{\mathcal{L}}$ for bounded
    Hamiltonians.
\end{example}

\begin{remark}[The continuous (Feller) variant]
    \label{rmk:ql-feller}
    \uses{def:ql-quasilocal-spec, prop:ql-continuous-iff}
    \lean{BoundedContinuousFunction.fellerQuasilocalFunctions, Specification.IsFellerQuasilocal, Specification.IsFellerQuasilocal_iff_onCylinder}
    \leanok

    The library also carries a topological analogue, for use in weak-convergence arguments: with $E$
    a topological space, $C_b(\Cfg)$ replaces $\ell^\infty(\Cfg)$, the cylinder functions replace
    $\mathcal{L}$, and a specification with Feller kernels is \emph{Feller-quasilocal} when its
    action preserves the uniform closure of the continuous cylinder functions; again it suffices to
    test cylinder observables. This is \emph{not} Georgii's (2.23): it is a strictly smaller class
    of test functions and it presupposes a topology on $E$, whereas Definition
    \ref{def:ql-quasilocal-spec} does not. By Proposition \ref{prop:ql-continuous-iff} the two
    algebras of observables coincide when $E$ is finite and discrete.
\end{remark}

\subsection*{Non-quasilocal specifications}

\begin{proposition}[Gluing a family of specifications, Georgii Remark (2.26)]
    \label{prop:ql-glued}
    \uses{def:spec, def:gibbs-meas, lem:ql-spec-preserves-tail}
    \lean{Specification.IsMeasurableFamily, Specification.glued, Specification.isGibbsMeasure_glued}
    \leanok

    Let $(X, \MCX)$ be a measurable space with measurable singletons, $\xi : \Cfg \to X$ be
    $\Tail$-measurable, and $(\gamma^x)_{x \in X}$ a family of specifications such that
    $(\omega, x) \mapsto \gamma^x_\Lambda(A \mid \omega)$ is jointly measurable. Then
    \[\gamma_\Lambda(A \mid \omega) \;=\; \gamma^{\xi(\omega)}_\Lambda(A \mid \omega)\]
    is again a specification, and every random field $\mu$ with $\mu(\xi = x) = 1$ and
    $\mu \in \Gibbs{\gamma^x}$ satisfies $\mu \in \Gibbs{\gamma}$. Hence
    $\{\mu^x : x \in X\} \subseteq \Gibbs{\gamma}$ for any family $(\mu^x)$ as in Georgii's
    hypotheses (i)--(ii).
\end{proposition}
\begin{proof}
    \leanok

    Properness and the Markov property are pointwise in $\omega$. Consistency is the whole content:
    by Lemma \ref{lem:ql-spec-preserves-tail}, $\gamma^{\xi(\omega)}_\Delta(\cdot \mid \omega)$ is
    carried by $\{\xi = \xi(\omega)\}$, on which the inner kernel $\gamma^{\xi(\sigma)}_\Lambda$ is
    the \emph{fixed} kernel $\gamma^{\xi(\omega)}_\Lambda$; consistency of $\gamma^{\xi(\omega)}$
    then applies. The Gibbs statement is the same argument with $\mu$-almost sure constancy of
    $\xi$.
\end{proof}

\begin{example}[Georgii (2.27): Bernoulli fields on $\{0,1\}^{\mathbb{N}}$]
    \label{ex:ql-2-27}
    \uses{prop:ql-glued, def:ql-quasilocal-spec, thm:ql-osc-criterion}
    \lean{MeasureTheory.GibbsMeasure.Exchangeable.gammaEx, MeasureTheory.GibbsMeasure.Exchangeable.isGibbsMeasure_bernoulliField, MeasureTheory.GibbsMeasure.Exchangeable.not_countable_gibbsMeasures, MeasureTheory.GibbsMeasure.Exchangeable.not_isQuasilocal_gammaEx}
    \leanok

    Take $E = \{0,1\}$, $S = \mathbb{N}$, $X = [0,1]$, and let $\mu^x$ be the Bernoulli field with
    parameter $x$, a Gibbs measure for the independent specification $\gamma^x = \ISSSD(\lambda^x)$
    with $\lambda^x = x\delta_1 + (1-x)\delta_0$. The tail function
    \[\xi \;=\; \liminf_{n\to\infty} \frac{1}{n}\sum_{i < n} \sigma_i\]
    separates their supports, by the strong law of large numbers. Gluing along $\xi$ as in
    Proposition \ref{prop:ql-glued} produces a specification $\gamma$ with $\mu^x \in \Gibbs{\gamma}$ for
    every $x \in [0,1]$; in particular $\Gibbs{\gamma}$ is uncountable, and $\gamma$ is not
    quasilocal.

    The library proves in addition that Dobrushin's interdependence matrix of this $\gamma$ vanishes
    identically, so $c(\gamma) = 0 < 1$: this is why the quasilocality conjunct in Dobrushin's
    condition is not decorative. Non-quasilocality is deduced from that, by contradiction with the
    uniqueness theorem, rather than from the oscillation criterion directly.

    Georgii's further claim that $\Gibbs{\gamma}$ is \emph{exactly} the set of mixtures
    $\int w(\mathrm{d}x)\,\mu^x$, i.e.\ the exchangeable random fields --- which needs de Finetti's
    theorem, proved by Georgii only later in Example (7.31) --- is not part of this file's Lean
    content.
\end{example}

\begin{example}[Georgii (4.16): a non-quasilocal specification with no Gibbs measure]
    \label{ex:ql-4-16}
    \uses{def:ql-quasilocal-spec, def:gibbs-meas, thm:ql-osc-criterion}
    \lean{MeasureTheory.GibbsMeasure.Example416.specification, MeasureTheory.GibbsMeasure.Example416.GP_eq_empty, MeasureTheory.GibbsMeasure.Example416.not_isQuasilocal_specification, MeasureTheory.GibbsMeasure.Example416.mem_GP_of_finite}
    \leanok

    On $E = \{0,1\}$ with $S$ countably infinite, let $\gamma_\Lambda(\cdot \mid \omega)$ be the
    uniform distribution on the ``spikes'' $\omega^a$, $a \in \Lambda$, when $\omega$ vanishes off
    $\Lambda$, and the Dirac mass at $0_\Lambda\omega$ otherwise. This is a proper, consistent
    specification with $\Gibbs{\gamma} = \emptyset$, and it is not quasilocal.

    This example is numbered (4.16) by Georgii, not (2.2$\cdot$); it is recorded here because it is
    the sharpest illustration of what Definition \ref{def:ql-quasilocal-spec} excludes --- it is
    Georgii's witness that quasilocality cannot be dropped from the existence theorems (4.17) and
    (4.22). The library also proves that for \emph{finite} $S$ the same kernels do admit a Gibbs
    measure, so the infiniteness hypothesis is sharp.

    The two arguments are Georgii's. Non-quasilocality: $f = 1_{\{\sigma_a = 1\}}$ is local, but
    $(\gamma_{\{a\}} f)(\eta) = 1_{\{\eta = 0 \text{ off } \{a\}\}}$, whose oscillation outside any
    finite $\Delta$ is $1$ --- put a spike at a site outside $\Delta \cup \{a\}$ --- contradicting
    Theorem \ref{thm:ql-osc-criterion}. Emptiness of $\Gibbs{\gamma}$: a Gibbs measure would give
    mass $0$ to every configuration with two or more $1$s, to the zero configuration, and (letting
    $|\Lambda| \to \infty$ in $\gamma_\Lambda(\{\omega^a\} \mid \cdot) \le |\Lambda|^{-1}$) to every
    single spike; these exhaust $\Cfg$.
\end{example}
