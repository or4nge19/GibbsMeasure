\section{Gibbs representation of pre-modifications}

Throughout, $S$ is a countable set, $(E,\mathcal{E})$ a measurable space, $\Cfg = E^S$, and
$\lambda \in \Meas(E,\mathcal{E})$ a $\sigma$-finite a priori measure.  A positive quasilocal
$\lambda$-modification $\rho = (\rho_\Lambda)_{\Lambda \in \Fin}$ is Gibbsian for a potential that
is unique once normalized.

% ---------------------------------------------------------------------------
\begin{definition}[Georgii (2.28), $\alpha$-normalized potential]
    \label{def:grep-alpha-normalized}

    Let $\alpha \in \Prob(E,\mathcal{E})$ and let $\alpha_B$ ($B \subset S$) be the probability
    kernels from $\Tail_B$ to $\mathcal{F}$ of Remark (1.25),
    \[
        \alpha_B f(\omega) = \int \alpha^B(\mathrm{d}\zeta)\, f(\zeta_B \omega_{S \setminus B}).
    \]
    A potential $\Phi$ is \textbf{$\alpha$-normalized} if
    \[
        \alpha_B \Phi_A = \int \alpha^B(\mathrm{d}\zeta)\, \Phi_A(\zeta_B \sigma_{S\setminus B}) = 0
        \qquad \text{whenever } \emptyset \neq B \subset A \in \Fin .
    \]
    For $\alpha = \delta_a$, $a \in E$, such a $\Phi$ is a \textbf{gas potential with vacuum
    state $a$}.
\end{definition}

\begin{definition}[Vacuum configurations]
    \label{def:grep-vacuum}
    \lean{Potential.vacuum, Potential.vacuumOn}
    \leanok

    For $a \in E$, $C, \Lambda \in \Finset(S)$ and $\eta \in \Cfg$ let
    \[
        (\eta_C a_{S \setminus C})_i = \begin{cases} \eta_i & i \in C \\ a & i \notin C\end{cases}
        \qquad
        (a_\Lambda \eta_{S\setminus\Lambda})_i = \begin{cases} a & i \in \Lambda \\ \eta_i & i \notin \Lambda .\end{cases}
    \]
    For $\alpha = \delta_a$ the kernel of Definition~\ref{def:grep-alpha-normalized} acts by
    $\alpha_{S\setminus C} f(\eta) = f(\eta_C a_{S\setminus C})$.
\end{definition}

\begin{definition}[Georgii (2.29)(1), gas potential]
    \label{def:grep-gas-potential}
    \uses{def:grep-alpha-normalized}
    \lean{Potential.IsGasPotential}
    \leanok

    A potential $\Phi$ is a \textbf{gas potential with vacuum state $a$} if
    \[
        \Phi_A(\omega) = 0 \quad \text{whenever } \omega_i = a \text{ for some } i \in A .
    \]
    This is the case $\alpha = \delta_a$ of Definition~\ref{def:grep-alpha-normalized}.
\end{definition}

\begin{definition}[Georgii (2.22), quasilocal function]
    \label{def:grep-quasilocal-fun}
    \lean{MeasureTheory.GibbsMeasure.IsQuasilocalFun}
    \leanok

    $f : \Cfg \to \R$ is \textbf{quasilocal} if for every $\varepsilon > 0$ there is
    $\Delta \in \Finset(S)$ with $|f(\zeta) - f(\eta)| \leq \varepsilon$ whenever
    $\zeta_\Delta = \eta_\Delta$.  A family $(\rho_\Lambda)_{\Lambda \in \Fin}$ of
    $[0,\infty]$-valued densities is quasilocal if each real-valued function
    $\eta \mapsto \rho_\Lambda(\eta)$ is.
\end{definition}

% ---------------------------------------------------------------------------
\begin{definition}[Georgii, proof of (2.30), step 1: the operator $p_A$]
    \label{def:grep-mobius}
    \uses{def:grep-vacuum}
    \lean{Potential.mobius}
    \leanok

    For $A \in \Fin \union \set{\emptyset}$ and $f : \Cfg \to \R$ put
    \[
        p_A f = \sum_{C \subset A} (-1)^{|A \setminus C|}\, \alpha_{S \setminus C} f ,
    \]
    which for $\alpha = \delta_a$ reads
    $p_A f(\eta) = \sum_{C \subset A} (-1)^{|A\setminus C|} f(\eta_C a_{S \setminus C})$.
\end{definition}

\begin{lemma}[Georgii, proof of (2.30), step 1 (i)--(iii)]
    \label{lem:grep-mobius-properties}
    \uses{def:grep-mobius, def:grep-gas-potential, def:grep-vacuum}
    \lean{Potential.dependsOn_mobius, Potential.mobius_eq_zero, Potential.sum_powerset_mobius}
    \leanok

    Let $f : \Cfg \to \R$ be arbitrary.
    \begin{enumerate}
        \item $p_A f(\eta)$ depends on $\eta$ only through $\eta_A$; in particular $p_A f$ is
        $\mathcal{F}_A$-measurable when $f$ is measurable.
        \item For all $\Lambda \in \Fin \union \set{\emptyset}$,
        $\displaystyle \alpha_{S\setminus\Lambda} f = \sum_{A \subset \Lambda} p_A f$.
        \item $\alpha_B(p_A f) = 0$ for all $\emptyset \neq B \subset A$; for $\alpha = \delta_a$,
        $p_A f(\eta) = 0$ as soon as $\eta_i = a$ for some $i \in A$.
    \end{enumerate}
\end{lemma}
\begin{proof}
    \leanok

    (i) Each summand $f(\eta_C a_{S\setminus C})$, $C \subset A$, reads only the coordinates in
    $C$.  (ii) Exchanging the two sums,
    $\sum_{C \subset A \subset \Lambda}(-1)^{|A \setminus C|} = \delta_{C,\Lambda}$.  (iii) It
    suffices to treat $B = \set{i}$.  Pair each $C \not\ni i$ with $C \union \set{i}$; since
    $\alpha_{\set i}\alpha_{S\setminus C} = \alpha_{S \setminus C}
    = \alpha_{\set i}\alpha_{S \setminus (C \union \set i)}$ (for $\alpha = \delta_a$: the two
    vacuum configurations coincide when $\eta_i = a$) and the exponents $|A \setminus C|$ differ
    by one, the two contributions cancel.
\end{proof}

\begin{definition}[$u_\Lambda = \log \rho_\Lambda$]
    \label{def:grep-log-density}
    \lean{Potential.logDensity}
    \leanok

    For a family $\rho$ of $[0,\infty]$-valued densities, $u_\Lambda = \log \rho_\Lambda$ is the
    logarithm of the real-valued function $\eta \mapsto \rho_\Lambda(\eta)$.
\end{definition}

\begin{lemma}[Georgii (1.31) in logarithmic form]
    \label{lem:grep-log-density-premod}
    \uses{def:premodif, def:grep-log-density}
    \lean{Potential.logDensity_sub_comm}
    \leanok

    Let $\rho$ be a pre-modification with $0 < \rho_\Lambda < \infty$ everywhere.  For
    $\Lambda_1 \subset \Lambda_2$ and $\zeta,\omega$ agreeing off $\Lambda_1$,
    \[
        u_{\Lambda_2}(\zeta) - u_{\Lambda_2}(\omega) = u_{\Lambda_1}(\zeta) - u_{\Lambda_1}(\omega).
    \]
\end{lemma}
\begin{proof}
    \leanok

    Take logarithms in
    $\rho_{\Lambda_2}(\zeta)\rho_{\Lambda_1}(\omega) = \rho_{\Lambda_1}(\zeta)\rho_{\Lambda_2}(\omega)$
    (Definition \ref{def:premodif}); all four factors are in $(0,\infty)$.
\end{proof}

\begin{definition}[The gas potential of a pre-modification, Georgii (2.30) step 2]
    \label{def:grep-gas-potential-of-premod}
    \uses{def:grep-mobius, def:grep-log-density}
    \lean{Potential.gasPotential}
    \leanok

    With $u_A = \log \rho_A$ set $\Phi^a_A = -p_A u_A$, i.e.
    \[
        \Phi^a_A(\omega) = -\sum_{C \subset A} (-1)^{|A \setminus C|} \log \rho_A(\omega_C a_{S\setminus C}).
    \]
\end{definition}

\begin{lemma}[Georgii (2.30), step 2]
    \label{lem:grep-gas-potential-step2}
    \uses{def:grep-gas-potential-of-premod, lem:grep-mobius-properties, def:grep-gas-potential}
    \lean{Potential.isGasPotential_gasPotential, Potential.isPotential_gasPotential}
    \leanok

    For any family $\rho$, $\Phi^a$ is a gas potential with vacuum state $a$.  If each
    $\rho_\Lambda$ is measurable, $\Phi^a$ is a potential in the sense of Georgii (2.2)(i): each
    $\Phi^a_A$ is $\mathcal{F}_A$-measurable.
\end{lemma}
\begin{proof}
    \leanok

    Lemma~\ref{lem:grep-mobius-properties}(iii) and (i) with $f = u_A$.  For measurability,
    $\eta \mapsto \eta_C a_{S \setminus C}$ is measurable and $\log$ is measurable on
    $(0,\infty)$; $\mathcal{F}_A$-measurability follows from (i).
\end{proof}

\begin{lemma}[Georgii (2.30), step 3]
    \label{lem:grep-gas-potential-step3}
    \uses{def:grep-gas-potential-of-premod, lem:grep-log-density-premod}
    \lean{Potential.gasPotential_eq_neg_mobius}
    \leanok

    For a pre-modification $\rho$ with $0 < \rho_\Lambda < \infty$ everywhere and
    $\emptyset \neq A \subset \Delta \in \Fin$,
    \[
        \Phi^a_A = -p_A u_\Delta .
    \]
\end{lemma}
\begin{proof}
    \leanok

    By Lemma~\ref{lem:grep-log-density-premod}, $u_A - u_\Delta$ takes the same value at every
    $\omega_C a_{S \setminus C}$, $C \subset A$, all of which agree off $A$ with the constant
    configuration $\mathbf{a}$.  Hence
    $p_A u_A - p_A u_\Delta = (u_A(\mathbf a) - u_\Delta(\mathbf a)) \sum_{C \subset A}(-1)^{|A\setminus C|} = 0$
    for nonempty $A$.
\end{proof}

\begin{lemma}[Georgii (2.30), step 4: the partial Hamiltonians]
    \label{lem:grep-partial-hamiltonian}
    \uses{lem:grep-gas-potential-step3, def:grep-vacuum}
    \lean{Potential.sum_powerset_hamiltonianTerms_gasPotential}
    \leanok

    Let $\rho$ be a pre-modification with $0 < \rho_\Lambda < \infty$ everywhere and let
    $H^\Phi_{\Lambda,\Delta} = \sum_{A \cap \Lambda \neq \emptyset,\ A \subset \Delta} \Phi_A$
    be the partial Hamiltonian (2.13).  For $\Lambda \subset \Delta \in \Fin$,
    \[
        H^{\Phi^a}_{\Lambda,\Delta} = \alpha_{S\setminus\Delta}\bigl(\alpha_\Lambda u_\Lambda - u_\Lambda\bigr),
        \qquad\text{i.e.}\qquad
        H^{\Phi^a}_{\Lambda,\Delta}(\omega) = \log \frac{\rho_\Lambda(\omega_{\Delta \setminus \Lambda} a_{S \setminus (\Delta \setminus \Lambda)})}{\rho_\Lambda(\omega_\Delta a_{S \setminus \Delta})} .
    \]
\end{lemma}
\begin{proof}
    \leanok

    Write $H^{\Phi^a}_{\Lambda,\Delta} = \sum_{\emptyset \neq A \subset \Delta}\Phi^a_A - \sum_{\emptyset \neq A \subset \Delta\setminus\Lambda}\Phi^a_A$
    and replace each $\Phi^a_A$ by $-p_A u_\Delta$ (Lemma \ref{lem:grep-gas-potential-step3}).
    Lemma~\ref{lem:grep-mobius-properties}(ii) collapses the two powerset sums to
    $-\alpha_{S\setminus\Delta}u_\Delta + \alpha_{S\setminus(\Delta\setminus\Lambda)}u_\Delta
    = \alpha_{S\setminus\Delta}(\alpha_\Lambda u_\Delta - u_\Delta)$, and
    $\alpha_\Lambda u_\Delta - u_\Delta = \alpha_\Lambda u_\Lambda - u_\Lambda$ by
    Lemma~\ref{lem:grep-log-density-premod}.
\end{proof}

\begin{theorem}[Georgii (2.30), step 4: the Hamiltonian exists]
    \label{thm:grep-hamiltonian-gas}
    \uses{lem:grep-partial-hamiltonian, def:grep-quasilocal-fun}
    \lean{Potential.hasSum_hamiltonianTerms_gasPotential, Potential.isSummable_gasPotential, Potential.hamiltonian_gasPotential}
    \leanok

    Let $\rho$ be a quasilocal pre-modification with $0 < \rho_\Lambda < \infty$ everywhere.  Then
    $\Phi^a$ satisfies Georgii (2.2)(ii): for every $\Lambda \in \Fin$ the limit
    $H^{\Phi^a}_\Lambda = \lim_\Delta H^{\Phi^a}_{\Lambda,\Delta}$ along $\Delta \uparrow S$
    (Convention (2.1)) exists, and
    \[
        H^{\Phi^a}_\Lambda = v_\Lambda := \alpha_\Lambda u_\Lambda - u_\Lambda
        = \log \rho_\Lambda(a_\Lambda \omega_{S\setminus\Lambda}) - \log \rho_\Lambda(\omega).
    \]
\end{theorem}
\begin{proof}
    \leanok

    For $\Delta \supset \Lambda$, Lemma~\ref{lem:grep-partial-hamiltonian} gives
    $H^{\Phi^a}_{\Lambda,\Delta}(\omega) = \log \rho_\Lambda(\omega_{\Delta\setminus\Lambda}a_{S\setminus(\Delta\setminus\Lambda)})
    - \log \rho_\Lambda(\omega_\Delta a_{S\setminus\Delta})$.  By quasilocality,
    $\rho_\Lambda(\omega_\Delta a_{S\setminus\Delta}) \to \rho_\Lambda(\omega)$ and
    $\rho_\Lambda(\omega_{\Delta\setminus\Lambda}a_{\cdots}) \to \rho_\Lambda(a_\Lambda\omega_{S\setminus\Lambda})$;
    compose with the continuity of $\log$ on $(0,\infty)$.
\end{proof}

\begin{lemma}[Georgii (2.30), step 4: Boltzmann factor and partition function]
    \label{lem:grep-partition-function}
    \uses{thm:grep-hamiltonian-gas}
    \lean{Potential.boltzmannFactor_gasPotential, Potential.sigmaFiniteLambdaZ_gasPotential, Potential.isSigmaFiniteLambdaAdmissible_gasPotential}
    \leanok

    Let $\rho$ be a quasilocal pre-modification with $0 < \rho_\Lambda < \infty$ everywhere and
    $\lambda_\Lambda \rho_\Lambda = 1$ for all $\Lambda \in \Fin$.  Then
    \[
        h^{\Phi^a}_\Lambda = \rho_\Lambda \exp(-\alpha_\Lambda u_\Lambda),
        \qquad
        Z^{\Phi^a}_\Lambda = \lambda_\Lambda h^{\Phi^a}_\Lambda = \exp(-\alpha_\Lambda u_\Lambda),
    \]
    and $\Phi^a$ is $\lambda$-admissible: $0 < Z^{\Phi^a}_\Lambda < \infty$ everywhere.
\end{lemma}
\begin{proof}
    \leanok

    The first identity is $h^\Phi_\Lambda = \exp(-H^\Phi_\Lambda)$ and
    Theorem~\ref{thm:grep-hamiltonian-gas}.  The factor $\exp(-\alpha_\Lambda u_\Lambda)$ is
    $\Tail_\Lambda$-measurable, so it comes out of the $\lambda_\Lambda$-integral, leaving
    $\lambda_\Lambda \rho_\Lambda = 1$.  The value is a strictly positive real.
\end{proof}

\begin{proposition}[Georgii (2.30): $\rho = \rho^{\Phi^a}$]
    \label{prop:grep-rho-eq}
    \uses{lem:grep-partition-function}
    \lean{Potential.sigmaFinitePremodifierNorm_gasPotential}
    \leanok

    Under the hypotheses of Lemma~\ref{lem:grep-partition-function},
    $\rho^{\Phi^a}_\Lambda = h^{\Phi^a}_\Lambda / Z^{\Phi^a}_\Lambda = \rho_\Lambda$
    for every $\Lambda \in \Fin$.
\end{proposition}
\begin{proof}
    \leanok

    Divide the two identities of Lemma~\ref{lem:grep-partition-function}; the common factor
    $\exp(-\alpha_\Lambda u_\Lambda)$ is in $(0,\infty)$.
\end{proof}

\begin{lemma}[Georgii (2.36) for $\alpha = \delta_a$]
    \label{lem:grep-hamiltonian-vacuum}
    \uses{def:grep-gas-potential, def:grep-vacuum}
    \lean{Potential.sum_powerset_hamiltonianTerms_vacuum, Potential.hamiltonian_vacuum}
    \leanok

    Let $\Theta$ be a summable gas potential with vacuum state $a$.  For $\Lambda \subset \Delta$
    the partial Hamiltonians $H^\Theta_{\Lambda,\Delta}(\omega_\Lambda a_{S\setminus\Lambda})$ do
    not depend on $\Delta$, and
    \[
        \alpha_{S\setminus\Lambda} H^\Theta_\Lambda
        = H^\Theta_\Lambda(\omega_\Lambda a_{S\setminus\Lambda})
        = \sum_{\emptyset \neq A \subset \Lambda} \Theta_A(\omega).
    \]
\end{lemma}
\begin{proof}
    \leanok

    A term $\Theta_A(\omega_\Lambda a_{S\setminus\Lambda})$ with $A \not\subset \Lambda$ has a
    site $i \in A \setminus \Lambda$ at the vacuum value and vanishes; a term with
    $A \cap \Lambda = \emptyset$ is absent from the partial Hamiltonian.  The remaining sum runs
    over $\emptyset \neq A \subset \Lambda$, where $\Theta_A$ reads only coordinates in $\Lambda$
    and therefore sees $\omega$.
\end{proof}

\begin{theorem}[Georgii (2.35)(a) for $\alpha = \delta_a$]
    \label{thm:grep-gas-potential-vanishes}
    \uses{lem:grep-hamiltonian-vacuum}
    \lean{Potential.eq_zero_of_isGasPotential}
    \leanok

    Let $\Theta$ be a summable gas potential with vacuum state $a$ such that for every
    $\Lambda \in \Fin$, $H^\Theta_\Lambda$ takes the same value at any two configurations
    agreeing off $\Lambda$ (in particular, if $H^\Theta_\Lambda$ is $\Tail_\Lambda$-measurable).
    Then $\Theta_A = 0$ for every $A \in \Fin$, i.e. every nonempty finite $A$.
\end{theorem}
\begin{proof}
    \leanok

    By hypothesis $H^\Theta_\Lambda$ takes the same value at
    $\omega_\Lambda a_{S \setminus \Lambda}$ and at $\mathbf a$, where every term vanishes.
    Lemma~\ref{lem:grep-hamiltonian-vacuum} gives
    $\sum_{\emptyset \neq A \subset B} \Theta_A(\omega) = 0$ for all finite $B$ and all $\omega$;
    induction on $|B|$ yields $\Theta_B = 0$.
\end{proof}

\begin{proposition}[Georgii (2.34), (ii) $\Rightarrow$ (i)]
    \label{prop:grep-2-34-ii-i}
    \uses{prop:grep-rho-eq}
    \lean{Potential.hamiltonian_sub_eq_log_sigmaFiniteLambdaZ, Potential.dependsOn_hamiltonian_sub_of_sigmaFinitePremodifierNorm_eq}
    \leanok

    Let $S$ be an arbitrary index set and let $\Phi,\Psi$ be summable $\lambda$-admissible
    potentials at inverse temperature $\beta = 1$ with $\rho^\Phi = \rho^\Psi$.  Then
    \[
        H^\Phi_\Lambda - H^\Psi_\Lambda = \log \frac{Z^\Psi_\Lambda}{Z^\Phi_\Lambda},
    \]
    and $H^\Phi_\Lambda - H^\Psi_\Lambda$ depends on the configuration only off $\Lambda$, i.e. is
    $\Tail_\Lambda$-measurable: $\Phi$ and $\Psi$ are equivalent in the sense of Georgii (2.33).
\end{proposition}
\begin{proof}
    \leanok

    Take logarithms in $h^\Phi_\Lambda / Z^\Phi_\Lambda = h^\Psi_\Lambda / Z^\Psi_\Lambda$ with
    $h^\Phi_\Lambda = \exp(-H^\Phi_\Lambda)$; all four quantities are in $(0,\infty)$ by
    admissibility.  $Z^\Phi_\Lambda, Z^\Psi_\Lambda$ are $\Tail_\Lambda$-measurable because
    $\lambda_\Lambda(\cdot \mid \eta)$ depends on $\eta$ only off $\Lambda$.
\end{proof}

\begin{corollary}[Georgii (2.30), step 5: uniqueness]
    \label{cor:grep-uniqueness-step5}
    \uses{thm:grep-gas-potential-vanishes, prop:grep-2-34-ii-i, def:grep-gas-potential}
    \lean{Potential.eq_of_isGasPotential_of_sigmaFinitePremodifierNorm_eq}
    \leanok

    Two summable $\lambda$-admissible gas potentials $\Phi,\Psi$ with the same vacuum state $a$ and
    $\rho^\Phi = \rho^\Psi$ satisfy $\Phi_A = \Psi_A$ for every $A \in \Fin$.
\end{corollary}
\begin{proof}
    \leanok

    $\Phi - \Psi$ is a summable gas potential with vacuum state $a$ (Georgii (2.29)(1)).  Its
    Hamiltonians are $\Tail_\Lambda$-measurable by Proposition~\ref{prop:grep-2-34-ii-i}, so
    Theorem~\ref{thm:grep-gas-potential-vanishes} applies.
\end{proof}

\begin{theorem}[Georgii (2.30), the Gibbs representation theorem]
    \label{thm:grep-2-30}
    \uses{def:premodif, def:grep-quasilocal-fun, def:grep-gas-potential, def:grep-gas-potential-of-premod, prop:grep-rho-eq, cor:grep-uniqueness-step5, lem:grep-gas-potential-step2, thm:grep-hamiltonian-gas}
    \lean{Potential.exists_unique_isGasPotential_sigmaFinitePremodifierNorm_eq}
    \leanok

    Let $S$ be countable, $\lambda \in \Meas(E,\mathcal{E})$ $\sigma$-finite, and $\rho$ a
    quasilocal pre-modification with $0 < \rho_\Lambda < \infty$ everywhere and
    $\lambda_\Lambda \rho_\Lambda = 1$ for all $\Lambda \in \Fin$, where $\lambda_\Lambda$ is the
    $\sigma$-finite reference kernel of Notation (1.26).  Then for each $a \in E$ there is a
    $\lambda$-admissible gas potential with vacuum state $a$ and $\rho = \rho^{\Phi^a}$, namely
    $\Phi^a$ of Definition~\ref{def:grep-gas-potential-of-premod}, and any two such potentials
    agree on $\Fin$.
\end{theorem}
\begin{proof}
    \leanok

    Existence: $\Phi^a_A = -p_A u_A$ is a potential and a gas potential
    (Lemma~\ref{lem:grep-gas-potential-step2}), its Hamiltonians converge with
    $H^{\Phi^a}_\Lambda = \alpha_\Lambda u_\Lambda - u_\Lambda$
    (Theorem~\ref{thm:grep-hamiltonian-gas}), it is $\lambda$-admissible with
    $Z^{\Phi^a}_\Lambda = \exp(-\alpha_\Lambda u_\Lambda)$
    (Lemma~\ref{lem:grep-partition-function}), and $\rho^{\Phi^a} = \rho$
    (Proposition~\ref{prop:grep-rho-eq}).  Uniqueness is
    Corollary~\ref{cor:grep-uniqueness-step5}.
\end{proof}

\begin{corollary}[Georgii (2.30): positive quasilocal $\lambda$-specifications are Gibbsian]
    \label{cor:grep-gibbsian-spec}
    \uses{thm:grep-2-30, def:premodif, lem:premodif-modif}
    \lean{Potential.sigmaFinitePremodifierKernel_gasPotential}
    \leanok

    Under the hypotheses of Theorem~\ref{thm:grep-2-30}, the Gibbsian kernels of $\Phi^a$ are the
    kernels of the $\lambda$-specification $\gamma = \rho\lambda$:
    \[
        \gamma^{\Phi^a}_\Lambda(\cdot \mid \eta) = \lambda_\Lambda(\cdot \mid \eta)\,\rho_\Lambda .
    \]
\end{corollary}
\begin{proof}
    \leanok

    Both sides are $\lambda_\Lambda(\cdot\mid\eta)$ with a density; the densities agree by
    Proposition~\ref{prop:grep-rho-eq}.
\end{proof}

\begin{theorem}[Georgii (2.30), the $\alpha$-normalized half]
    \label{thm:grep-2-30-normalized}
    \uses{thm:grep-2-30, def:grep-alpha-normalized}
    \lean{Potential.normalizedPotential, Potential.IsNormalized,
      Potential.IsUniformlyConvergent,
      Potential.exists_unique_isNormalized_sigmaFinitePremodifierNorm_eq}
    \leanok

    If in the situation of Theorem~\ref{thm:grep-2-30} the function $\log \rho_\Lambda$ is bounded
    for every $\Lambda \in \Fin$, then for each $\alpha \in \Prob(E,\mathcal{E})$ there is a unique
    uniformly convergent $\alpha$-normalized $\lambda$-admissible potential $\Phi^\alpha$ with
    $\rho = \rho^{\Phi^\alpha}$.  Quasilocality may here be weakened to
    $\lim_\Delta \sup_{\zeta} |\rho_\Lambda(\omega_\Delta \zeta_{S\setminus\Delta}) - \rho_\Lambda(\omega)| = 0$
    for each $\omega$ and $\Lambda$.
\end{theorem}
\begin{proof}
    \leanok

    Put $\Phi^\alpha_A = -p_A u_A$ with $p_A$ taken for $\alpha$; equivalently
    $\Phi^\alpha_A(\eta) = \int \Phi^\zeta_A(\eta)\,\alpha^S(\mathrm{d}\zeta)$, the average over
    the reference configuration $\zeta$ of the potentials of
    Definition~\ref{def:grep-gas-potential-of-premod}.  Steps 1--3 of
    Theorem~\ref{thm:grep-2-30} integrate to give $\alpha$-normalization and
    $H^{\Phi^\alpha}_{\Lambda,\Delta} = \alpha_{S\setminus\Delta} v_\Lambda$ with
    $v_\Lambda = \alpha_\Lambda u_\Lambda - u_\Lambda$.  As $u_\Lambda$ is bounded, $v_\Lambda$ is
    quasilocal, so
    \[
        \lim_\Delta \|H^{\Phi^\alpha}_{\Lambda,\Delta} - v_\Lambda\| \leq \lim_\Delta \int \alpha^{S\setminus\Delta}(\mathrm{d}\zeta)\sup_\omega |v_\Lambda(\omega_\Delta \zeta_{S\setminus\Delta}) - v_\Lambda(\omega)| = 0 ,
    \]
    which is uniform convergence with $H^{\Phi^\alpha}_\Lambda = v_\Lambda$; admissibility and
    $\rho^{\Phi^\alpha} = \rho$ follow as in Lemma~\ref{lem:grep-partition-function}.  Uniqueness
    is the vanishing theorem (2.35)(a) for an $\alpha$-normalized uniformly convergent potential
    with $\Tail_\Lambda$-measurable Hamiltonians, applied to a difference as in
    Corollary~\ref{cor:grep-uniqueness-step5}.
\end{proof}

\begin{corollary}[Georgii (2.31)]
    \label{cor:grep-2-31}
    \uses{thm:grep-2-30, thm:grep-2-30-normalized}
    \lean{Potential.exists_unique_isNormalized_of_finite}
    \leanok

    Suppose $E$ is finite, $\lambda$ is counting measure, and $\gamma$ is a positive quasilocal
    specification.  Then for each $\alpha \in \Prob(E,\mathcal{E})$ there is a unique
    $\alpha$-normalized $\lambda$-admissible potential $\Phi$ with $\gamma = \gamma^\Phi$; it is
    uniformly convergent and $H^\Phi_\Lambda \in \overline{\mathcal{L}}$ for all $\Lambda \in \Fin$.
    This is Theorem~\ref{thm:grep-2-30-normalized} for the $\lambda$-modification $\rho$ of
    $\gamma$, a quasilocal pre-modification (Georgii (2.24)(c)) with $\log \rho_\Lambda$ bounded
    by continuity on the compact space $\Cfg$.
\end{corollary}

\begin{corollary}[Georgii (2.32)]
    \label{cor:grep-2-32}
    \uses{thm:grep-2-30, thm:grep-2-30-normalized, def:grep-gas-potential-of-premod}
    \lean{Potential.IsNearestNeighbour, Potential.isNearestNeighbour_normalizedPotential,
      Potential.isNearestNeighbour_gasPotential}
    \leanok

    Let $\rho$ and $\alpha$ satisfy the hypotheses of Theorem~\ref{thm:grep-2-30-normalized} and
    let $\Phi = \Phi^\alpha$ be the corresponding $\alpha$-normalized potential.  Suppose $S$ is
    the vertex set of a simple graph with bond set $B$ and $\rho$ is Markovian: $\rho_{\set i}$ is
    $\mathcal{F}_{B(i)}$-measurable for every $i \in S$, where
    $B(i) = \set i \union \set{j \in S : \set{i,j} \in B}$.  Then $\Phi$ is a nearest-neighbour
    potential in the sense of Georgii (2.17): $\Phi_A = 0$ unless $A$ is a complete subgraph.
    For a non-adjacent pair $i,j \in A$, the subsets $C \subset A$ group into quadruples
    $D, D \union \set i, D \union \set j, D \union \set{i,j}$; by
    $\alpha_{\set i}u_A - u_A = \alpha_{\set i}u_{\set i} - u_{\set i}$ and
    $B(i) \setminus D_1 = B(i) \setminus D_3$, $B(i) \setminus D_2 = B(i)\setminus D_4$ each
    quadruple cancels.  The same holds for the gas potential $\Phi^a$ of
    Definition~\ref{def:grep-gas-potential-of-premod}, without boundedness of $\log\rho_\Lambda$.
\end{corollary}
