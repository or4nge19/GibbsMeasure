\section{Specific entropy}

Throughout, $\lambda$ is a probability measure on $E$, $\lambda^S$ the product measure on
$\Cfg$, $\mathcal H_\Lambda(\mu \mid \nu)$ the relative entropy on $\mathcal F_\Lambda$
(Definition \ref{def:ent-relative}), and $\mathcal H_\Lambda(\mu) = -\mathcal H_\Lambda(\mu \mid \lambda^S) \in [-\infty, 0]$.

\begin{proposition}[Georgii (15.10)]
    \label{prop:sent-15-10}
    \uses{def:ent-relative, prop:ent-155}
    \lean{MeasureTheory.GibbsMeasure.relativeEntropyIn, MeasureTheory.GibbsMeasure.relativeEntropyIn_add_relativeEntropyIn_le, MeasureTheory.GibbsMeasure.entropyIn_add_entropyIn_le, MeasureTheory.GibbsMeasure.entropyIn_union_le, MeasureTheory.GibbsMeasure.relativeEntropyIn_image_add, MeasureTheory.GibbsMeasure.entropyIn_image_add}
    \leanok

    For a probability measure $\mu$ on $\Cfg$ and arbitrary $\Lambda, \Delta \subseteq S$,
    \[
        \mathcal H_{\Lambda \cup \Delta}(\mu) + \mathcal H_{\Lambda \cap \Delta}(\mu)
        \leq \mathcal H_\Lambda(\mu) + \mathcal H_\Delta(\mu),
    \]
    and $\mathcal H_{\Lambda + i}(\mu) = \mathcal H_\Lambda(\mu)$ for shift-invariant $\mu$.
\end{proposition}
\begin{proof}
    \leanok
    Chain rule $\mathcal H(\mu \mid \lambda^S) = \mathcal H(\mu \mid g\lambda^S) + \mathcal H_m(\mu \mid \lambda^S)$
    for an $m$-measurable density $g$ with $\mu = g\lambda^S$ on $m$, applied with
    $m = \mathcal F_{\Lambda \cap \Delta}$ and the Markov property of the independent
    $\sigma$-algebras $\mathcal F_{\Lambda \setminus \Delta}$, $\mathcal F_{\Lambda \cap \Delta}$,
    $\mathcal F_{\Delta \setminus \Lambda}$ under $\lambda^S$.
\end{proof}

\begin{theorem}[Specific entropy, Georgii (15.12)--(15.13)]
    \label{thm:sent-15-12}
    \uses{prop:sent-15-10, lem:ent-1511, def:erg-invariant}
    \lean{MeasureTheory.GibbsMeasure.specificEntropy, MeasureTheory.GibbsMeasure.boxSubadditive_entropyIn, MeasureTheory.GibbsMeasure.tendsto_entropyIn_div_card, MeasureTheory.GibbsMeasure.tendsto_entropyIn_div_card_cube, MeasureTheory.GibbsMeasure.specificEntropy_nonpos}
    \leanok

    Let $S = \mathbb Z^d$ and $\mu \in \Prob_\Theta$. Then
    $h(\mu) = \lim_n |\Lambda_n|^{-1}\mathcal H_{\Lambda_n}(\mu) \in [-\infty, 0]$ exists along
    every sequence of boxes all of whose side lengths tend to infinity, and equals the infimum
    of $|\Lambda|^{-1}\mathcal H_\Lambda(\mu)$ over all boxes $\Lambda$.
\end{theorem}
\begin{proof}
    \leanok
    Proposition \ref{prop:sent-15-10} makes $\Lambda \mapsto -\mathcal H_\Lambda(\mu)$
    shift-invariant and box-subadditive; Lemma \ref{lem:ent-1511}.
\end{proof}

\begin{proposition}[Georgii (15.14)(a),(b)]
    \label{prop:sent-15-14}
    \uses{thm:sent-15-12, def:lc-topology}
    \lean{MeasureTheory.GibbsMeasure.smul_specificEntropy_add_smul_specificEntropy_le, MeasureTheory.GibbsMeasure.specificEntropy_smul_add_smul_le, MeasureTheory.GibbsMeasure.upperSemicontinuous_specificEntropy}
    \leanok

    For $d \geq 1$, $h$ is affine on $\Prob_\Theta$, and $\mu \mapsto h(\mu)$ is upper
    semicontinuous on $\Prob(\Cfg)$ in the topology of local convergence.
\end{proposition}
\begin{proof}
    \leanok
    Concavity is that of $\mathcal H_\Lambda$; convexity up to the defect
    $s\log(1/s) + t\log(1/t)$ in each finite volume, which vanishes after division by
    $|\Lambda_n| \to \infty$. Upper semicontinuity: each $\mathcal H_\Lambda(\cdot \mid \lambda^S)$
    is a supremum over finite partitions of continuous functions.
\end{proof}

\begin{proposition}[Georgii (15.14)(c)]
    \label{prop:sent-15-14c}
    \uses{thm:sent-15-12, prop:sent-15-14, prop:cp-49}
    \lean{MeasureTheory.GibbsMeasure.isCompact_setOf_le_specificEntropy, InformationTheory.exists_forall_measure_le_of_klDiv_le}
    \leanok

    Let $E$ be standard Borel and $\lambda$ a probability measure. For every $c \in \mathbb R$
    the set $\{\mu : h(\mu) \geq c\}$ is compact in the topology of local convergence.
\end{proposition}
\begin{proof}
    \leanok
    A bound $\mathcal H(\mu \mid \nu) \leq c$ makes $\mu$ uniformly absolutely continuous with
    respect to $\nu$; on each box this gives local equicontinuity, and Proposition
    \ref{prop:cp-49} yields a cluster point, which lies in the set by upper semicontinuity.
\end{proof}

\begin{theorem}[Georgii (15.16)--(15.18)]
    \label{thm:sent-15-16}
    \uses{thm:sent-15-12, def:ent-relative}
    \lean{MeasureTheory.GibbsMeasure.lexPast, MeasureTheory.GibbsMeasure.specificEntropy_eq_neg_relativeEntropyIn_lexPast, MeasureTheory.GibbsMeasure.specificEntropy_eq_iInf_finset, MeasureTheory.GibbsMeasure.neg_relativeEntropyIn_lexPast_le_specificEntropy, MeasureTheory.GibbsMeasure.specificEntropy_le_neg_relativeEntropyIn_lexPast, MeasureTheory.GibbsMeasure.specificEntropy_eq_neg_lintegral_klDiv_withDensity, MeasureTheory.GibbsMeasure.toReal_lintegral_update_ae_eq_condExp}
    \leanok

    Let $\mu \in \Prob_\Theta$ on $S = \mathbb Z^d$ with the lexicographic order and
    $V(j) = \{i : i \leq j\}$. Then $h(\mu) = -\mathcal H_{V(0)}(\mu \mid \mu\lambda_0)$
    (15.16), $h(\mu) = \inf_\Delta |\Delta|^{-1}\mathcal H_\Delta(\mu)$, and the two
    inequalities of (15.17) hold. If $\mu\lambda_0 \gg \mu$ on $\mathcal F_{V(0)}$ with density
    $g$, then $h(\mu) = -\int \mathcal H(q^\omega \mid \lambda)\,\mu(d\omega)$ with
    $q^\omega = \lambda(g(\omega_{V(0)\setminus 0}\,\cdot))$ (15.18), and
    $q^\omega(A) = \mu(\sigma_0 \in A \mid \mathcal F_{V(0)\setminus\{0\}})(\omega)$ a.s.
\end{theorem}
\begin{proof}
    \leanok
    The chain rule for the relative entropy of a resampling and shift equivariance telescope
    $\mathcal H_\Delta(\mu)$ over the sites of $\Delta$ in lexicographic order; the martingale
    limit of $\mathcal H_{V(0) \cap [-n,n]^d}$ gives (15.17), and (15.18) is the conditional
    form of (15.16).
\end{proof}

\begin{proposition}[Georgii (15.19)]
    \label{prop:sent-15-19}
    \uses{thm:sent-15-16}
    \lean{MeasureTheory.GibbsMeasure.specificEntropy_uniformOn_eq_neg_integral_sum_mul_log, InformationTheory.klDiv_uniformOn_univ}
    \leanok

    For finite $E$ and $\lambda$ the uniform distribution,
    $h(\mu) = -\int \sum_{x} p_x \log p_x\,d\mu - \log|E|$ with
    $p_x = \mu(\sigma_0 = x \mid \mathcal F_{V(0)\setminus\{0\}})$; Georgii's counting-measure
    normalisation differs by the constant $\log|E|$.
\end{proposition}
\begin{proof}
    \leanok
    Shannon's formula for the relative entropy with respect to the uniform distribution, in (15.18).
\end{proof}

\begin{theorem}[Georgii (15.20)]
    \label{thm:sent-15-20}
    \uses{thm:sent-15-16, thm:erg-1410}
    \lean{MeasureTheory.GibbsMeasure.exists_measurable_specificEntropy_eq_neg_lintegral, MeasureTheory.GibbsMeasure.specificEntropy_join_eq_neg_lintegral, MeasureTheory.GibbsMeasure.tailErgodicKernel, MeasureTheory.GibbsMeasure.marginalErgodicKernel, MeasureTheory.GibbsMeasure.exists_isPAKernel_invariantFields_shiftGroup_int_inf_tail}
    \leanok

    Let $E$ be standard Borel and $d \geq 1$. There is an
    $\mathcal I \cap \mathcal T$-measurable $h_0 \leq 0$ with $h(\mu) = \int h_0\,d\mu$ for all
    $\mu \in \Prob_\Theta$, and $h(\mu) = \int h(\nu)\,w_\mu(d\nu)$ for the ergodic decomposition
    $w_\mu$ of Theorem \ref{thm:erg-1410}.
\end{theorem}
\begin{proof}
    \leanok
    $h_0 = h(\pi(\cdot \mid \omega))$ for a $(\Prob_\Theta, \mathcal I)$-kernel $\pi$ that is
    moreover $\mathcal I \cap \mathcal T$-measurable: the ergodic averages of the indicators of
    cylinder events over a finite volume are tail measurable, so the marginal kernels are, and
    Kolmogorov extension reassembles them. (This fails for a finite site group, where
    $\mathcal T$ is trivial, and the hypothesis $\abs{F_n} \to \infty$ excludes that.) The
    integral identity is affinity of $h$ under the decomposition.
\end{proof}
