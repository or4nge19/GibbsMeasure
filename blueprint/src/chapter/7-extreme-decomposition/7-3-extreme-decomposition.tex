\section{The extreme decomposition of a Gibbs simplex}

Throughout, $\gamma$ is a specification with countable parameter set $S$ and state space
$(E, \mathcal{E})$, and $\Tail$ is the tail $\sigma$-algebra of $\Cfg = E^S$.

\begin{theorem}[Georgii, Theorem (7.7)]
    \label{thm:ext-77-concrete}
    \uses{def:spec, def:gibbs-meas, def:ql-tail}
    \lean{MeasureTheory.GibbsMeasure.not_mem_extremePoints_G_of_tail_prob,
      MeasureTheory.GibbsMeasure.tailTrivial_of_mem_extremePoints_G,
      MeasureTheory.GibbsMeasure.mem_extremePoints_G_of_isTailTrivial,
      MeasureTheory.GibbsMeasure.eq_of_forall_measurableSet_tail_eq,
      MeasureTheory.GibbsMeasure.exists_tail_eq_one_eq_zero_of_mem_extremePoints,
      MeasureTheory.GibbsMeasure.mutuallySingular_of_mem_extremePoints}
    \leanok

    No assumption is made on $(E, \mathcal{E})$ or on $\mathcal{G}(\gamma) \neq \emptyset$.
    \begin{enumerate}
        \item[(a)] $\mu \in \mathcal{G}(\gamma)$ is extreme iff $\mu$ is trivial on $\Tail$.
        \item[(c)] Two Gibbs measures agreeing on every tail event are equal.
        \item[(d)] Distinct extreme $\mu, \nu \in \mathcal{G}(\gamma)$ are separated by a tail
          event $A$ with $\mu(A) = 1$ and $\nu(A) = 0$; in particular $\mu \perp \nu$.
    \end{enumerate}
    (c) follows from (b): for $\rho = \tfrac12(\mu + \nu) \in \mathcal{G}(\gamma)$, $\mu$ and
    $\nu$ have $\Tail$-measurable densities with respect to $\rho$, which coincide once $\mu$ and
    $\nu$ agree on $\Tail$.
\end{theorem}

\begin{proposition}[The Gibbs kernel, Georgii (7.25)]
    \label{prop:ext-gibbs-kernel}
    \uses{thm:ext-77-concrete}
    \lean{MeasureTheory.GibbsMeasure.gibbsKernel,
      MeasureTheory.GibbsMeasure.gibbsKernel_mem_G,
      MeasureTheory.GibbsMeasure.condExp_ae_eq_gibbsKernel,
      MeasureTheory.GibbsMeasure.isPAKernel_gibbsKernel,
      MeasureTheory.GibbsMeasure.ae_mem_extremePoints_G_gibbsKernel}
    \leanok

    Let $(E, \mathcal{E})$ be standard Borel and $\mathcal{G}(\gamma) \neq \emptyset$. There is a
    $\Tail$-measurable probability kernel $\omega \mapsto \pi(\cdot \mid \omega)$ with
    $\pi(\cdot \mid \omega) \in \mathcal{G}(\gamma)$ for every $\omega$ which is a version of
    $\mu(\cdot \mid \Tail)$ for every $\mu \in \mathcal{G}(\gamma)$ simultaneously (a
    $(\mathcal{G}(\gamma), \Tail)$-kernel), and for every $\mu \in \mathcal{G}(\gamma)$,
    $\pi(\cdot \mid \omega) \in \operatorname{ex}\mathcal{G}(\gamma)$ for $\mu$-a.e.\ $\omega$.
\end{proposition}

\begin{theorem}[The extreme decomposition, Georgii (7.26)]
    \label{thm:ext-726}
    \uses{prop:ext-gibbs-kernel}
    \lean{MeasureTheory.GibbsMeasure.weightOf,
      MeasureTheory.GibbsMeasure.exists_unique_weight_extremePoints,
      MeasureTheory.GibbsMeasure.eq_weightOf_of_join_eq,
      MeasureTheory.GibbsMeasure.weightOf_join,
      MeasureTheory.GibbsMeasure.bijOn_weightOf}
    \leanok

    Let $(E, \mathcal{E})$ be standard Borel and $\mathcal{G}(\gamma) \neq \emptyset$. Every
    $\mu \in \mathcal{G}(\gamma)$ is the barycentre $\mu = \int \nu\, w_\mu(\mathrm{d}\nu)$ of a
    unique probability weight $w_\mu$ concentrated on $\operatorname{ex}\mathcal{G}(\gamma)$,
    namely the law under $\mu$ of $\omega \mapsto \pi(\cdot \mid \omega)$ for the kernel $\pi$ of
    Proposition \ref{prop:ext-gibbs-kernel}; and $\mu \mapsto w_\mu$ is a bijection from
    $\mathcal{G}(\gamma)$ onto the probability weights concentrated on
    $\operatorname{ex}\mathcal{G}(\gamma)$ (the Gibbs simplex).
\end{theorem}

\begin{corollary}[Georgii (7.28)--(7.30)]
    \label{cor:ext-728-730}
    \uses{thm:ext-726}
    \lean{MeasureTheory.GibbsMeasure.weightOf_map,
      MeasureTheory.GibbsMeasure.map_eq_self_iff_weightOf_map_eq_self,
      MeasureTheory.GibbsMeasure.measurePreserving_iff_measurePreserving_weightOf,
      MeasureTheory.GibbsMeasure.linearIndependent_of_mem_extremePoints,
      MeasureTheory.GibbsMeasure.le_encard_extremePoints_iff,
      MeasureTheory.GibbsMeasure.setOf_mem_GP_eq_closure_convexCombosLimitGibbs,
      MeasureTheory.GibbsMeasure.setOf_mem_GP_eq_closure_convexCombosLimitGibbs_modification_isssd,
      MeasureTheory.GibbsMeasure.setOf_mem_GP_eq_closure_convexCombosLimitGibbs_lambdaSpecification}
    \leanok

    Let $(E, \mathcal{E})$ be standard Borel.
    \begin{enumerate}
        \item[(7.28)] If $\mathcal{G}(\gamma) \neq \emptyset$ and $\tau$ is a symmetry of
          $\gamma$, then $w_{\tau\mu} = \tau_* w_\mu$ for every $\mu \in \mathcal{G}(\gamma)$,
          where $\tau_*$ is the push-forward along $\nu \mapsto \tau\nu$; in particular
          $\tau\mu = \mu$ iff $\tau_* w_\mu = w_\mu$.
        \item[(7.29)] If $\mathcal{G}(\gamma) \neq \emptyset$ and $N \in \mathbb{N}$, then
          $|\operatorname{ex}\mathcal{G}(\gamma)| \geq N$ iff $\mathcal{G}(\gamma)$ contains $N$
          measures linearly independent over $[0, \infty]$.
        \item[(7.30)] If $\gamma$ is a quasilocal $\lambda$-specification for a $\sigma$-finite
          non-zero a priori measure $\lambda$, then $\mathcal{G}(\gamma)$ is the closure, in the
          topology of local convergence, of the convex hull of the limiting Gibbs distributions;
          the same holds for every quasilocal $\gamma$ when $E$ is finite.
    \end{enumerate}
\end{corollary}
