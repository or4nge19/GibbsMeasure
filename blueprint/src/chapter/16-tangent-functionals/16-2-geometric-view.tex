\section{A geometric view of Gibbs measures}

\begin{definition}[Local potentials, Georgii (16.10)]
    \label{def:gv-16-10}
    \uses{def:tf-btheta, def:pr-energy-density}
    \lean{Potential.localPotential, Potential.normAt_localPotential, Potential.specificEnergy_localPotential, Potential.BTheta.specificEnergyFunctional, Potential.BTheta.specificEnergyFunctional_injective}
    \leanok

    For a finite volume $\Lambda$ and a bounded local $f$, the shift-invariant potential
    $\Phi^{\Lambda,f}$ obtained by spreading $f$ over the translates of $\Lambda$ satisfies
    $\norm{\Phi^{\Lambda,f}} = \abs{\Lambda}\norm{f}$ and $\langle\mu,\Phi^{\Lambda,f}\rangle =
    \mu(f)$ for $\mu \in \Prob_\Theta$. Hence $j : \mu \mapsto (\Psi \mapsto -\langle\mu,\Psi\rangle)$
    is injective on $\Prob_\Theta$.
\end{definition}

\begin{theorem}[Georgii (16.11)]
    \label{thm:gv-16-11}
    \uses{def:gv-16-10, thm:vp-15-39, prop:sent-15-14c, thm:pr-15-30}
    \lean{Potential.BTheta.specificEntropyDual_eq_iInf, Potential.BTheta.isCompact_setOf_le_specificEntropyDual}
    \leanok

    For $E$ standard Borel and every linear $L$ on $\mathcal B_\Theta$,
    $h^*(L) = \inf_\Phi (P(\Phi) - L(\Phi))$, where $h^*$ is the specific entropy transported
    along $j$.
\end{theorem}
\begin{proof}
    \leanok
    The hypograph of $h^*$ is nonempty, convex and closed in the weak topology, since the level
    sets $\{h \geq c\}$ are compact and $j$ is continuous; separate a point from it by
    Hahn--Banach and represent the separating functional by the weak representation theorem. The
    other inequality is Corollary (15.35), with attainment on $\mathcal G_\Theta(\Phi)$.
\end{proof}

\begin{corollary}[Georgii (16.12), (16.13)]
    \label{cor:gv-16-12}
    \uses{thm:gv-16-11, def:tf-16-4}
    \lean{Potential.BTheta.specificEntropy_eq_iInf, Potential.BTheta.isBoundedBy_pressure_iff, Potential.BTheta.fieldsOf}
    \leanok

    $h(\mu) = \inf_\Phi (\langle\mu,\Phi\rangle + P(\Phi))$ for $\mu \in \Prob_\Theta$, and a
    linear $L$ is bounded by $P$ iff $L = j(\mu)$ for some $\mu \in \Prob_\Theta$ with
    $h(\mu) > -\infty$; such a $\mu$ is unique.
\end{corollary}

\begin{theorem}[Georgii (16.14)]
    \label{thm:gv-16-14}
    \uses{thm:gv-16-11, thm:tf-16-6, thm:vp-15-39}
    \lean{Potential.BTheta.mem_subgradientAt_pressure_of_mem_invariantG, Potential.BTheta.mem_subgradientAt_pressure_iff_mem_invariantG', Potential.BTheta.bijOn_specificEnergyFunctional_invariantG, Potential.BTheta.image_specificEnergyFunctional_invariantG, Potential.BTheta.subsingleton_invariantG_of_gateauxDifferentiable, Potential.BTheta.gateauxDifferentiable_pressure_iff_subsingleton}
    \leanok

    For $E$ standard Borel, $j$ maps $\mathcal G_\Theta(\Phi)$ bijectively onto the tangent
    functionals of $P$ at $\Phi$. Consequently $P$ is Gateaux differentiable at $\Phi$ iff
    $\mathcal G_\Theta(\Phi)$ is a singleton, and by Theorem \ref{thm:tf-16-3} that happens on a
    dense $G_\delta$ when $E$ is finite.
\end{theorem}
\begin{proof}
    \leanok
    Tangency of $j(\mu)$ at $\Phi$ says exactly that $\mu$ minimises the specific free energy,
    which by the variational principle characterises $\mathcal G_\Theta(\Phi)$; surjectivity is
    Theorem \ref{thm:gv-16-11} with the attainment of the infimum.
\end{proof}

\begin{theorem}[Georgii (16.15), (16.16)]
    \label{thm:gv-16-15}
    \uses{thm:gv-16-14, thm:equiv-tfae}
    \lean{Potential.BTheta.isEquivalent_iff_forall_pressure_smul_add_smul_eq, Potential.BTheta.pressure_smul_add_smul_lt_of_isNormalized, Potential.BTheta.isEquivalent_of_specificEnergy_sub_eq}
    \leanok

    Assume the a priori measure is everywhere dense and the Hamiltonians of $\Phi - \Psi$ are
    continuous. (a) $\Phi \sim \Psi$ iff $P$ is affine on the segment from $\Phi$ to $\Psi$.
    (b) If $\Phi$ and $\Psi$ are normalised and distinct, $P$ is strictly convex on that segment. (16.16): equality
    of specific energies on $\mathcal G_\Theta$ forces equivalence.
\end{theorem}
\begin{proof}
    \leanok
    Affineness at one interior point gives a common tangent functional, hence by Theorem
    \ref{thm:gv-16-14} a common Gibbs measure, hence equivalence by Theorem \ref{thm:equiv-tfae};
    normalisation upgrades equivalence to equality.
\end{proof}

\begin{theorem}[Georgii (16.17), the two derivatives]
    \label{thm:gv-16-17}
    \uses{thm:gv-16-14, thm:dob-uniqueness, cor:dob-8-37}
    \lean{MeasureTheory.GibbsMeasure.Dobrushin.cardNormAt, MeasureTheory.GibbsMeasure.Dobrushin.isDobrushin_gibbsSpecification_of_cardNormAt_le, Potential.BTheta.dobrushinRegion, Potential.BTheta.exists_invariantG_eq_singleton_of_mem_dobrushinRegion, Potential.BTheta.leftDirDeriv_eq_and_rightDirDeriv_eq_pressure_of_mem_dobrushinRegion, Potential.BTheta.hasDerivAt_rightDirDeriv_pressure_of_mem_dobrushinRegion, ENNReal.exists_pos_add_ofReal_mul_lt_one}
    \leanok

    On the Dobrushin region $\mathcal D = \{\Phi : \sum_{A \ni 0}\abs{A}\norm{\Phi_A} < 1\}$ the
    shift-invariant Gibbs measure is unique and $P$ is Gateaux differentiable, with
    $\frac{d}{dt}P(\Phi + t\Psi)\big|_{t=0} = -\langle\mu_\Phi,\Psi\rangle$; and for
    $\Psi$ of finite $(8.36)$-norm and any $\tilde\Psi \in \mathcal B_\Theta$,
    $t \mapsto \frac{\partial}{\partial s}P(\Phi + t\Psi + s\tilde\Psi)\big|_{s=0}$ is
    differentiable at $0$ with derivative
    $\sum_i [\mu_\Phi(f_{\tilde\Psi}\,f_\Psi\circ\theta_i) - \mu_\Phi(f_{\tilde\Psi})\mu_\Phi(f_\Psi)]$.
    Only the outer direction needs the $(8.36)$-norm, which keeps the segment inside
    $\mathcal D$; Georgii asks it of both. The continuity of the second derivative in his
    ``twice continuously differentiable'' is not proved.
\end{theorem}
\begin{proof}
    \leanok
    Georgii (8.36): the norm of $\widetilde{\mathcal B}_\Theta$ dominates the interaction strength,
    so $\mathcal D$ lies in Dobrushin's regime (8.8) and uniqueness is (8.7); differentiability is
    Theorem \ref{thm:gv-16-14} with a single tangent functional. For the second derivative, a
    $t_0 > 0$ with $\Phi + t\Psi \in \mathcal D$ for $|t| \leq t_0$ gives the family of unique
    phases, the inner derivative is $-\mu_{\Phi+t\Psi}(f_{\tilde\Psi})$, and Corollary
    \ref{cor:dob-8-37} with $g = f_{\tilde\Psi}$ differentiates it.
\end{proof}

\begin{example}[The Ising potential as a witness]
    \label{ex:gv-ising-witness}
    \uses{thm:gv-16-17, prop:ld-15-49, cor:dob-8-37}
    \lean{MeasureTheory.GibbsMeasure.isingPotential_mem_BTheta, MeasureTheory.GibbsMeasure.isingBTheta, MeasureTheory.GibbsMeasure.isingBTheta_ne_zero, MeasureTheory.GibbsMeasure.card_neighborFinset_latticeGraph, MeasureTheory.GibbsMeasure.cardNormAt_isingPotential, MeasureTheory.GibbsMeasure.cardNormAt_isingPotential_latticeGraph, MeasureTheory.GibbsMeasure.cardNormAt_isingPotential_of_zero_coupling, MeasureTheory.GibbsMeasure.isingBTheta_mem_dobrushinRegion, MeasureTheory.GibbsMeasure.isDobrushin_isingSpecification_of_cardNormAt_lt_one, MeasureTheory.GibbsMeasure.exists_ne_zero_mem_dobrushinRegion, MeasureTheory.GibbsMeasure.exists_ldRate_isingBTheta_eq_zero, MeasureTheory.GibbsMeasure.isingPotential_add_smul_field, MeasureTheory.GibbsMeasure.leftDirDeriv_eq_and_rightDirDeriv_eq_pressure_isingBTheta_field, MeasureTheory.GibbsMeasure.hasDerivAt_rightDirDeriv_pressure_isingBTheta_field, MeasureTheory.GibbsMeasure.exists_hasDerivAt_integral_isingGibbsMeasure_field}
    \leanok

    The $\mathbb Z^d$ Ising potential lies in $\mathcal B_\Theta$ and is nonzero when $h \neq 0$;
    Georgii's norm (8.36) is exactly $\abs h + 2\deg(i)\abs J$ at every site of a locally finite
    graph, so $\abs h + 4d\abs J$ on $\mathbb Z^d$, and for $\abs h + 4d\abs J < 1$ the potential
    lies in the Dobrushin region (a criterion strictly smaller than (8.8)'s $2d\abs J < 1$, which
    ignores the field). There Theorem \ref{thm:gv-16-17} reads $\partial P/\partial h =
    \mu(\sigma_0)$, the magnetisation, and $\partial^2 P/\partial h^2 = \sum_k
    \operatorname{cov}_\mu(\sigma_0, \sigma_k)$, the susceptibility, and Corollary
    \ref{cor:dob-8-37} is the fluctuation--response identity with the Gibbs family constructed;
    the rate function vanishes somewhere. None of these objects is vacuous.
\end{example}
