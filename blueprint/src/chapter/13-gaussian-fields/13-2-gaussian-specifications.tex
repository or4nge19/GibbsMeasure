\section{Gibbs measures for Gaussian specifications}

\begin{theorem}[Georgii (13.20)]
    \label{thm:gs-13-20}
    \uses{thm:gf-13-7, lem:gf-13-10, def:gf-13-11}
    \lean{MeasureTheory.GibbsMeasure.georgii_13_20, Specification.isGibbsMeasure_of_forall_singleton_condExp_ae_eq, Specification.bind_eq_of_condExp_ae_eq}
    \leanok

    Let $\mu$ be a Gauss field with $\Gamma(j,j) > 0$ for all $j$ and each $\xi_j^\mu$ measurable
    with respect to a neighbourhood $N(j) \not\ni j$. Then the conditional couplings $J$ and
    external field $h$ of Theorem \ref{thm:gf-13-7} define a Gaussian specification and
    $\mu \in \mathcal G(\gamma^{J,h})$.
\end{theorem}
\begin{proof}
    \leanok
    (13.7) gives $J$ symmetric with finite rows and positive definite, (13.10) identifies the
    one-site conditional distributions with $\gamma^{J,h}_{\{i\}}$, and a conditional expectation
    identity at every singleton is the DLR equation by (1.33).
\end{proof}

\begin{theorem}[Georgii (13.22)]
    \label{thm:gs-13-22}
    \uses{thm:gs-13-20, def:gf-13-18}
    \lean{MeasureTheory.GibbsMeasure.georgii_13_22_of_finiteRowSupport, MeasureTheory.GibbsMeasure.georgii_13_22_iff, MeasureTheory.GibbsMeasure.isInverse_covariance_of_isGibbsMeasure, MeasureTheory.GibbsMeasure.condExpOutside_ae_eq_gaussianCondMean_of_isGibbsMeasure, ProbabilityTheory.integral_eval_sq_multivariateGaussianPi}
    \leanok

    Let $J$ be symmetric of finite row support with every $\mathcal J_\Lambda$ positive definite.
    A Gauss field $\mu$ is a Gibbs measure for $\gamma^{J,h}$ if and only if its mean lies in
    $M_{J,h}$ and its covariance $C$ satisfies $\sum_j J(i,j)C(j,k) = \delta_{ik}$.
\end{theorem}
\begin{proof}
    \leanok
    The row forms $X_i = \sum_j J(i,j)\sigma_j$ are jointly Gaussian with the field, uncorrelated
    with every coordinate outside $\{i\}$, so the residual is independent of the exterior; this is
    (13.5), and Theorem \ref{thm:gs-13-20} applies.
\end{proof}

\begin{theorem}[Georgii (13.23)]
    \label{thm:gs-13-23}
    \uses{def:gf-13-18, rmk:sym-transport-gibbs}
    \lean{MeasureTheory.GibbsMeasure.spinTranslation, Potential.gaussianSpecification_map_spinTranslation, Potential.gaussianSpecification_G_eq_image_of_mem_gaussianMeanSet, Potential.isInvariant_spinTranslation_gaussianSpecification, Potential.map_add_isGibbsMeasure_of_mem_gaussianMeanSubmodule}
    \leanok

    For $m \in M_{J,h}$ the translation $\tau^m$ maps $\gamma^{J,h'}$ to $\gamma^{J,h+h'}$, hence
    $\mathcal G(\gamma^{J,h})$ is the image of $\mathcal G(\gamma^{J,0})$ under $\mu \mapsto \mu(\cdot - m)$;
    for $m \in M_{J,0}$ the translation is a symmetry of $\gamma^{J,h}$ and preserves
    $\mathcal G(\gamma^{J,h})$.
\end{theorem}

\begin{proposition}[Georgii (13.26), (13.31), the closing step]
    \label{prop:gs-13-26}
    \uses{thm:gs-13-22, thm:gs-13-23}
    \lean{MeasureTheory.GibbsMeasure.isGibbsMeasure_map_add_of_centered_of_isInverse, MeasureTheory.GibbsMeasure.nonempty_G_gaussianSpecification_of_centered_of_isInverse}
    \leanok

    If $\mu_C$ is a centred Gauss field whose covariance inverts $J$, then
    $\tau^m(\mu_C) \in \mathcal G(\gamma^{J,h})$ for every $m \in M_{J,h}$; in particular
    $\mathcal G(\gamma^{J,h}) \neq \emptyset$.
\end{proposition}

\begin{theorem}[Georgii (13.A7)]
    \label{thm:gs-13-A7}
    \lean{ProbabilityTheory.covMatrix, ProbabilityTheory.gaussianField, ProbabilityTheory.exists_isGaussianProcess_covariance_eq, ProbabilityTheory.existsUnique_isGaussianProcess_covariance_eq, ProbabilityTheory.covariance_eval_gaussianField}
    \leanok

    For a symmetric $C : \iota \times \iota \to \mathbb R$ all of whose finite submatrices are
    nonnegative definite there is a unique centred Gaussian process on $\mathbb R^\iota$ with
    covariance $C$.
\end{theorem}
\begin{proof}
    \leanok
    The finite-dimensional marginals are the multivariate Gaussians of the submatrices, which are
    consistent under restriction; Kolmogorov extension glues them. Degeneracy is harmless because
    a multivariate Gaussian is the image of a standard one under a square root of its covariance.
\end{proof}

\begin{corollary}[Georgii (13.26), existence]
    \label{cor:gs-13-26}
    \uses{prop:gs-13-26, thm:gs-13-A7}
    \lean{MeasureTheory.GibbsMeasure.nonempty_G_gaussianSpecification_of_posSemidef_of_isInverse}
    \leanok

    If $J$ is symmetric with finite rows, every $\mathcal J_\Lambda$ positive definite, and $C$ is
    nonnegative definite with $\sum_j J(i,j)C(j,k) = \delta_{ik}$, then
    $\mathcal G(\gamma^{J,h}) \neq \emptyset$ for every $h$ with $M_{J,h} \neq \emptyset$.
\end{corollary}

\begin{theorem}[Georgii (13.26), (13.27)]
    \label{thm:gs-13-27}
    \uses{thm:gs-13-22, thm:gs-13-A7}
    \lean{Matrix.PosDef.dotProduct_mulVec_inv_submatrix_le, MeasureTheory.GibbsMeasure.dotProduct_mulVec_inv_gaussianCouplingMatrix_mono, MeasureTheory.GibbsMeasure.exists_tendsto_gaussianCovEntry, MeasureTheory.GibbsMeasure.isInverse_of_tendsto_gaussianCovEntry, MeasureTheory.GibbsMeasure.nonempty_G_gaussianSpecification_of_bddAbove}
    \leanok

    If $\sup_\Lambda \mathcal J_\Lambda^{-1}(i,i) < \infty$ for every $i$ (13.27) and
    $M_{J,h} \neq \emptyset$, then the entries $\mathcal J_\Lambda^{-1}(i,j)$ converge as
    $\Lambda \uparrow S$ to a nonnegative-definite $C$ inverting $J$, and
    $\mathcal G(\gamma^{J,h}) \neq \emptyset$.
\end{theorem}
\begin{proof}
    \leanok
    The monotonicity $(A_{\Lambda\Lambda})^{-1} \preceq (A^{-1})_{\Lambda\Lambda}$ is linear
    algebra, from $2t\cdot x - x\cdot Ax \leq t\cdot A^{-1}t$ with equality at $x = A^{-1}t$;
    Georgii derives it from (13.13) and Jensen. Monotone bounded convergence then gives $C$, and
    Theorem \ref{thm:gs-13-A7} supplies the field.
\end{proof}

\begin{proposition}[Georgii §13.3, the homogeneous case]
    \label{prop:gs-13-3}
    \uses{thm:gs-13-23, thm:gs-13-27}
    \lean{Potential.homogeneousCoupling, Potential.const_mem_gaussianMeanSet_iff, Potential.map_add_const_isGibbsMeasure_of_tsum_eq_zero, Potential.nonempty_G_homogeneousCoupling_of_bddAbove}
    \leanok

    For a translation-invariant coupling $J(i,j) = J'(j-i)$ on a countable abelian group, a
    constant configuration lies in $M_{J,h}$ iff $h' + \hat J(1)c = 0$; when $\hat J(1) = 0$ the
    constant shifts are symmetries of $\gamma^{J,h}$ and preserve $\mathcal G$.
\end{proposition}
