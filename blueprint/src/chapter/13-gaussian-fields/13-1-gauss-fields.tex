\section{Gauss fields as Gibbs measures}

Throughout, $E = \mathbb R$ with Lebesgue measure and $S$ is countable.

\begin{definition}[Gauss field, Georgii (13.1)--(13.3)]
    \label{def:gf-13-1}
    \lean{MeasureTheory.GibbsMeasure.integral_cexp_I_sum_eq_of_isGaussianProcess, MeasureTheory.GibbsMeasure.posSemidef_covar_of_isGaussianProcess, Matrix.posSemidef_covariance}
    \leanok

    A probability measure $\mu$ on $\mathbb R^S$ is a \textbf{Gauss field} if the coordinate
    process is a Gaussian process (every finite-dimensional marginal is Gaussian). Then
    $\mu\bigl(e^{i\sum_{j\in\Lambda} t_j\sigma_j}\bigr) = \exp\bigl(i\sum_j t_j m_j - \tfrac12\sum_{j,k} t_j t_k c_{jk}\bigr)$
    with $m$ the mean and $c$ the covariance function (13.2), and $c$ is nonnegative definite (13.3).
\end{definition}

\begin{theorem}[Georgii (13.4)--(13.8): the conditional structure of a Gauss field]
    \label{thm:gf-13-7}
    \uses{def:gf-13-1}
    \lean{MeasureTheory.GibbsMeasure.condExpOutside, MeasureTheory.GibbsMeasure.condCovariance, MeasureTheory.GibbsMeasure.georgii_13_7, MeasureTheory.GibbsMeasure.exists_condExpOutside_ae_eq_affine}
    \leanok

    Let $\mu$ be a Gauss field on $S$ countable, $\xi_i^\mu = \mu(\sigma_i \mid \mathcal T_{\{i\}})$
    (13.4) and $\Gamma(i,j) = \mu((\sigma_i - \xi_i^\mu)(\sigma_j - \xi_j^\mu))$ (13.6). If
    $\Gamma(j,j) > 0$ for all $j$ and each $\xi_i^\mu$ is $\mathcal F_{N(i)}$-measurable for a
    neighbourhood $N(i) \not\ni i$, then with $J(i,j) = -\Gamma(i,j)/(\Gamma(i,i)\Gamma(j,j))$-type
    conditional couplings and $h$ the conditional external field: $J(i,j) = 0$ for
    $j \notin N(i)$, $J$ is positive definite, and $\sigma_i - \xi_i^\mu = J(i,i)^{-1}(h_i + \sum_j J(i,j)\sigma_j)$
    a.s.\ (13.5); (13.8): $\xi_i^\mu = u_i - \sum_{j \in \partial i}\Gamma(i,j)\Gamma(j,j)^{-1}\sigma_j$ a.s.
\end{theorem}
\begin{proof}
    \leanok
    Gaussian regression: the conditional expectation of a Gaussian family on a sub-family is
    affine, and the residual is independent of the conditioning $\sigma$-algebra.
\end{proof}

\begin{lemma}[Georgii (13.10)]
    \label{lem:gf-13-10}
    \uses{thm:gf-13-7, def:gf-13-11}
    \lean{MeasureTheory.GibbsMeasure.georgii_13_10, MeasureTheory.GibbsMeasure.condExp_indicator_ae_eq_gaussianReal_map_update, MeasureTheory.GibbsMeasure.condExp_indicator_ae_eq_gaussianSpecification_singleton}
    \leanok

    Let $J$ be positive definite and $\mu$ a Gauss field with (i) $\mu(\Omega_J) = 1$,
    (ii) $\mu((\sigma_i - \xi_i^\mu)^2) = J(i,i)^{-1}$ and (iii) (13.5) for all $i$. Then the
    conditional distribution of $\sigma_i$ given $\mathcal T_{\{i\}}$ is the Gaussian with mean
    $\xi_i = -J(i,i)^{-1}(h_i + \sum_{j\neq i}J(i,j)\sigma_j)$ and variance $J(i,i)^{-1}$; for $J$
    of finite row support this is $\gamma^{J,h}_{\{i\}}(\cdot \mid \omega)$.
\end{lemma}
\begin{proof}
    \leanok
    For every Gauss field the conditional law of $\sigma_i$ given $\mathcal T_{\{i\}}$ is
    $\mathcal N(\xi_i^\mu, \Gamma(i,i))$, by independence of the residual; (ii) and (iii)
    identify mean and variance.
\end{proof}

\begin{definition}[Gaussian potential, Georgii (13.11)--(13.12)]
    \label{def:gf-13-11}
    \uses{def:pot-potential, def:sb-pair-potential}
    \lean{Potential.gaussianPotential, Potential.isPotential_gaussianPotential, Potential.gaussianCovMatrix}
    \leanok

    For a symmetric $J \colon S \times S \to \mathbb R$ and $h \colon S \to \mathbb R$, the
    \textbf{Gaussian potential} has $\Phi_{\set i}(\sigma) = \tfrac12 J(i,i)\sigma_i^2 - h_i\sigma_i$
    and $\Phi_{\set{i,j}}(\sigma) = J(i,j)\sigma_i\sigma_j$ for $i \neq j$, and $\Phi_A = 0$
    otherwise; $\mathcal J_\Lambda = (J(i,j))_{i,j\in\Lambda}$ is its finite-volume matrix.
\end{definition}

\begin{theorem}[Georgii (13.13)]
    \label{thm:gf-13-13}
    \uses{def:gf-13-11, def:pot-gibbs-spec}
    \lean{Potential.hamiltonian_gaussianPotential_juxt_eq, Potential.isSigmaFiniteLambdaAdmissible_gaussianPotential_boltzmannFactor, Potential.gaussianSpecification, Potential.gaussianMean, Potential.gaussianSpecification_apply}
    \leanok

    Let $J$ be symmetric with finitely many nonzero entries in each row and every
    $\mathcal J_\Lambda$ positive definite, and $\beta > 0$. Then $\beta\Phi^{J,h}$ is
    $\lambda$-admissible, $H_\Lambda(\sigma_\Lambda\omega_{\Lambda^c}) = \tfrac12 \sigma_\Lambda \cdot \mathcal J_\Lambda\sigma_\Lambda + (h_\Lambda + J_{\Lambda\Lambda^c}\omega_{\Lambda^c})\cdot\sigma_\Lambda$,
    and $\gamma^{\beta\Phi}_\Lambda(\cdot \mid \omega)$ is the image under
    $\sigma_\Lambda \mapsto \sigma_\Lambda\omega_{\Lambda^c}$ of the Gaussian measure on
    $\mathbb R^\Lambda$ with precision matrix $\beta\mathcal J_\Lambda$ and mean
    $-\mathcal J_\Lambda^{-1}(h_\Lambda + J_{\Lambda\Lambda^c}\omega_{\Lambda^c})$.
\end{theorem}
\begin{proof}
    \leanok
    The Hamiltonian is the quadratic form above; completing the square and the
    $n$-dimensional Gaussian integral give the partition function and the density of
    $\gamma_\Lambda$ against $\lambda_\Lambda$, which is the Gaussian density.
\end{proof}

\begin{definition}[Georgii (13.9), (13.18), (13.21)]
    \label{def:gf-13-18}
    \uses{thm:gf-13-13, def:ql-tail}
    \lean{Potential.gaussianConvergenceSet, Potential.measurableSet_tail_gaussianConvergenceSet, Specification.zeroDirac, Potential.gaussianGluedSpecification, Potential.gaussianGluedSpecification_eq_gaussianSpecification, Potential.gaussianMeanSet, Potential.gaussianMeanSubmodule}
    \leanok

    $\Omega_J = \set{\omega : \sum_j |J(i,j)\omega_j| < \infty \ \forall i}$ is a tail-measurable
    linear subspace (13.9), all of $\Omega$ for $J$ of finite range. The Gaussian specification
    $\gamma^{J,h}$ (13.18) is the specification $\gamma^{\beta\Phi}$ on $\Omega_J$ glued with the
    Dirac specification at $0_\Lambda\omega_{\Lambda^c}$ off $\Omega_J$; for finite-range $J$ it is
    $\gamma^{\beta\Phi}$. $M_{J,h} = \set{m : h + Jm = 0}$ (13.21) is a coset of the subspace $M_{J,0}$.
\end{definition}
