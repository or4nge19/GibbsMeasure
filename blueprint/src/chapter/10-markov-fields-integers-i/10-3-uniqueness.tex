\section{Uniqueness of the shift-invariant Markov field}

Throughout, $\lambda$ is a probability measure on $E$ and $\gamma = \rho\lambda$ a Markov
specification on $\mathbb Z$ with Markovian densities $\rho$ (Definition \ref{def:mfi-markov-spec});
$\rho$ is \emph{homogeneous} if $\rho_{\Lambda + 1} = \rho_\Lambda \circ \theta_{-1}$.

\begin{definition}[Irreducibility, Georgii (10.23)]
    \label{def:mfu-irreducible}
    \uses{def:mfi-markov-spec}
    \lean{Specification.IsHomogeneousInt, Specification.IsIrreducibleInt}
    \leanok

    A homogeneous Markovian $\rho$ is \textbf{irreducible} if there are sets
    $C_N \uparrow E$, integers $n(N) \geq 1$ and measurable $h_N \geq 0$ on $E$ with
    $\lambda(h_N) > 0$ for large $N$, such that
    $\rho^0_{]-n(N), n(N)[}(\omega) \geq h_N(\omega_0)$ whenever
    $\omega_{-n(N)}, \omega_{n(N)} \in C_N$.
\end{definition}

\begin{example}[Georgii (10.24)]
    \label{ex:mfu-10-24}
    \uses{def:mfu-irreducible, def:pot-nearest-neighbour}
    \lean{MeasureTheory.GibbsMeasure.Markov.isIrreducibleInt_premodifierNorm_boltzmannFactor_of_isNearestNeighbour, MeasureTheory.GibbsMeasure.Markov.isIrreducibleInt_gibbsModifier_of_isNearestNeighbour, MeasureTheory.GibbsMeasure.Markov.isIrreducibleInt_premodifierNorm_boltzmannFactor_markovPotential, MeasureTheory.GibbsMeasure.Markov.isIrreducibleInt_rescaledTransferDensity_of_eventually, MeasureTheory.GibbsMeasure.Markov.isIrreducibleInt_rescaledTransferDensity_of_isAperiodic, ProbabilityTheory.Kernel.eventually_pow_apply_singleton_pos}
    \leanok

    \begin{enumerate}
    \item Let $\Phi$ be a shift-invariant nearest-neighbour potential on $\mathbb Z$ with
      $c = \sup_\omega Z^\Phi_{\{0\}}(\omega) < \infty$, and measurable $C_N \uparrow E$ with
      $\Phi_{\{0\}} \leq c_0(N)$ and $\Phi_{\{0,1\}} \leq c_1(N)$ on configurations with
      $\omega_0, \omega_1 \in C_N$. Then $\rho^\Phi$ is irreducible, with $n(N) = 1$ and
      $h_N = 1_{C_N} e^{-(c_0(N) + 2c_1(N))}/c$. A positive stochastic matrix on a finite $E$
      is an instance.
    \item Let $E$ be countable, $\lambda$ counting measure, $P$ a stochastic matrix, and
      $\rho$ the Markovian modification of Example (10.3) with $p_j = P$, rescaled to the
      probability a priori measure of Remark (1.28)(3). If $P$ is irreducible and aperiodic
      then $\rho$ is irreducible: for all $x, y$, $P^n(x,y) > 0$ for all large $n$, and the
      marginal density is $\rho^0_{]-n,n[}(\omega) = P^n(\omega_{-n},\omega_0)
      P^n(\omega_0,\omega_n)/(r(\omega_0) P^{2n}(\omega_{-n},\omega_n))$.
    \end{enumerate}
\end{example}
\begin{proof}
    \leanok
    (1) $\rho^0_{\{0\}} = e^{-H_{\{0\}}}/Z_{\{0\}}$ and $H_{\{0\}} = \Phi_{\{0\}} +
    \Phi_{\{-1,0\}} + \Phi_{\{0,1\}}$ for a nearest-neighbour potential on $\mathbb Z$.
    (2) The period of a state is the gcd of its return times; supermultiplicativity of the
    diagonal and the additive submonoid generated by a set of gcd $1$ give eventual positivity
    of $P^n(x,x)$, then of $P^n(x,y)$ by irreducibility. Choose $n(N)$ with $P^{n(N)}$ and
    $P^{2n(N)}$ positive on $C_N \times C_N$ and take the infimum of the marginal density.
\end{proof}

\begin{theorem}[Georgii (10.25)--(10.26)]
    \label{thm:mfu-10-25}
    \uses{def:mfu-irreducible, lem:mfc-10-20, def:erg-invariant}
    \lean{MeasureTheory.GibbsMeasure.Markov.exists_ae_eq_pair_of_isIrreducibleInt, MeasureTheory.GibbsMeasure.Markov.exists_isMarkovChain_of_measurePreserving_shift}
    \leanok

    Let $\rho$ be irreducible and $\mu \in \mathcal G(\gamma)$ shift-invariant. Then the
    conditional density $\rho^0_{]-1,\infty[}$ is $\nu_0$-a.s.\ a function of
    $(\sigma_{-1}, \sigma_0)$ (Proposition (10.26)), and hence $\mu$ is a Markov chain with a
    homogeneous transition density (Theorem (10.25)).
\end{theorem}
\begin{proof}
    \leanok
    Two copies of $\mu$ coupled through the tail kernel are compared on the exterior of
    $]-n, n[$; irreducibility bounds the averaged density $\bar\varphi$ away from zero, and the
    estimate (10.28) forces the two-sided conditional densities to agree (Georgii's three steps,
    each a separate lemma). Lemma \ref{lem:mfc-10-20} completes the chain.
\end{proof}

\begin{theorem}[Georgii (10.30)--(10.34)]
    \label{thm:mfu-10-34}
    \uses{thm:mfu-10-25}
    \lean{MeasureTheory.GibbsMeasure.Markov.toReal_densityPow_ae_eq_condExp, MeasureTheory.GibbsMeasure.Markov.tendsto_lintegral_ofReal_abs_densityPow_sub, MeasureTheory.GibbsMeasure.Markov.mul_le_stationaryDensity_of_irreducibility}
    \leanok

    For the chain of Theorem \ref{thm:mfu-10-25} with transition kernel $P$ and invariant
    one-dimensional marginal density $\alpha$: the $n$-step densities $p^{(n)}(x, \cdot)$ form a
    backward martingale (10.30) converging to $\alpha$ in $L^1(\lambda)$ (10.34), and
    $\alpha \geq c\, h_N$ on $C_N$ (10.33), so $\alpha > 0$ $\lambda$-a.e.
\end{theorem}
\begin{proof}
    \leanok
    (10.30)–(10.32) identify $p^{(n)}(\sigma_0, \sigma_n)$ with a conditional expectation of
    $\rho_{\set 0}$ under $\nu_0$; reversed-martingale convergence and the tail triviality of the
    chain give the limit.
\end{proof}

\begin{theorem}[Georgii (10.35)]
    \label{thm:mfu-10-35}
    \uses{thm:mfu-10-34, thm:ext-77-concrete}
    \lean{MeasureTheory.GibbsMeasure.Markov.exists_abs_measureReal_inter_sub_le_of_cylinderEvents, MeasureTheory.GibbsMeasure.Markov.isTailTrivial_of_isMarkovChain, MeasureTheory.GibbsMeasure.Markov.mem_extremePoints_G_of_measurePreserving_shift, MeasureTheory.GibbsMeasure.Markov.eq_of_isGibbsMeasure_of_measurePreserving_shift}
    \leanok

    Let $E$ be standard Borel and $\rho$ irreducible. Then every shift-invariant
    $\mu \in \mathcal G(\gamma)$ is extreme in $\mathcal G(\gamma)$, and $\mathcal G(\gamma)$
    contains at most one shift-invariant measure.
\end{theorem}
\begin{proof}
    \leanok
    (10.36)–(10.37): the mixing bound of Theorem \ref{thm:mfu-10-34} gives
    $|\mu(A \cap B) - \mu(A)\mu(B)| \leq \varepsilon$ for $A$ local and $B$ far away, first for
    cylinder $B$ and then for all measurable $B$ by approximation; hence $\mu$ is tail trivial and
    extreme (Theorem (7.7)). Two distinct shift-invariant Gibbs measures would have an extreme
    midpoint, a contradiction.
\end{proof}
