\section{Preliminaries}

Throughout this section $(X,\MCX)$ is a measurable space and $\MCB$ a sub-$\sigma$-algebra of
$\MCX$.

\begin{definition}[Juxtaposition]
    \label{def:juxtaposition}
    \lean{juxt}
    \leanok{}

    Let $E$ and $S$ be sets, $\Delta \subseteq S$ and $\omega\in E^S$. The
    \textbf{juxtaposition of $\zeta$ and $\omega$}, for $\zeta\in E^\Delta$, is
    \begin{align}
        \juxtBy{\omega} : E^\Delta&\to E^S\\
        \zeta&\mapsto \delta\mapsto\begin{cases}
            \zeta_\delta & \delta\in\Delta\\
            \omega_\delta & \delta\notin\Delta.
        \end{cases}
    \end{align}
\end{definition}

\begin{definition}[Cylinder events]
    \label{def:cylinder-event}
    \lean{MeasureTheory.cylinderEvents}
    \leanok{}

    Let $(E,\mathcal{E})$ be a measurable space and $S$ a set. The $\sigma$-algebra of
    \textbf{cylinder events in $\Delta$}, for $\Delta \subseteq S$, is
    \begin{align}
        \mathcal{F}:\mathcal{P}(S)&\to \set{\sigma\text{-algebras on } E^S}\\
        \Delta&\mapsto \sigma(\{\text{proj}_\delta:E^S\to E\mid \delta\in \Delta\}),
    \end{align}
    where $\text{proj}_\delta$ is the coordinate projection at $\delta$.
\end{definition}

\begin{definition}[Kernel]
    \label{def:kernel}
    \lean{ProbabilityTheory.Kernel}
    \leanok{}

    Let $(X,\MCX)$ and $(Y,\mathcal{Y})$ be measurable spaces. The set of \textbf{kernels from
    $\mathcal{Y}$ to $\MCX$} is
    \[\text{Ker}_{\mathcal{Y},\MCX}:=\left\{\pi:\MCX\times Y\to[0,\infty]~\middle\vert~ \forall y\in Y,\pi(\cdot\mid y)\in\mathfrak{M}(X,\MCX);~\forall A\in\MCX,\pi(A\mid\cdot)\text{ is }\mathcal{Y}\text{-measurable}\right\},\]
    where $\mathfrak{M}(X,\MCX)$ is the set of measures on $(X,\MCX)$.
\end{definition}

\begin{definition}[Markov kernel]
    \label{def:markov-ker}
    \uses{def:kernel}
    \lean{ProbabilityTheory.IsMarkovKernel}
    \leanok{}

    A kernel $\pi\in\text{Ker}_{\mathcal{Y},\MCX}$ is a \textbf{Markov kernel} iff $\pi(X\mid y)=1$
    for all $y \in Y$.
\end{definition}

\begin{definition}[Proper kernel]
    \label{def:proper-ker}
    \uses{def:kernel}
    \lean{ProbabilityTheory.Kernel.IsProper}
    \leanok{}

    A kernel $\pi\in\text{Ker}_{\MCB,\MCX}$ is \textbf{proper} iff
    $\pi(A\cap B\mid x)=\pi(A\mid x)\,\mathbf{1}_B(x)$ for all $A\in\MCX$, $B\in\MCB$ and $x\in X$.
\end{definition}

\begin{lemma}[Lebesgue integral characterisation of proper kernels]
    \label{lem:proper-ker-lintegral}
    \uses{def:proper-ker}
    \lean{ProbabilityTheory.Kernel.IsProper.lintegral_mul}
    \leanok{}

    If $\pi\in\text{Ker}_{\MCB,\MCX}$ is proper, then
    \[\int f(x) g(x)\ \pi(dx\mid x_0) = g(x_0)\int f(x)\ \pi(dx\mid x_0)\]
    for all $x_0 \in X$, all $\MCX$-measurable $f : X \to [0, \infty]$ and all $\MCB$-measurable
    $g : X \to [0,\infty]$.
\end{lemma}
\begin{proof}
    % \uses{}
    \leanok{}

    For $g = \mathbf 1_B$, $B \in \MCB$, this is the definition; the general case follows by
    linearity and monotone convergence.
\end{proof}

\begin{lemma}[Integral characterisation of proper kernels]
    \label{lem:proper-ker-integral}
    \uses{def:proper-ker}
    \lean{ProbabilityTheory.Kernel.IsProper.integral_bdd_mul}
    \leanok{}

    If $\pi\in\text{Ker}_{\MCB,\MCX}$ is a proper finite kernel, then
    \[\int f(x) g(x)\ \pi(dx\mid x_0) = g(x_0)\int f(x)\ \pi(dx\mid x_0)\]
    for all $x_0 \in X$, all $f : X \to \R$ integrable with respect to $\pi(\cdot\mid x_0)$ and
    all bounded $\MCB$-measurable $g : X \to \R$.
\end{lemma}
\begin{proof}
    % \uses{}
    \leanok

    For indicators $f = \mathbf 1_A$, $A \in \MCX$, this is the definition, extended to bounded
    $g$ by approximation. Both sides are continuous in $f \in L^1(\pi(\cdot\mid x_0))$, so the
    identity extends from simple functions to integrable $f$.
\end{proof}

\begin{definition}[Conditional expectation kernel]
    \label{def:cond-exp-ker}
    \uses{def:kernel}
    \lean{ProbabilityTheory.condExpKernel, ProbabilityTheory.Kernel.IsCondExp}
    \leanok{}

    Let $\mu\in\mathfrak{M}(X,\MCX)$. A kernel $\pi\in\text{Ker}_{\MCB,\MCX}$ is a
    \textbf{conditional expectation kernel for $\mu$} iff $\mu(A\mid \MCB)=\pi(A\mid\cdot)$
    $\mu$-a.e.\ for every $A \in \MCX$.
\end{definition}

\begin{lemma}[Lebesgue integral characterisation of proper conditional expectation kernels]
    \label{lem:cond-exp-proper-ker-lintegral}
    \uses{def:cond-exp-ker, def:proper-ker}
    % \lean{}
    % \leanok

    If $\pi \in \text{Ker}_{\MCB, \MCX}$ is a conditional expectation kernel for $\mu$, then
    \[\mu(f\mid \MCB) = \int f(x)\ \pi(dx\mid\cdot)\quad \mu\text{-a.e.}\]
    for all $\MCX$-measurable $f : X \to [0, \infty]$.
\end{lemma}
\begin{proof}
    \uses{lem:lebesgue-cond-exp-char}
    % \leanok

    Monotone convergence from the indicator case of Definition \ref{def:cond-exp-ker}.
\end{proof}

\begin{lemma}[Integral characterisation of proper conditional expectation kernels]
    \label{lem:cond-exp-proper-ker-integral}
    \uses{def:cond-exp-ker, def:proper-ker}
    \lean{ProbabilityTheory.Kernel.condExp_ae_eq_integral}
    \leanok

    Let $\mu$ be a finite measure and $\pi \in \text{Ker}_{\MCB, \MCX}$ a proper Markov kernel
    which is a conditional expectation kernel for $\mu$. Then
    \[\mu(f\mid \MCB) = \int f(x)\ \pi(dx\mid\cdot)\quad \mu\text{-a.e.}\]
    for all $\mu$-integrable $f : X \to \R$.
\end{lemma}
\begin{proof}
    \uses{lem:lebesgue-cond-exp-char}
    % \leanok

    By Lemma \ref{lem:cond-exp-proper-ker-bind}, $\mu\pi = \mu$, so
    $f \mapsto \int f(x)\,\pi(dx\mid\cdot)$ is a contraction of $L^1(\mu)$. The identity holds for
    indicators by Definition \ref{def:cond-exp-ker}, hence for simple functions, hence on
    $L^1(\mu)$ by continuity of both sides.
\end{proof}

\begin{lemma}[Characterisation of proper conditional expectation kernels, Remark 1.20]
    \label{lem:cond-exp-proper-ker-bind}
    \uses{def:cond-exp-ker}
    \lean{ProbabilityTheory.Kernel.isCondExp_iff_bind_eq_left}
    \leanok

    Let $\mu\in\mathfrak{M}(X,\MCX)$ be a finite measure and $\pi\in\text{Ker}_{\MCB,\MCX}$ a
    proper Markov kernel. Then $\pi$ is a conditional expectation kernel for $\mu$ iff
    $\mu\pi = \mu$.
\end{lemma}
\begin{proof}
    \uses{lem:lebesgue-cond-exp-char}
    \leanok

    By the characterisation of conditional expectation, $\pi$ is a conditional expectation
    kernel for $\mu$ iff $\mu(A \inter B) = \int_B \pi(A\mid\cdot)\ d\mu$ for all $A \in \MCX$,
    $B \in \MCB$. By properness, $\int_B \pi(A\mid\cdot)\ d\mu = \mu\pi(A \inter B)$. Hence
    \begin{align}
        \pi \text{ is a conditional expectation kernel for }\mu
        & \iff \forall A \in \MCX, \forall B \in \MCB, \mu(A \inter B) = \mu\pi(A \inter B) \\
        & \iff \forall A \in \MCX, \mu(A) = \mu\pi(A) \\
        & \iff \mu = \mu\pi.
    \end{align}
\end{proof}
