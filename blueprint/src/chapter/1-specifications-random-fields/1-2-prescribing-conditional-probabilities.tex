\section{Prescribing conditional probabilities}

Throughout, $S$ is the parameter set and $(E, \mathcal{E})$ the state space.

\begin{definition}[Specification]
    \label{def:spec}
    \uses{def:cylinder-event, def:kernel}
    \lean{Specification}
    \leanok{}

    A {\bf specification} (Georgii (1.23)) is a family of kernels
    $\gamma : \Finset S \to \Ker_{\mathcal{F}_{S\setminus\Lambda}, \mathcal{E}^S}$, each
    $\gamma_\Lambda$ a {\bf probability} kernel and {\bf proper}, which is {\bf consistent}:
    \[\gamma_{\Lambda_1} \circ_k \gamma_{\Lambda_2} = \gamma_{\Lambda_2}
      \quad \text{whenever } \Lambda_1 \subseteq \Lambda_2 \in \Finset(S).\]
    A consistent family is a {\bf pre-specification}.
\end{definition}

\begin{definition}[Independent specification]
    \label{def:indep-spec}
    \uses{def:spec}
    \lean{IsStronglyConsistent}
    \leanok{}

    A family of kernels $\gamma : \Finset S \to \Ker_{\mathcal{F}_{S\setminus\Lambda}, \mathcal{E}^S}$
    (in particular a specification) is \textbf{independent} if
    \[\gamma_{\Lambda_1} \circ_k \gamma_{\Lambda_2} = \gamma_{\Lambda_1\cup\Lambda_2}
      \quad (\Lambda_1, \Lambda_2 \in \Finset(S)).\]
\end{definition}

\begin{remark}[The probability and properness conditions]
    \label{def:markov-spec}
    \lean{Specification.isMarkovKernel', Specification.IsProper, Specification.isProper}
    \leanok{}
    \uses{def:spec, def:proper-ker}

    Every specification consists of probability kernels and is proper (Definition
    \ref{def:spec}); on a pre-specification, where only consistency is assumed, each is a
    separate hypothesis.
\end{remark}

\begin{definition}[Gibbs measures]
    \label{def:gibbs-meas}
    \uses{def:spec, def:cond-exp-ker}
    \lean{Specification.IsGibbsMeasure}
    \leanok

    Let $\gamma$ be a specification. A measure $\nu \in \mathfrak{M}(E^S, \mathcal{E}^S)$ is a
    \textbf{Gibbs measure specified by $\gamma$} if $\gamma_\Lambda(A \mid \cdot)$ is a
    conditional expectation kernel for $\nu$ for all $A \in \mathcal E^S$ and
    $\Lambda \in \Finset(S)$.
\end{definition}

\begin{lemma}[Characterisation of Gibbs measures, Remark 1.24]
    \label{lem:gibbs-meas-tfae}
    \uses{def:gibbs-meas, def:markov-spec}
    \lean{Specification.isGibbsMeasure_iff_forall_bind_eq, Specification.isGibbsMeasure_iff_frequently_bind_eq}
    \leanok

    Let $\gamma$ be a specification and $\nu$ a finite measure on $(E^S, \mathcal{E}^S)$. TFAE:
    \begin{enumerate}
        \item $\nu$ is a Gibbs measure specified by $\gamma$.
        \item $\gamma_\Lambda\circ_m\nu = \nu$ for all $\Lambda\in\Finset(S)$.
        \item $\gamma_\Lambda\circ_m\nu = \nu$ frequently as $\Lambda \to S$.
    \end{enumerate}
\end{lemma}
\begin{proof}
    \uses{lem:cond-exp-proper-ker-lintegral}
    \leanok

    (1)$\Leftrightarrow$(2) is Lemma \ref{lem:cond-exp-proper-ker-lintegral}, and
    (2)$\Rightarrow$(3) is trivial. (3)$\Rightarrow$(2): given $\Lambda$, choose
    $\Lambda' \supseteq \Lambda$ with $\gamma_{\Lambda'}\circ_m\nu = \nu$; by consistency
    \[\gamma_\Lambda\circ_m\nu = \gamma_\Lambda\circ_m(\gamma_{\Lambda'}\circ_m\nu)
      = \gamma_{\Lambda'}\circ_m\nu = \nu.\]
\end{proof}
